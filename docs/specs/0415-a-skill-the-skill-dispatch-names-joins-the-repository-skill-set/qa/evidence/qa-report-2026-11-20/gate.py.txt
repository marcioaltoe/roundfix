def run(name,args,cwd=ROOT,env=None,expect=None):
    r=subprocess.run(args,cwd=cwd,env=env or ENV,stdin=subprocess.DEVNULL,capture_output=True,text=True)
    out='$ '+shlex.join(args)+'\nworking directory: '+str(cwd)+'\nstdout:\n'+r.stdout+'\nstderr:\n'+r.stderr+'\nexit: '+str(r.returncode)+'\n'
    (E/(name+'.txt')).write_text(out)
    print(name,'exit',r.returncode,flush=True)
    if expect is not None and r.returncode!=expect: raise AssertionError(name+': unexpected exit '+str(r.returncode)+'; see transcript')
    return r

def observe(row,status,message,files):
    text=REPORT.read_text()
    text,n=re.subn(r'^\| '+re.escape(row)+r' \| [^|]* \|',lambda m:'| '+row+' | '+status+' |',text,count=1,flags=re.M)
    assert n==1,row
    start=text.index('\n### '+row+'\n'); end=text.find('\n### ',start+1)
    if end<0: end=len(text)
    block=text[start:end]
    links=', '.join('['+f+'](evidence/qa-report-2026-11-20/'+f+')' for f in files)
    block=block.replace('Observation: pending.','Observation: '+message+' Evidence: '+links+'.')
    REPORT.write_text(text[:start]+block+text[end:])
    print(row,status,message,flush=True)

def test(name,pattern,package,verbose=True,tags=None,cwd=ROOT):
    args=['go','test','-count=1','-timeout','20m']
    if verbose: args+=['-v']
    if tags: args+=['-tags',tags]
    if pattern: args+=['-run',pattern]
    args+=[package]
    return run(name,args,cwd=cwd)

def passed(r,names):

    for name in names: assert re.search(r'--- PASS: '+re.escape(name)+r'\s',r.stdout), 'missing pass '+name
    assert 'no tests to run' not in r.stdout

def tree_digest(d):
    rows=[]
    for p in sorted(d.rglob('*')):
        if '.git' in p.relative_to(d).parts: continue
        if p.is_symlink(): rows.append((str(p.relative_to(d)),'symlink',os.readlink(p)))
        elif p.is_file(): rows.append((str(p.relative_to(d)),p.stat().st_mode & 0o111,hashlib.sha256(p.read_bytes()).hexdigest()))
    return rows

def diagnose_regenerate():
    with archived() as d:
        before={r[0]:r[1:] for r in tree_digest(d)}
        with product_env() as env:
            run('regenerate-diagnosis',['./bin/roundfix','baseline','update','--repo',str(d),'--no-skills','--no-history','--yes','--format','text'],ROOT,env,0)
        after={r[0]:r[1:] for r in tree_digest(d)}
        delta=[p for p in sorted(set(before)|set(after)) if before.get(p)!=after.get(p)]
        status=git(['status','--porcelain=v1','--untracked-files=all'],d)
        text='Files differing when ignored operational files are also counted:\n'+'\n'.join(delta)+'\n\nGit repository-content status:\n'+status
        for p in delta:
            r=subprocess.run(['git','check-ignore','--',p],cwd=d,env=ENV,capture_output=True,text=True)
            text+='\n'+p+': '+('ignored' if r.returncode==0 else 'NOT IGNORED')
        (E/'regenerate-diagnosis.txt').write_text((E/'regenerate-diagnosis.txt').read_text()+'\n'+text+'\n')
        comparisons=[{'path':p,'before_mode':oct(before[p][0]),'after_mode':oct(after[p][0]),'content_unchanged':before[p][1:]==after[p][1:]} for p in delta]
        with (E/'regenerate-diagnosis.txt').open('a') as f: f.write(json.dumps(comparisons,indent=2)+'\n')
        print(text); print(json.dumps(comparisons,indent=2))
        assert not status.strip() and all(row['content_unchanged'] for row in comparisons)
