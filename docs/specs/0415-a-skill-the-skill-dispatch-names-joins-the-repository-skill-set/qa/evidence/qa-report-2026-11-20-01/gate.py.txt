def run(name,args,cwd=ROOT,env=None,expected=None):
    p=subprocess.run(args,cwd=cwd,env=env or ENV,stdin=subprocess.DEVNULL,stdout=subprocess.PIPE,stderr=subprocess.PIPE,text=True)
    output='command: '+shlex.join([str(a) for a in args])+'\ncwd: '+str(cwd)+'\nexit: '+str(p.returncode)+'\nstdout:\n'+p.stdout+'\nstderr:\n'+p.stderr
    (E/(name+'.txt')).write_text(output)
    print(name+': exit '+str(p.returncode),flush=True)
    if expected is not None and p.returncode!=expected: raise AssertionError(name+' unexpected exit; see '+str(E/(name+'.txt')))
    return p
class Product:
    def __enter__(self):
        self.temp=pathlib.Path(tempfile.mkdtemp(prefix='qa0415-'))
        self.env=ENV.copy()
        for var in ['GOMODCACHE','GOCACHE']:
            self.env[var]=subprocess.check_output(['go','env',var],env=ENV,text=True).strip()
        self.env['RUSTUP_HOME']=os.environ.get('RUSTUP_HOME',str(pathlib.Path.home()/'.rustup'))
        self.env['CARGO_HOME']=os.environ.get('CARGO_HOME',str(pathlib.Path.home()/'.cargo'))
        for var,part in [('HOME','home'),('TMPDIR','tmp')]:
            (self.temp/part).mkdir(); self.env[var]=str(self.temp/part)
        (self.temp/'stub').mkdir()
        gh=self.temp/'stub/gh'; gh.write_text('#!/bin/sh\nexit 1\n'); gh.chmod(0o755)
        self.env['PATH']=str(gh.parent)+os.pathsep+ENV['PATH']
        self.env['GIT_CONFIG_NOSYSTEM']='1'; self.env['GIT_CONFIG_GLOBAL']=os.devnull
        return self
    def __exit__(self,*exc):
        subprocess.run(['chmod','-R','u+w',str(self.temp)],stdout=subprocess.DEVNULL,stderr=subprocess.DEVNULL)
        shutil.rmtree(self.temp)
    def cli(self,name,args,cwd=ROOT,expected=None):
        # A repo operand reaches fixtures outside the Run; executable is always Run-built.
        return run(name,[str(ROOT/'bin/roundfix')]+args,cwd,self.env,expected)
