import os, pathlib, subprocess, tempfile, json, sys
ROOT=pathlib.Path.cwd().resolve()
STATE=pathlib.Path('/private/tmp/qa0293-state.json')
if STATE.exists():
 T=pathlib.Path(json.loads(STATE.read_text())['tmp'])
else:
 T=pathlib.Path(tempfile.mkdtemp(prefix='rf293-',dir='/private/tmp')).resolve(); STATE.write_text(json.dumps({'tmp':str(T)}))
E=os.environ.copy()
for k in list(E):
 if any(s in k for s in ('API_KEY','TOKEN','SECRET','PASSWORD')) or k.startswith(('GIT_','ROUNDFIX_')): E.pop(k,None)
for d in ['home','tmp','xdg']: (T/d).mkdir(exist_ok=True)
E.update(HOME=str(T/'home'),GIT_CONFIG_GLOBAL=str(T/'gitconfig'),GIT_CONFIG_NOSYSTEM='1',GIT_TERMINAL_PROMPT='0',TMPDIR=str(T/'tmp'),XDG_CONFIG_HOME=str(T/'xdg'),GOPROXY='off',GOSUMDB='off',GOTOOLCHAIN='local')
(T/'gitconfig').touch(exist_ok=True)
# Reuse installed Go modules offline without reading the host Roundfix store.
E['GOMODCACHE']=subprocess.check_output(['go','env','GOMODCACHE'],text=True).strip()
def run(args,cwd=ROOT,log=None,extra=None):
 e=E.copy(); e.update(extra or {}); e['PWD']=str(cwd)
 p=subprocess.run(args,cwd=cwd,env=e,text=True,stdout=subprocess.PIPE,stderr=subprocess.PIPE)
 out='$ '+ ' '.join(map(str,args))+'\nexit: '+str(p.returncode)+'\nstdout:\n'+p.stdout+'stderr:\n'+p.stderr
 if log: pathlib.Path(log).write_text(out)
 return p,out
if __name__=='__main__':
 p,out=run(sys.argv[1:]); print(out); sys.exit(p.returncode)
