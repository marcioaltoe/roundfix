def norm(s): return ' '.join(s.split())
def git(*args,where=ROOT): return subprocess.check_output(['git',*args],cwd=where,env=ENV,text=True).strip()
def show(rev,path): return subprocess.check_output(['git','show',f'{rev}:{path}'],cwd=ROOT,env=ENV)
COMMAND_LOGS=[]
def commands(args,where=ROOT,env=None,expect=0):
    print('RUN', ' '.join(map(str,args)),flush=True)
    p=subprocess.run(list(map(str,args)),cwd=where,env=env or ENV,stdout=subprocess.PIPE,stderr=subprocess.PIPE,text=True)
    log='command: '+' '.join(map(str,args))+'\nworking directory: '+str(where)+'\nSTDOUT:\n'+p.stdout+'\nSTDERR:\n'+p.stderr+'\nexit: '+str(p.returncode)+'\n'
    COMMAND_LOGS.append(log)
    if expect is not None and p.returncode!=expect: raise RuntimeError(log)
    return p,log

        commands(['git','add','.'],d)

def record(row,status,observation,log):
    path=EV/(row.lower()+'.txt'); path.write_text(log)
    s=REPORT.read_text(); s=re.sub(r'(?m)^\| '+row+r' \| [^|]* \|', '| '+row+' | '+status+' |',s)
    start=s.index('### '+row+'\n'); end=s.find('\n### ',start+1)
    if end<0: end=len(s)
    block=s[start:end].replace('Observation: pending.',observation+'\n\nEvidence: ['+row+' transcript](evidence/qa-report-2026-11-14/'+path.name+').')
    s=s[:start]+block+s[end:]; REPORT.write_text(s)
    print(row,status,observation,flush=True)

def perform(row,fn):
    COMMAND_LOGS.clear()
    try:
        result=fn()
        obs,log=result
        record(row,'pass',obs,log)
    except Exception as ex:
        record(row,'fail','Expected observable was not met; see transcript.','\n'.join(COMMAND_LOGS)+'\n'+traceback.format_exc())
