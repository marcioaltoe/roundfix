func TestDeliverRetryReproducesSurfaceTranscriptAfterAQAOnlyFailedRun(t *testing.T) {
	t.Parallel()
	f := newRetryOwnerProofFixture(t, true, "live")
	code, out, diagnostic := f.retry(t)
 t.Logf("Observed exit=%d stdout=%q stderr=%q", code, out, diagnostic)
	want := "Carried forward from Run " + f.runID + ": task_01, task_02, task_03, task_04, task_05\nRetried " + implementTestSlug + ": run-unresolved -> running\n" + f.report
	if code != exitOK || out != want || diagnostic != "" {
		t.Fatalf("exit=%d stdout=%q stderr=%q want=%q", code, out, diagnostic, want)
	}
	f.assertCarried(t)
}

func TestDeliverRetryReproducesSurfaceTranscriptOwnerDidNotStart(t *testing.T) {
	t.Parallel()
	f := newRetryOwnerProofFixture(t, false, "failed")
	code, out, diagnostic := f.retry(t)
 t.Logf("Observed exit=%d stdout=%q stderr=%q", code, out, diagnostic)
	want := "owner start diagnostic\nRetry failed\n\nReason:\n  the Queue Owner did not start\n\nItem:\n  stage: parked; blocker: run-unresolved\n\nNext:\n  roundfix deliver retry " + implementTestSlug + "\n"
	if code != exitPreflight || out != "" || diagnostic != want {
		t.Fatalf("exit=%d stdout=%q stderr=%q want=%q", code, out, diagnostic, want)
	}
	f.assertReparked(t)
}

func TestDeliverRetryReproducesSurfaceTranscriptOwnerEndedBeforeTheProof(t *testing.T) {
	t.Parallel()
	f := newRetryOwnerProofFixture(t, true, "gone")
	code, out, diagnostic := f.retry(t)
 t.Logf("Observed exit=%d stdout=%q stderr=%q", code, out, diagnostic)
	want := fmt.Sprintf("Retry failed\n\nReason:\n  Queue Owner PID %d is not running\n\nItem:\n  stage: parked; blocker: run-unresolved\n\nCarried forward:\n  Run %s: task_01, task_02, task_03, task_04, task_05\n\nNext:\n  roundfix deliver retry %s\n", 1<<30, f.runID, implementTestSlug)
	if code != exitPreflight || out != "" || diagnostic != want {
		t.Fatalf("exit=%d stdout=%q stderr=%q want=%q", code, out, diagnostic, want)
	}
	f.assertReparked(t)
	f.assertCarried(t)
}

func TestDeliverRetryProvesTheQueueOwnerItStarted(t *testing.T) {
	t.Parallel()
	for _, name := range []string{"different_identity_parks_the_item_again", "unreadable_identity_leaves_the_item_and_the_record", "owner_that_already_parked_the_item_exits_zero", "no_recorded_owner_parks_the_item_again"} {
		t.Run(name, func(t *testing.T) {
			t.Parallel()
			f := newRetryOwnerProofFixture(t, false, name)
			code, out, diagnostic := f.retry(t)
 t.Logf("Observed exit=%d stdout=%q stderr=%q", code, out, diagnostic)
			q := openDeliveryQueueForCLI(t, f.home, f.repo)
			switch name {
			case "owner_that_already_parked_the_item_exits_zero":
				if code != exitOK || diagnostic != "" || out != "Retried "+implementTestSlug+": run-unresolved -> running\n"+f.report || q.Items[0].Blocker != "new-owner-blocker" {
					t.Fatalf("exit=%d stdout=%q stderr=%q queue=%+v", code, out, diagnostic, q)
				}
			case "unreadable_identity_leaves_the_item_and_the_record":
				if code != exitPreflight || out != "" || q.OwnerPID != os.Getpid() || q.Items[0].Stage != store.DeliveryStageRunning || q.Items[0].RetryCount != f.before.RetryCount+1 || !strings.Contains(diagnostic, "could not be checked: unreadable identity") || !strings.Contains(diagnostic, "stage: running; blocker: -") || !strings.HasSuffix(diagnostic, "roundfix deliver status\n") {
					t.Fatalf("exit=%d stdout=%q stderr=%q queue=%+v", code, out, diagnostic, q)
				}
			default:
				f.assertReparked(t)
				reason := "no Queue Owner holds the Delivery Queue after the start"
				if name == "different_identity_parks_the_item_again" {
					reason = fmt.Sprintf("Queue Owner PID %d has a different process identity", os.Getpid())
				}
				want := "Retry failed\n\nReason:\n  " + reason + "\n\nItem:\n  stage: parked; blocker: run-unresolved\n\nNext:\n  roundfix deliver retry " + implementTestSlug + "\n"
				if code != exitPreflight || out != "" || diagnostic != want {
					t.Fatalf("exit=%d stdout=%q stderr=%q", code, out, diagnostic)
				}
			}
		})
	}
}

type retryOwnerProofFixture struct {
	home, repo, runID, report string
	before                    store.DeliveryQueueItem
}

type storingRetryOwnerProofEngine struct {
	*retryCommandDeliveryEngine
	db       *store.Store
	workflow *commandDeliveryWorkflow
	before   store.DeliveryQueueItem
}

func (e *storingRetryOwnerProofEngine) Retry(ctx context.Context, root, slug string) (delivery.RetryResult, error) {
	carried := delivery.CarryForwardResult{}
	var err error
	if e.before.RunID != "" {
		carried, err = e.workflow.CarryForward(ctx, root, slug, e.before.Branch, e.before.RunID)
		if err != nil {
			return delivery.RetryResult{}, err
		}
	}
	item := e.before
	item.Stage, item.Blocker = store.DeliveryStageRunning, ""
	if carried.RunID != "" {
		item.RunID = carried.RunID
	}
	item.CandidateCommits = []string{"retried-candidate"}
	pid, identity, err := e.db.RetryDeliveryQueueItem(ctx, root, item, e.before.Blocker)
	return delivery.RetryResult{SpecSlug: slug, Blocker: e.before.Blocker, Stage: item.Stage, OwnerPID: pid, OwnerIdentity: identity, CarriedFrom: carried}, err
}

func newRetryOwnerProofFixture(t *testing.T, carry bool, mode string) *retryOwnerProofFixture {
	t.Helper()
	seeds := []implementSeed{}
	ids := []string{}
	for i := 1; i <= 5; i++ {
		id := fmt.Sprintf("task_%02d", i)
		ids = append(ids, id)
		seeds = append(seeds, implementSeed{id: id})
	}
	home, repo := newImplementWorkspace(t, append(append([]implementSeed{}, seeds...), implementQAGateSeed("pending", ids...)))
	workflow := newItemRecoveryWorkflowForRepository(t, home, repo)
	f := &retryOwnerProofFixture{home: home, repo: repo}
	if carry {
		f.runID = createDeliveryRetryRun(t, workflow, repo, "ma/widget-flow", seeds, store.StateFailed).run.ID
	}
	q, err := workflow.store.CreateDeliveryQueue(t.Context(), repo, []string{implementTestSlug})
	if err != nil {
		t.Fatal(err)
	}
	if _, _, _, err := workflow.store.RecordDeliveryQueueItemWorktree(t.Context(), repo, implementTestSlug, "ma/widget-flow", repo); err != nil {
		t.Fatal(err)
	}
	f.before = q.Items[0]
	f.before.Stage, f.before.Blocker = store.DeliveryStageParked, "run-unresolved"
	f.before.Branch, f.before.Worktree, f.before.RunID = "ma/widget-flow", repo, f.runID
	f.before.CandidateCommits = []string{"original-candidate"}
	if err := workflow.store.UpdateDeliveryQueueItem(t.Context(), repo, f.before); err != nil {
		t.Fatal(err)
	}
	f.report = "Delivery Owner: test-owner\nConsole Log: " + filepath.Join(home, "owner.log") + "\nStatus: roundfix deliver status\nStop: roundfix deliver stop\n"
	processes := &fakeDeliveryOwnerProcesses{}
	if mode == "different_identity_parks_the_item_again" {
		processes.proveErr = store.ErrOwnerProcessIdentityUnproven
	}
	if mode == "unreadable_identity_leaves_the_item_and_the_record" {
		processes.proveErr = errors.New("unreadable identity")
	}
	updateCommandDependenciesForTest(t, func(d *commandDependencies) {
		d.ownerProcesses = processes
		d.newDeliveryEngine = func(db *store.Store, _ roundconfig.Loaded) deliveryEngine {
			copyWorkflow := *workflow
			copyWorkflow.store = db
			return &storingRetryOwnerProofEngine{retryCommandDeliveryEngine: &retryCommandDeliveryEngine{}, db: db, workflow: &copyWorkflow, before: f.before}
		}
		d.startDeliveryOwner = func(ctx context.Context, _ roundconfig.Loaded, _ commandEnvironment, out, diagnostic io.Writer) int {
			if mode == "failed" {
				fmt.Fprintln(out, "discarded partial report")
				fmt.Fprintln(diagnostic, "owner start diagnostic")
				return exitPreflight
			}
			if mode != "no_recorded_owner_parks_the_item_again" {
				pid := os.Getpid()
				identity, err := store.OwnerProcessIdentity(ctx, pid)
				if err != nil {
					t.Fatal(err)
				}
				if mode == "gone" {
					pid = 1 << 30
				}
				if err := workflow.store.ClaimDeliveryQueueOwner(ctx, repo, pid, identity); err != nil {
					t.Fatal(err)
				}
				if mode == "owner_that_already_parked_the_item_exits_zero" {
					q, _, err := workflow.store.DeliveryQueue(ctx, repo)
					if err != nil {
						t.Fatal(err)
					}
					item := q.Items[0]
					item.Stage, item.Blocker = store.DeliveryStageParked, "new-owner-blocker"
					if err := workflow.store.UpdateDeliveryQueueItem(ctx, repo, item); err != nil {
						t.Fatal(err)
					}
					if released, err := workflow.store.ReleaseIdleDeliveryQueueOwner(ctx, repo, pid, identity); err != nil || !released {
						t.Fatalf("idle release=%v error=%v", released, err)
					}
				}
			}
			fmt.Fprint(out, f.report)
			return exitOK
		}
	})
	return f
}

func (f *retryOwnerProofFixture) retry(t *testing.T) (int, string, string) {
	t.Helper()
	var out, diagnostic bytes.Buffer
	code := runCLI(t, []string{"deliver", "retry", implementTestSlug}, &out, &diagnostic)
	return code, out.String(), diagnostic.String()
}

func (f *retryOwnerProofFixture) assertReparked(t *testing.T) {
	t.Helper()
	q := openDeliveryQueueForCLI(t, f.home, f.repo)
	item := q.Items[0]
	if q.OwnerPID != 0 || q.OwnerIdentity != "" || item.Stage != store.DeliveryStageParked || item.Blocker != f.before.Blocker || item.RetryCount != f.before.RetryCount || strings.Join(item.CandidateCommits, ",") != strings.Join(f.before.CandidateCommits, ",") || item.RunID != f.before.RunID || item.Branch != f.before.Branch || item.Worktree != f.before.Worktree {
		t.Fatalf("queue after failed retry=%+v", q)
	}
}

func (f *retryOwnerProofFixture) assertCarried(t *testing.T) {
	t.Helper()
	for i := 1; i <= 5; i++ {
		id := fmt.Sprintf("task_%02d", i)
		content := mustRead(t, implementTaskPath(f.repo, id))
		if !strings.Contains(content, "status: completed") || !strings.Contains(content, f.runID) || mustRead(t, filepath.Join(f.repo, "src", id+".txt")) != id+" settled\n" {
			t.Fatalf("missing carried Task %s", id)
		}
	}
}
