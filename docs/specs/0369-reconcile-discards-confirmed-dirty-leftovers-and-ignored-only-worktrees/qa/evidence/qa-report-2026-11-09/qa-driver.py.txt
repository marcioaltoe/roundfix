def run(args, cwd=ROOT, environ=None):
    p = subprocess.run(args, cwd=cwd, env=environ, capture_output=True)
    return p, 'command: ' + shlex.join(map(str,args)) + '\nexit: ' + str(p.returncode) + '\nstdout:\n' + p.stdout.decode(errors='replace') + '\nstderr:\n' + p.stderr.decode(errors='replace') + '\n'

def save(name, text):
    (EVIDENCE / (name + '.txt')).write_text(text)

def row(n, status, detail, logs=(), commits=False):
    s = REPORT.read_text()
    s = re.sub(r'(\| Q%02d \| )[^|]+(\|)' % n, lambda m:m[1]+status+' '+m[2], s)
    s += '\n### Q%02d\n\n```yaml\ninputs:\n  - kind: repository_path\n    ref: internal/cli/reconcile.go\n' % n
    if commits:
        s += '  - kind: commit_range\n    ref: ' + TASK_RANGE + '\n'
    s += '```\n\n' + detail + '\n\n' + ' '.join('[%s](evidence/qa-report-2026-11-09/%s.txt)' % (x,x) for x in logs) + '\n'
    REPORT.write_text(s)
    print('Q%02d: %s' % (n,status), flush=True)

def cli_env(home, tmp):
    e = os.environ.copy()
    e.update(HOME=str(home), TMPDIR=str(tmp))
    for name in ('ROUNDFIX_OPENROUTER_API_KEY','ROUNDFIX_OPENROUTER_JUDGE_API_KEY','ROUNDFIX_TYPESAFE_API_KEY','TYPESAFE_API_KEY'):
        e.pop(name,None)
    for name in ('GOMODCACHE','GOCACHE'):
        e[name] = subprocess.check_output(['go','env',name], text=True).strip()
    return e

def archive(dest, revision=HEAD):
    p = subprocess.run(['git','archive',revision],capture_output=True,check=True)
    with tarfile.open(fileobj=io.BytesIO(p.stdout)) as t:
        t.extractall(dest, filter='data')
