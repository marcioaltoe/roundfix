package main

import (
 "context"
 "encoding/json"
 "os"
 "os/exec"
 "strings"
 "roundfix/internal/store"
 "roundfix/internal/worktree"
)

func main() {
 ctx:=context.Background()
 home,repo,location,slug:=os.Args[1],os.Args[2],os.Args[3],os.Args[4]
 cmd:=exec.Command("git","rev-parse","HEAD");cmd.Dir=repo
 b,err:=cmd.Output();must(err)
 db,err:=store.Open(ctx,home);must(err)
 run,err:=db.CreateRun(ctx,store.CreateRunRequest{Kind:store.KindImplement,GitRoot:repo,LocalBranch:"delivery/"+slug,HeadSHA:strings.TrimSpace(string(b)),SpecSlug:slug,Agent:"codex"});must(err)
 ref,err:=worktree.Create(ctx,worktree.CreateOptions{UserRoot:repo,Location:location,RunID:run.ID,HeadSHA:run.HeadSHA});must(err)
 _,err=db.SetRunWorkDir(ctx,run.ID,ref.Path);must(err)
 _,err=db.CompleteRun(ctx,run.ID,store.StateStopped);must(err)
 must(db.Close())
 must(json.NewEncoder(os.Stdout).Encode(map[string]string{"id":run.ID,"path":ref.Path,"branch":ref.Branch}))
}

func must(err error) { if err!=nil { panic(err) } }
