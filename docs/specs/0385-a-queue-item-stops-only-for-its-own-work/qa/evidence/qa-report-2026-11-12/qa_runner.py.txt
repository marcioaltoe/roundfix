def integrity():
    s=REPORT.read_text(); lines=[]; errors=[]
    def check(ok,label):
        lines.append(("PASS " if ok else "FAIL ")+label)
        if not ok: errors.append(label)
    check(s.startswith("---\n") and s.count("\n---\n")==1,"one frontmatter block")
    table=s.split("## Results\n",1)[1].lstrip("\n").split("\n\n",1)[0]
    check(len(re.findall(r"(?m)^\| [A-Z][0-9]{2} \|",table))==25,"25 rows contiguous inside Results table")
    check("| pending |" not in s,"no pending rows")
    check("rows_blocked_environment: 1" in s and "rows_blocked_declared: 1" in s and "rows_blocked_finding: 0" in s,"exact typed blocked counts")
    check("evidence_snapshots" not in s,"Daemon snapshots not authored")
    for target in re.findall(r"\]\(([^)]+)\)",s):
        if "://" in target: continue
        p=(REPORT.parent/target).resolve(); check(p.exists(),"link exists: "+target)
        r=call(["git","check-ignore",str(p)]); check(r.returncode==1,"link not ignored: "+target)
    for rid,_,_ in ROWS:
        section=s.split("### "+rid+"\n",1)[1].split("\n### ",1)[0]
        check("```yaml\ninputs:\n  - kind: repository_path" in section,"typed inputs: "+rid)
    forbidden=[]
    pattern_text=(ROOT/"internal/secretscan/patterns.go").read_text()
    expressions=re.findall(r'\{"([^"\n]+)", regexp\.MustCompile\(`([^`]+)`\)\}',pattern_text)
    for p in sorted((S/"qa").rglob("*")):
        if not p.is_file(): continue
        body=p.read_bytes(); check(b"\0" not in body,"text evidence: "+str(p.relative_to(ROOT)))
        for n,line in enumerate(body.decode().splitlines(),1):
            for kind,pattern in expressions:
                m=re.search(pattern,line)
                if m and (kind!="credential-assignment" or (re.search(r"[A-Za-z]",m.group(1)) and re.search(r"[0-9]",m.group(1)))):
                    forbidden.append(str(p.relative_to(ROOT))+":"+str(n)+": "+kind); break
    check(not forbidden,"no Secret Pattern matches; current repository pattern table")
    lines.extend(forbidden)
    check(not call(["git","diff","--name-only",HEAD]).stdout.strip(),"tracked implementation and Task files unchanged")
    check(call(["git","diff","--check"]).returncode==0,"git diff --check")
    (E/"report-integrity.txt").write_text("\n".join(lines)+"\n")
    print("Report integrity: "+str(len(errors))+" errors; "+str(len(forbidden))+" Secret Matches.")
    return not errors
