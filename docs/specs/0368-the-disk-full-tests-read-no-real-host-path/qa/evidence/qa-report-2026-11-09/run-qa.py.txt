def raw(args, cwd=ROOT, env=None):
    return subprocess.run(args, cwd=cwd, env=env, text=True, stdout=subprocess.PIPE, stderr=subprocess.PIPE)

def clean(text):
    for name in ['HOME', 'TMPDIR', 'GOCACHE', 'GOMODCACHE']:
        value = os.environ.get(name)
        if value:
            text = text.replace(value, '<' + name + '>')
    return text

def save(name, text):
    (EVIDENCE / name).write_text(clean(text))

def run(name, args, cwd=ROOT, env=None, expected=0):
    result = raw(args, cwd, env)
    save(name, '$ ' + shlex.join(args) + '\nstdout:\n' + result.stdout + '\nstderr:\n' + result.stderr + '\nexit: ' + str(result.returncode) + '\n')
    print(name + ': exit ' + str(result.returncode), flush=True)
    if result.returncode != expected:
        raise RuntimeError(name + ': unexpected exit; inspect retained evidence')
    return result

def temporary(label):
    return Path(subprocess.check_output(['mktemp', '-d', str(ROOT / ('.qa-0368-' + label + '.XXXXXX'))], text=True).strip())

def cli(name, args, expected=0):
    home = Path(subprocess.check_output(['mktemp', '-d', '/private/tmp/qa-0368-home.XXXXXX'], text=True).strip())
    tmp = Path(subprocess.check_output(['mktemp', '-d', '/private/tmp/qa-0368-tmp.XXXXXX'], text=True).strip())
    try:
        cli_env = os.environ.copy()
        for key in KEYS:
            cli_env.pop(key, None)
        for key in ['GOMODCACHE', 'GOCACHE']:
            cli_env[key] = subprocess.check_output(['go', 'env', key], text=True).strip()
        cli_env.update(HOME=str(home), TMPDIR=str(tmp))
        return run(name, ['./bin/roundfix'] + args, env=cli_env, expected=expected)
    finally:
        shutil.rmtree(home)
        shutil.rmtree(tmp)

def status(row, value):
    text = REPORT.read_text()
    text = re.sub(r'(?m)^\| ' + row + r' \| [^|]+ \|', '| ' + row + ' | ' + value + ' |', text)
    REPORT.write_text(text)

def snapshot(label, revision):
    path = temporary(label)
    data = subprocess.check_output(['git', 'archive', revision], cwd=ROOT)
    with tarfile.open(fileobj=io.BytesIO(data)) as archive:
        archive.extractall(path, filter='data')
    return path

def gotest(name, selector, package='./internal/cli', cwd=ROOT, expected=0):
    # The package test executable is explicitly emitted inside this Run Worktree.
    binary = cwd / '.qa-cli.test'
    args = ['go', 'test', '-count=1', '-timeout', '20m', '-v', '-o', str(binary), '-run', selector, package]
    result = run(name, args, cwd=cwd, expected=expected)
    with (EVIDENCE / 'test-executables.txt').open('a') as stream:
        stream.write(str(binary.relative_to(ROOT)) + '\n')
    return result
