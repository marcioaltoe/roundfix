
func TestQA0270ParkAllFiveShapes(t *testing.T) {
 t.Parallel()
 count := 0
 for _, shape := range qaPartialShapes() {
  if !strings.Contains(shape.name, "network prohibited") { continue }
  count++
  t.Run(shape.name, func(t *testing.T) {
   home, repo := newImplementWorkspace(t, []implementSeed{{id: "task_01"}})
   workflow := newItemRecoveryWorkflowForRepository(t, home, repo)
   head := itemRecoveryHead(t, repo)
   const runID = "qa0270-five-shapes"
   gittest.Run(t, repo, "checkout", "-b", store.RunBranchPrefix+runID)
   dir := filepath.Join(repo, "docs", "specs", implementTestSlug, "qa")
   writeQAReportTestFile(t, filepath.Join(dir, "qa-report-2026-10-14.md"), shape.report())
   prd := filepath.Join(repo, "docs", "specs", implementTestSlug, "_prd.md")
   mustWrite(t, prd, mustRead(t, prd)+shape.unreachable())
   gittest.Run(t, repo, "add", "docs/specs")
   gittest.Run(t, repo, "commit", "-m", "docs: disposable queue policy evidence")
   gittest.Run(t, repo, "checkout", "--detach", head)
   result, err := workflow.runResult(t.Context(), repo, implementTestSlug, roundfixCommandResult{exitCode: exitRunFailed}, "", nil, &store.Run{ID: runID, State: store.StateUnresolved})
   if err != nil { t.Fatal(err) }
   want := shape.reason != ""
   if result.Outcome != delivery.RunOutcomeUnresolved || result.QAEnvironmentPartial != want {
    t.Fatalf("result=%+v want park=%v", result, want)
   }
   t.Logf("shape=%s park=%v", shape.name, result.QAEnvironmentPartial)
  })
 }
 if count != 5 { t.Fatalf("exercised %d shapes, want 5", count) }
}
