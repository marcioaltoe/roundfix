def yaml_profiles(profiles):
    out='profiles:\n'
    def inline(s): return '{runtime: '+s['runtime']+', model: '+s['model']+', reasoning_effort: '+s['reasoning_effort']+'}'
    for cat,p in profiles.items():
        out+='  '+cat+':\n    preferred: '+inline(p['preferred'])+'\n    fallbacks:\n'
        for f in p['fallbacks']: out+='      - '+inline(f)+'\n'
        if 'deviation' in p: out+='    deviation: {from: '+p['deviation']['from']+', reason: '+p['deviation']['reason']+'}\n'
    return out

def main(row):
    global CURRENT,LOG
    LOG=[]; home=Path(tempfile.mkdtemp(prefix='q403-home-')); CURRENT=ORIGINAL.copy()
    CURRENT.update(HOME=str(home),GOMODCACHE=MODCACHE,GOPROXY='off',GOSUMDB='off',RUSTUP_HOME=RUST,CARGO_HOME=CARGO,GIT_CONFIG_NOSYSTEM='1',GIT_CONFIG_GLOBAL=os.devnull)
    for key in ['GIT_DIR','GIT_WORK_TREE','GIT_INDEX_FILE','GIT_COMMON_DIR','GIT_CONFIG_COUNT']: CURRENT.pop(key,None)
    try:
        observation=globals()[row](); value='blocked (environment: no open Pull Request)' if row=='Q10' else 'pass'
        status(row,value,observation); print(row,value,flush=True)
    except Exception as e:
        import traceback
        note(traceback.format_exc()); print(row,'CHECK FAILED:',e,flush=True); raise
    finally:
        (EV/(row+'.txt')).write_text('\n'.join(LOG)+'\n')
        shutil.rmtree(home)
        for p in TEMPS: shutil.rmtree(p)
        TEMPS.clear()
for row in sys.argv[1:]: main(row)
