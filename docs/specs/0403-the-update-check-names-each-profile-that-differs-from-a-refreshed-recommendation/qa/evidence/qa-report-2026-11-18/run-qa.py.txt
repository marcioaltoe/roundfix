def command(args,cwd=R,env=None,ok=True):
    p=subprocess.run([str(x) for x in args],cwd=cwd,env=env or ENV,text=True,capture_output=True,timeout=300)
    LOG.append('$ '+ ' '.join(str(x) for x in args)+'\nstdout:\n'+p.stdout+'stderr:\n'+p.stderr+'exit: '+str(p.returncode)+'\n')
    if ok and p.returncode: raise AssertionError('command failed: '+str(args)+'\n'+p.stdout[-1500:]+p.stderr[-1500:])
    return p

def update(row,status,detail):
    t=REPORT.read_text(); t=re.sub(r'^\| '+row+r' \| [^|]+ \|', '| '+row+' | '+status+' |',t,flags=re.M)
    start=t.index('### '+row+'\n'); end=t.find('\n### ',start+5)
    if end<0: end=len(t)
    old=t[start:end]; closing=old.index('```',old.index('```')+3)+3
    intro=old[closing:].split('\n\n',2)[1] if '\n\n' in old[closing:] else ''
    block=old[:closing]+'\n\n'+intro+'\n\n'+detail+'\n'
    REPORT.write_text(t[:start]+block+t[end:]); print(row+': '+status+' — '+detail[:180],flush=True)

def row(rowid,fn):
    LOG.clear()
    try: detail=fn(); status='pass'
    except Exception as e:
        detail=str(e); status='fail'; LOG.append(traceback.format_exc())
    (E/(rowid+'.txt')).write_text('\n'.join(LOG)+'\nResult: '+status+'\n'+detail+'\n')
    update(rowid,status,detail+' [Evidence](evidence/qa-report-2026-11-18/'+rowid+'.txt).')
    return status

def cli(f,*args,base=False):

def snapshot(paths):
    out={}
    for root in paths:
        for p in root.rglob('*'):
            if '.git' in p.parts or p.is_symlink() or not p.is_file(): continue
            out[str(p.relative_to(root))+'@'+str(root)]=(hashlib.sha256(p.read_bytes()).hexdigest(),p.stat().st_mode)
    return out

def compile_test(pkg,name,tags=None,overlay=None):
    args=['go','test','-c','-o',str(R/'bin'/name)]

def run_test(name,pattern,fail=False):
    pkg={'qa-agent.test':'internal/agent','qa-cli.test':'internal/cli','qa-config.test':'internal/config','qa-skills.test':'skills','qa-docscontract.test':'internal/docscontract'}[name]
    p=command([R/'bin'/name,'-test.v','-test.timeout=5m','-test.run='+pattern],cwd=R/pkg,ok=not fail)
    if fail: assert p.returncode==1 and '--- FAIL:' in p.stdout and 'stdout' in p.stdout, 'sabotage must fail its output assertion'
    else: assert 'PASS' in p.stdout and 'no tests to run' not in p.stderr
    return p
