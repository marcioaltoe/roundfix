import os,sys,json,subprocess,tempfile,shlex
from pathlib import Path
state=json.loads(Path('/private/tmp/0359-qa-state.json').read_text());root=Path(state['root']);q=Path(state['q']);e=q/'evidence'
def update(row,status,detail):
 p=q/'qa-report-2026-11-06.md';s=p.read_text();s=s.replace('| '+row+' | pending |','| '+row+' | '+status+' |');s=s.replace('### '+row+'\n','### '+row+'\n',1);s += '\n<!-- observation '+row+' -->\n'+detail+'\n';p.write_text(s)
def run(row,name,args,cwd=None,env=None,expected=0):
 print('START',row,shlex.join(args),flush=True)
 with tempfile.TemporaryFile(mode='w+') as out:
  p=subprocess.run(args,cwd=cwd or root,env=env,stdout=out,stderr=subprocess.STDOUT,text=True)
  out.seek(0);body=out.read()
 (e/(name+'.txt')).write_text('$ '+shlex.join(args)+'\n'+body+'\nexit: '+str(p.returncode)+'\n')
 passed=p.returncode==expected
 print('END',row,'exit',p.returncode,'lines',len(body.splitlines()),flush=True)
 update(row,'pass' if passed else 'fail',row+': '+('Expected outcome observed.' if passed else 'Unexpected exit.')+' Evidence: [transcript](evidence/'+name+'.txt).')
 return p.returncode,body
