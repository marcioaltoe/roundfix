package main
import("context";"crypto/sha256";"encoding/json";"fmt";"os";"path/filepath";"strconv";"roundfix/internal/store";"roundfix/internal/delivery";"roundfix/internal/config")
func must(err error){if err!=nil{panic(err)}}
func main(){ctx:=context.Background();repo,home:=os.Args[1],os.Args[2];pid,err:=strconv.Atoi(os.Args[3]);must(err);identity,err:=store.OwnerProcessIdentity(ctx,pid);must(err);s,err:=store.Open(ctx,home);must(err);defer s.Close();q,err:=s.CreateDeliveryQueue(ctx,repo,[]string{"0149-example"});must(err);item:=q.Items[0];item.Stage=store.DeliveryStageChecking;item.Branch="roundfix/deliver-0149-example";item.Worktree="/worktrees/0149-example";_,_,_,err=s.RecordDeliveryQueueItemWorktree(ctx,repo,item.SpecSlug,item.Branch,item.Worktree);must(err);must(s.UpdateDeliveryQueueItem(ctx,repo,item));must(s.ClaimDeliveryQueueOwner(ctx,repo,pid,identity));artifact,err:=config.ResolveArtifactDirectory("",repo,home);must(err);digest:=sha256.Sum256([]byte(filepath.Clean(repo)));path:=filepath.Join(artifact,"delivery",fmt.Sprintf("delivery-%x-%d",digest[:6],pid),"turn.json");must(os.MkdirAll(filepath.Dir(path),0700));turn:=delivery.Turn{ReviewCapacity:2,Items:[]delivery.ItemTurn{{SpecSlug:item.SpecSlug,Stage:item.Stage,State:delivery.TurnAdvancing,Checks:[]string{"CI | Backend / Lint, Type Check, Test & Build",".github/workflows/ci-frontends.yml (no run yet)"}}}};data,err:=json.Marshal(struct{Schema string `json:"schema"`;OwnerPID int `json:"owner_pid"`;OwnerIdentity string `json:"owner_identity"`;delivery.Turn}{"roundfix/delivery-turn/v1",pid,identity,turn});must(err);must(os.WriteFile(path,data,0600))}
