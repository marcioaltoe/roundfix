package cli
import("encoding/json";"os";"os/exec";"path/filepath";"reflect";"strings";"testing")
func TestQARetiredBuiltBinary(t *testing.T){
 t.Parallel()
 root,err:=os.Getwd();if err!=nil{t.Fatal(err)};root=filepath.Clean(filepath.Join(root,"../.."))
 run:=func(t *testing.T,repo,format string, extra ...string)(map[string]any,string,string,int){
  t.Helper(); args:=[]string{"baseline","update","--repo",repo,"--format="+format};args=append(args,extra...)
  cmd:=exec.Command("./bin/roundfix",args...);cmd.Dir=root;cmd.Env=append(os.Environ(),"HOME="+t.TempDir(),"TMPDIR="+t.TempDir())
  var out,stderr strings.Builder;cmd.Stdout=&out;cmd.Stderr=&stderr;err:=cmd.Run();code:=0;if err!=nil{if e,ok:=err.(*exec.ExitError);ok{code=e.ExitCode()}else{t.Fatal(err)}}
  t.Logf("./bin/roundfix baseline update --repo <temporary-adopted-repository> --format=%s %s\nexit: %d\nstdout:\n%s\nstderr:\n%s",format,strings.Join(extra," "),code,out.String(),stderr.String())
  var result map[string]any;if format=="json"{if err:=json.Unmarshal([]byte(out.String()),&result);err!=nil{t.Fatal(err)}}
  return result,out.String(),stderr.String(),code
 }
 for _,format:=range []string{"json","text"}{t.Run(format,func(t *testing.T){
  repo:=deletableRetiredSkillsRepository(t); survivors:=map[string]string{};var lockBefore map[string]any
  if err:=json.Unmarshal(mustReadBytes(t,filepath.Join(repo,"skills-lock.json")),&lockBefore);err!=nil{t.Fatal(err)}
  for _,n:=range []string{"design-patterns","context7-cli","exa-web-search","local-only"}{survivors[n]=baselinePlanTestTree(t,filepath.Join(repo,".agents/skills",n))}
  before:=baselinePlanTestTree(t,repo);preview,_,stderr,code:=run(t,repo,"json");if code!=3||stderr!=""||preview["state"]!="plan_ready"{t.Fatalf("preview: %+v code %d",preview,code)}
  digest:=preview["planDigest"].(string)
  skipped,_,_,code:=run(t,repo,"json","--no-skills");if code!=0||skipped["planDigest"]==digest{t.Fatal("no-skills did not omit deletion digest")}
  if before!=baselinePlanTestTree(t,repo){t.Fatal("preview changed tree")}
  applied,out,stderr,code:=run(t,repo,format,"--confirm-plan",digest);if code!=0||stderr!=""{t.Fatalf("apply failed %d",code)}
  if format=="json"{if applied["state"]!="verified"{t.Fatal(applied)}; deleted:=applied["skills"].(map[string]any)["deleted"].([]any);if len(deleted)!=1||deleted[0].(map[string]any)["skill"]!="tdd"{t.Fatal(deleted)}}else if !strings.Contains(out,"Skills deleted: 1\n- deleted tdd:"){t.Fatal("missing deletion text")}
  for _,r:=range []string{".agents",".claude"}{if _,err:=os.Lstat(filepath.Join(repo,r,"skills/tdd"));!os.IsNotExist(err){t.Fatal("tdd remains",r,err)}}
  var lockAfter map[string]any;if err:=json.Unmarshal(mustReadBytes(t,filepath.Join(repo,"skills-lock.json")),&lockAfter);err!=nil{t.Fatal(err)}
  if _,ok:=lockAfter["skills"].(map[string]any)["tdd"];ok{t.Fatal("tdd lock remains")}
  for n,tree:=range survivors{if tree!=baselinePlanTestTree(t,filepath.Join(repo,".agents/skills",n)){t.Fatal("survivor changed",n)};target,err:=os.Readlink(filepath.Join(repo,".claude/skills",n));if err!=nil||target!="../../.agents/skills/"+n{t.Fatal("survivor link",n)};if !reflect.DeepEqual(lockBefore["skills"].(map[string]any)[n],lockAfter["skills"].(map[string]any)[n]){t.Fatal("survivor lock changed",n)}}
  fresh,_,_,code:=run(t,repo,"json");if code!=0||fresh["state"]!="current"{t.Fatal("fresh preview",fresh,code)}
  after:=baselinePlanTestTree(t,repo);_,_,_,code=run(t,repo,"json","--yes");if code!=0{t.Fatal("repeat apply",code)};if after!=baselinePlanTestTree(t,repo){t.Fatal("repeat apply changed bytes")}
  t.Log("Independent disk read confirms tdd folder/link/lock absent, survivor trees/links/lock entries unchanged; fresh command current; repeat apply byte-idempotent.")
 })}
 t.Run("stale approval",func(t *testing.T){repo:=deletableRetiredSkillsRepository(t);preview,_,_,code:=run(t,repo,"json");if code!=3{t.Fatal(code)};mustWrite(t,filepath.Join(repo,".agents/skills/tdd/SKILL.md"),"# edited after preview\n");before:=baselinePlanTestTree(t,repo);result,_,stderr,code:=run(t,repo,"json","--confirm-plan",preview["planDigest"].(string));if code!=3||result["state"]!="action_required"||!strings.Contains(stderr,"does not match supplied plan"){t.Fatal("stale refusal",result,code)};if before!=baselinePlanTestTree(t,repo){t.Fatal("stale approval changed bytes")}})
}
