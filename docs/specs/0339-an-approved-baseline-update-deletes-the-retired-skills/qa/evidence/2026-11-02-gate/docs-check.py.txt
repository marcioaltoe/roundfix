from pathlib import Path
import re
roots=[Path('docs/user-guide'),Path('docs/adr'),Path('.agents/skills/roundfix'),Path('skills/roundfix'),Path('CONTEXT.md')]
files=[]
for root in roots:files.extend([root] if root.is_file() else sorted(root.rglob('*.md')))
old='0246-a-retired-skill-leaves-the-baseline-and-the-adopter-removes-its-copy.md'
links=[];stale=[]
for f in files:
 s=f.read_text()
 for n,line in enumerate(s.splitlines(),1):
  if old in line:links.append(f'{f}:{n}: {line}')
 for para in re.split(r'\n\s*\n',s):
  flat=' '.join(para.split())
  if re.search(r'(?:never deletes? (?:an installed|the) cop|without deleting it)',flat,re.I) and 'retired' in flat.lower():stale.append(f'{f}: {flat}')
assert not links,links;assert not stale,stale;assert not Path('docs/adr/'+old).exists()
