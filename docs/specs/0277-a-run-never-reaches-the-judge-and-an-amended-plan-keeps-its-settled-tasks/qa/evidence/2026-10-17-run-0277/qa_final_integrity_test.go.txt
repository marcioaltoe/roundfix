package spec_test
import ("testing"; "os"; "path/filepath"; "roundfix/internal/spec"; "roundfix/internal/secretscan")
func TestQAReportFinalIntegrity(t *testing.T) {
 t.Parallel()
 root:=os.Getenv("ROUNDFIX_QA_REPORT_DIR")
 report,err:=spec.ReadQAReport(root);if err!=nil{t.Fatal(err)}
 if err:=spec.QAReportEligibility(root,report);err!=nil{t.Fatal(err)}
 if report.RowsBlockedEnvironment!=4 || report.RowsBlockedFinding!=0 || report.RowsBlockedDeclared!=0 || report.EnvironmentRowsNeedingOverride()!=0 {t.Fatalf("unexpected blocked counts: %+v",report)}
 files:=0
 err=filepath.WalkDir(filepath.Join(root,"qa"),func(path string,entry os.DirEntry,err error)error{if err!=nil{return err};if entry.IsDir(){return nil};data,err:=os.ReadFile(path);if err!=nil{return err};if matches:=secretscan.ScanContent(path,data);len(matches)>0{t.Fatalf("secret locations: %v",matches)};files++;return nil});if err!=nil{t.Fatal(err)}
 t.Logf("qualifying partial: 4 exempt environment rows; 0 rows needing override; %d report/evidence files scanned; no Secret Match",files)
}
