def find_id(obj,needle):
    if isinstance(obj,dict):
        if obj.get('id')==needle:return obj
        for val in obj.values():
            got=find_id(val,needle)
            if got is not None:return got
    elif isinstance(obj,list):
        for val in obj:
            got=find_id(val,needle)
            if got is not None:return got
