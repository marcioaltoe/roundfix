def run(args,cwd=ROOT,log=None,expected=0,env=None):
 p=subprocess.run(args,cwd=cwd,env=env or ENV,text=True,stdout=subprocess.PIPE,stderr=subprocess.STDOUT)
 text='$ '+shlex.join(map(str,args))+'\n'+p.stdout+'\nexit: '+str(p.returncode)+'\n'
 if log: log.write(text); log.flush()
 print(shlex.join(map(str,args[:5])), 'exit',p.returncode,flush=True)
 if expected is not None and p.returncode!=expected: raise AssertionError(text)
 return p

def result(row,status,detail):

 assert n==1
 s=s.replace('### '+row+'\n','### '+row+'\n\nObservation: '+detail+'\n',1)
 REPORT.write_text(s)

def executable(path):

  s=s.replace('user_flow_executables:\n','user_flow_executables:\n  - '+rel+'\n',1)

@contextlib.contextmanager
def scratch(label):
 p=pathlib.Path(subprocess.check_output(['mktemp','-d',str(ROOT/'bin'/('qa-'+label+'.XXXXXX'))],env=ENV,text=True).strip())
 try: yield p
 finally:
  shutil.rmtree(p)
  print('removed scratch '+label,flush=True)

@contextlib.contextmanager
def cli_env(tmp=False):
 dirs=[]; e=ENV.copy(); e.update(CACHE)
 try:
  for name in (('HOME','TMPDIR') if tmp else ('HOME',)):
   p=pathlib.Path(subprocess.check_output(['mktemp','-d'],env=ENV,text=True).strip()); dirs.append(p); e[name]=str(p)
  yield e
 finally:
  for p in dirs: shutil.rmtree(p)
