def body_replace(text,signature,newbody):
 start=text.index(signature); opening=text.index('{',start); depth=1; pos=opening+1
 while depth:
  if text[pos]=='{':depth+=1
  elif text[pos]=='}':depth-=1
  pos+=1
 return text[:opening+1]+'\n'+newbody+'\n'+text[pos-1:]
def replace_once(text,a,b):
 assert text.count(a)==1,(a,text.count(a));return text.replace(a,b,1)
