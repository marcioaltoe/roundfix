def cleanenv():
    return {k:v for k,v in os.environ.items() if k not in KEYS and not k.endswith('_API_KEY')}

def raw(args, cwd=ROOT):
    return subprocess.check_output(args, cwd=cwd, env=cleanenv()).decode()

def run(name, args, cwd=ROOT, isolated=False, expected=0):
    e = cleanenv()
    with tempfile.TemporaryDirectory(prefix='qah-') as h, tempfile.TemporaryDirectory(prefix='qat-') as t:
        if isolated:
            caches = json.loads(raw(['go','env','-json','GOMODCACHE','GOCACHE']))
            e.update(caches)
            e.update(HOME=h,TMPDIR=t)
        p = subprocess.run(args, cwd=cwd, env=e, capture_output=True, text=True)
    (EV/(name+'.txt')).write_text('command: '+shlex.join(map(str,args))+'\ncwd: '+str(cwd)+'\nexit: '+str(p.returncode)+'\nstdout:\n'+p.stdout+'\nstderr:\n'+p.stderr)
    print(name, 'exit', p.returncode, flush=True)
    if expected is not None:
        assert p.returncode == expected, name
    return p

def cli(name, args, cwd=ROOT, base=False, expected=None):
    a = ['./bin/roundfix-delivery-base' if base else './bin/roundfix']
    if cwd != ROOT:
        (cwd/'bin').mkdir(exist_ok=True)
        link=cwd/'bin'/('roundfix-base' if base else 'roundfix')
        if not link.exists():
            link.symlink_to(ROOT/('bin/roundfix-delivery-base' if base else 'bin/roundfix'))
        a=['./bin/roundfix-base' if base else './bin/roundfix']
    return run(name, a+args, cwd=cwd, isolated=True, expected=expected)

def setrow(row, status):
    p = SD/'qa/qa-report-2026-11-09.md'
    text = p.read_text()
    text, n = re.subn(r'(?m)^\| '+str(row)+r' \| [^|]+ \|', '| '+str(row)+' | '+status+' |', text)
    assert n == 1, (row,n)
    p.write_text(text)
