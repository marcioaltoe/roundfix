def run(args, cwd=ROOT, environ=None, timeout=2700):
    return subprocess.run(args, cwd=cwd, env=environ, text=True, stdout=subprocess.PIPE, stderr=subprocess.PIPE, timeout=timeout)

def save(name, args, p):
    output = 'Command: ' + shlex.join(args) + '\nExit: ' + str(p.returncode) + '\nstdout:\n' + p.stdout + '\nstderr:\n' + p.stderr
    # Test assertions sometimes include inherited fixture variables; retain only numeric budget values.
    output = re.sub(r'(?m)^.*(?:PATH=|HOME=|TMPDIR=|GOCACHE=|GOMODCACHE=|KEEP=).*$', '[fixture environment omitted]', output)
    (EVIDENCE / (name + '.txt')).write_text(output)

def cli(name, args):
    caches = {k: run(['go', 'env', k]).stdout.strip() for k in ['GOCACHE', 'GOMODCACHE']}
    with tempfile.TemporaryDirectory() as home, tempfile.TemporaryDirectory() as tmp:
        environ = os.environ.copy()
        for key in KEYS:
            environ.pop(key, None)
        environ.update(caches, HOME=home, TMPDIR=tmp)
        cmd = ['./bin/roundfix'] + args
        p = run(cmd, environ=environ)
    save(name, cmd, p)
    return p

def update(row, status, refs, detail):
    data = REPORT.read_text()
    data, n = re.subn(r'^\| ' + row + r' \| pending \|', '| ' + row + ' | ' + status + ' |', data, flags=re.M)
    assert n == 1, (row, n)
    inputs = '\n'.join('  - kind: ' + kind + '\n    ref: ' + ref for kind, ref in refs)
    data += '\n### ' + row + '\n\n```yaml\ninputs:\n' + inputs + '\n```\n\n' + detail + '\n'
    REPORT.write_text(data)
