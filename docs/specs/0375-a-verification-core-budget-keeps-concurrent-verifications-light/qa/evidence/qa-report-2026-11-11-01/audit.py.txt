import importlib.util

loader = importlib.machinery.SourceFileLoader('qa_runner', str(Path(__file__).with_name('runner.py.txt')))
spec = importlib.util.spec_from_loader(loader.name, loader)

flat = lambda s: ' '.join(s.split())
observations = []
checks = []
def check(label, condition):
    observations.append(('PASS ' if condition else 'FAIL ') + label)
    checks.append(condition)

config = flat(read('docs/user-guide/configuration.md'))
for sentence in [
    'Roundfix gives every Verification and every Agent Session of an Implement Run, and the repository gate of a Delivery Queue, a Verification Core Budget: half of the logical CPUs, at least 1, divided by `runs.max_verifying`, or by `runs.max_active` while `runs.max_verifying` is `0`, and never below 1.',
    'It sets `GOMAXPROCS` and `ROUNDFIX_VERIFICATION_CORES` to that number.',
    'A `GOMAXPROCS` that holds a positive whole number in the environment of the process that starts the Run or the queue takes its place.',
    'A test runner that reads neither variable keeps its own parallelism; changing its configuration is a tooling change your repository authorizes.']:
    check('configuration sentence: '+sentence, sentence in config)
check('budget section follows capacity section', config.index('## Machine Verification Capacity') < config.index('## Verification Core Budget'))
check('Vitest worker example opts in', 'maxWorkers' in config and 'process.env.ROUNDFIX_VERIFICATION_CORES' in config)
check('doctor guide names the cores line', 'verification-cores:' in read('docs/user-guide/commands/doctor.md'))

    check('deliver guide names '+name, name in read('docs/user-guide/commands/deliver.md'))

    a=read('.agents/skills/roundfix/references/'+ref+'.md')

    check(ref+' mirror byte equality', a == b)
    required = ['Verification Core Budget','verification-cores:'] if ref == 'setup' else ['Verification Core Budget','GOMAXPROCS','ROUNDFIX_VERIFICATION_CORES']
    check(ref+' reference names its Task 03 Requirement 7 contract', all(n in a for n in required))
context = read('CONTEXT.md')

contract = flat(tech.split('7. **Glossary text**',1)[1].split('exactly: "',1)[1].split('" The closing',1)[0])
glossary_text = re.sub(r'\[ADR-0385\]\([^)]*\)', 'ADR-0385', glossary).split(' _Avoid_:',1)[0]
check('glossary matches API Contract 7', glossary_text == contract)
check('glossary follows Machine Verification Capacity', context.index('**Machine Verification Capacity**:') < context.index('**Verification Core Budget**:'))
check('glossary avoid line exact', '_Avoid_: core limit, CPU quota, Machine Verification Capacity' in glossary)

    section=q.SD.joinpath(name).read_text().split('## Glossary',1)[1].split('\n## ',1)[0]
    check(name+' declares added term', 'adds:' in section and 'Verification Core Budget' in section)

writers=[]
users=[]
for p in sorted(q.ROOT.joinpath('internal').rglob('*.go')):
    rel=p.relative_to(q.ROOT).as_posix()
    if p.name.endswith('_test.go') or rel.startswith('internal/corebudget/'):
        continue
    content=p.read_text()
    for i,line in enumerate(content.splitlines(),1):
        if re.search(r'"(?:GOMAXPROCS|ROUNDFIX_VERIFICATION_CORES)=',line):
            writers.append(f'{rel}:{i}')
        if re.search(r'corebudget\.(GoVariable|Variable|Environment)\b',line):
            users.append(rel)
            observations.append('environment boundary '+rel+':'+str(i)+': '+line.strip())

(q.EVIDENCE/'D01.txt').write_text('\n'.join(observations)+'\n')

    sys.exit(0 if all(checks) else 1)

observations=[]
checks=[]
commits={'task_01':'29babf98','task_02':'7d13a542','task_03':'97d14077','task_04':'a6b0d6c0','task_06':'84d94eab'}
authorized=set(re.findall(r'^  - (\S+)$',q.SD.joinpath('_authorization.md').read_text().split('paths:',1)[1].split('operations:',1)[0],re.M))
all_paths=set()
for task,commit in commits.items():
    text=q.SD.joinpath(task+'.md').read_text()
    declared=set(re.findall(r'^- (?:instruction|interface|creates): `([^`]+)`',text,re.M))
    declared.add(str(q.SD.joinpath(task+'.md').relative_to(q.ROOT)))
    if '## Recorded paths' in text:
        declared.update(re.findall(r'`([^`]+)`',text.split('## Recorded paths',1)[1].split('\n## ',1)[0]))
    args=['git','diff-tree','--no-commit-id','--name-only','-r',commit]
    p=q.run(args)
    assert p.returncode==0
    changed=set(p.stdout.splitlines())
    all_paths.update(changed)
    observations.append('Command: '+' '.join(args)+'\n'+p.stdout)
    check(task+' changed paths within declarations plus assigned task file', changed <= declared)
    if not changed <= declared:
        observations.append('Undeclared: '+', '.join(sorted(changed-declared)))
    governed={p for p in changed if p.startswith('.agents/skills/') or re.match(r'skills/[^/]+/SKILL.md$',p)}
    check(task+' governed changes bounded by operative grant', governed <= authorized)
    if governed:
        check(task+' grant predates commit',q.run(['git','merge-base','--is-ancestor',q.BASE,commit]).returncode==0)
        delta=q.run(['git','diff',commit+'^',commit,'--',str(q.SD.joinpath('_authorization.md').relative_to(q.ROOT))])
        check(task+' no authorization folded into implementation commit',delta.stdout=='')
        observations.append(task+' operative grant and chronology also established by preserved Daemon authorization audit inputs.')

        prohibited.append(path)
