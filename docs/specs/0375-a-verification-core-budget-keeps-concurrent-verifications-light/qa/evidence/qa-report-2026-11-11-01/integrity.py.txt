import importlib.machinery
import importlib.util
from pathlib import Path
import re
import sys

sys.dont_write_bytecode = True
loader=importlib.machinery.SourceFileLoader('qa_runner',str(Path(__file__).with_name('runner.py.txt')))
spec=importlib.util.spec_from_loader(loader.name,loader)
q=importlib.util.module_from_spec(spec)
loader.exec_module(q)
text=q.REPORT.read_text()
checks=[]
def check(name, condition):
    checks.append((name,bool(condition)))
    print(('PASS ' if condition else 'FAIL ')+name,flush=True)

table=text.split('## Results\n',1)[1].lstrip().split('\n\n',1)[0]
rows=[]
for line in table.splitlines()[2:]:
    cells=[c.strip() for c in line.split('|')[1:-1]]
    check('row has three columns '+cells[0],len(cells)==3)
    rows.append(cells)
check('43 unique results',len(rows)==43 and len({r[0] for r in rows})==43)
check('no pending or failed row',not any(r[1] in {'pending','fail'} for r in rows))
carried=[r for r in rows if r[1].startswith('carried (')]
check('17 seeded carried rows kept',len(carried)==17)
check('24 fresh rows pass',sum(r[1]=='pass' for r in rows)==24)
check('one environment block',sum(r[1].startswith('blocked (environment:') for r in rows)==1)
check('one declared block',sum(r[1].startswith('blocked (declared:') for r in rows)==1)
check('only one frontmatter block',text.startswith('---\n') and text.count('\n---\n')==1)
check('Daemon auditor identity unchanged','auditing_binary: "0.138.0 (2cb2b816, built 2026-11-11 05:33:27 -0300)"' in text and 'auditor_staleness: "current: commit ancestry: build commit does not predate the delivery base"' in text)
check('Daemon-owned snapshots absent','evidence_snapshots:' not in text)
for row,status,provenance in rows:
    if status.startswith('carried ('):
        continue
    match=re.search(r'^### '+row+r'\n(.*?)(?=^### |\Z)',text,re.M|re.S)
    check(row+' typed repository inputs',match is not None and 'kind: repository_path' in match.group(1))
    if row.startswith('M') or row in {'S01','PR01'}:
        check(row+' commit range declared',match is not None and 'kind: commit_range' in match.group(1) and q.BASE+'..'+q.HEAD in match.group(1))
results_section=text.split('## Results\n',1)[1].split('\n## ',1)[0]
check('no detached result rows',len(re.findall(r'^\| (?:[A-Z]\d{2}|PR01) \|',results_section,re.M))==43)
for target in re.findall(r'\[[^\]]*\]\(([^)]+)\)',text):
    if '://' in target or target.startswith('#'):
        continue
    p=(q.REPORT.parent/target.split('#',1)[0]).resolve()
    check('evidence exists '+p.name,p.exists())
    ignored=q.run(['git','check-ignore','--',str(p)])
    check('evidence unignored '+p.name,ignored.returncode==1)
for line in text.split('user_flow_executables:\n',1)[1].split('\nauditing_binary:',1)[0].splitlines():
    if line.startswith('  - '):
        p=(q.ROOT/line[4:]).resolve()
        check('executable in this Run '+p.name,p.is_relative_to(q.ROOT) and p.exists())
patterns=re.findall(r'\{"([^"]+)", regexp.MustCompile\(`([^`]+)`\)\}',(q.ROOT/'internal/secretscan/patterns.go').read_text())
check('loaded all canonical secret patterns',len(patterns)==10)
for file in [q.REPORT]+sorted(q.EVIDENCE.rglob('*')):
    if not file.is_file():
        continue
    data=file.read_bytes()
    try:
        content=data.decode('utf-8')
    except UnicodeDecodeError:
        check('text evidence '+file.name,False)
        continue
    check('no binary evidence '+file.name,b'\x00' not in data)
    for i,line in enumerate(content.splitlines(),1):
        for kind,pattern in patterns:
            for match in re.finditer(pattern,line):
                if kind=='credential-assignment':
                    value=match.group(1)
                    if not (re.search('[A-Za-z]',value) and re.search('[0-9]',value)):
                        continue
                check('secret match '+file.name+':'+str(i)+':'+kind,False)
failed=[name for name,ok in checks if not ok]
print('Checks:',len(checks),'Failures:',len(failed),flush=True)
sys.exit(1 if failed else 0)
