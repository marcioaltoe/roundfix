def git(*args):

def evidence(row, text, ok, detail):
    (HERE / (row + '.txt')).write_text(text)
    g.update(row, 'pass' if ok else 'fail', detail + '\n\nEvidence: [audit and observations]'
             '(evidence/qa-report-2026-11-11/' + row + '.txt).')

          config.index('## Machine Verification Capacity') < config.index('## Verification Core Budget'))

          doctor.index('- `verification-capacity:`') < doctor.index('- `verification-cores:`'))

    check('deliver guide names both variables', all(v in deliver for v in ('GOMAXPROCS', 'ROUNDFIX_VERIFICATION_CORES')))

def scope():
    commits = ['9f0b955bb9cf6fa91f0c1f1569ed02af0d7dd104',
               '4a313873276b5776400cd948b42297435cbc86e8',
               'a77ea3a6e532307aa3fa27eca66e30eb305c321a', g.HEAD]
    auth = g.SPEC / '_authorization.md'
    grant = auth.read_text()
    bounded = set(re.findall(r'^  - (.+)$', grant.split('paths:\n', 1)[1].split('operations:', 1)[0], re.M))
    log, bad = [], []
    auth_ref = str(auth.relative_to(g.ROOT))
    assert git('show', g.BASE + ':' + auth_ref) == grant
    log += ['commit_range: ' + g.RANGE,
            'Approved authorization is byte-identical to the record at the Delivery Base.',
            'Seeded mechanical audit already covers the governed Task commits 02 and 03.']
    forbidden_exact = {'Makefile', 'go.mod', 'go.sum', '.roundfixrc.yml', 'skills-lock.json',
                       'CHANGELOG.md', 'internal/app/version.go', 'dist/npm/roundfix/package.json',
                       'docs/references/coverage-record.json'}
    forbidden_prefixes = ('.github/workflows/', 'internal/baseline/', 'docs/agents/', '.claude/', '.codex/', '.opencode/')
    changed_all = set()
    for i, commit in enumerate(commits, 1):
        task_path = g.SPEC / f'task_{i:02}.md'
        task = task_path.read_text()
        declared = set(re.findall(r'^- (?:instruction|creates|interface|deletes): `([^`]+)`', task, re.M))
        declared.add(str(task_path.relative_to(g.ROOT)))
        if '## Recorded paths' in task:
            section = task.split('## Recorded paths', 1)[1].split('\n## ', 1)[0]
            declared.update(re.findall(r'`([^`]+)`', section))
            log.append(f'task_{i:02} Recorded paths included: ' + section.strip())
        changed = git('diff-tree', '--no-commit-id', '--name-only', '-r', commit).splitlines()
        changed_all.update(changed)
        git('merge-base', '--is-ancestor', g.BASE, commit)
        log.append(f'task_{i:02}: ' + commit + '; changed files:')
        for path in changed:
            log.append('  ' + path)
            if path not in declared:
                bad.append('undeclared Task path: ' + path)
            if path in forbidden_exact or path.startswith(forbidden_prefixes):
                bad.append('forbidden path: ' + path)
            if path.startswith(('.agents/skills/', 'skills/')) and '/roundfix/' not in path and path != 'skills/testdata/owned-skill-versions.json':
                bad.append('another skill changed: ' + path)
            # Every changed path falls in ordinary Go/docs or the current governed skill class.
            governed = path.startswith('.agents/skills/') or path == 'skills/roundfix/SKILL.md'
            if governed and path not in bounded:
                bad.append('unbounded governed path: ' + path)
    for path in changed_all:
        if '/roundfix/' not in path:
            continue
        old, new = git('show', g.BASE + ':' + path), git('show', g.HEAD + ':' + path)
        def settlement(text):
            found = re.search(r'^### QA settlement\n.*?(?=^###? |\Z)', text, re.M | re.S)
            return found.group(0) if found else ''
        if settlement(old) != settlement(new):
            bad.append('QA settlement section changed: ' + path)
    if git('diff', '--name-only', 'HEAD').strip():
        bad.append('tracked working-tree delta exists')
    log += ['All five changed governed paths are bounded by the approved grant.',
            'Authorization predates every consuming Task; no authorization or fix is folded into a Task commit.',
            'No prerequisite/consequent fix is declared; the seeded consequent-fix mechanical skip is preserved.',
            'No forbidden file, Baseline module, vendored skill or QA settlement section changed.',
            'No tracked working-tree delta; Agent-authored output is confined to this Spec QA directory.']
    log += ['FAIL: ' + message for message in bad]
    evidence('S01', '\n'.join(log) + '\n', not bad,
             'Maintainer, repository history. Audited each Task with `git diff-tree --no-commit-id '
             '--name-only -r <commit>`; every changed path is declared by its Task, including the assigned '
             'Task record. No Recorded paths additions exist. Compared the changed governed set with the '
             'grant, verified authorization ancestry from the Delivery Base, checked every prohibited '
             'path family and compared QA settlement sections byte for byte. The seeded authorization '
             'audit remains authoritative for the two governed commits; this row supplies the explicitly '
             'required changed-file and chronology observations. No unauthorized or out-of-scope edit was found.')
