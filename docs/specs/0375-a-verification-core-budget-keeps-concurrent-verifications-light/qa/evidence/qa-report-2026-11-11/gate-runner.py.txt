def sanitize(text):
    # Mutation assertions may print fixture environment entries; keep only budget numbers.
    def entry(match):
        name, value = match.group(1), match.group(2)
        if name in ('GOMAXPROCS', 'ROUNDFIX_VERIFICATION_CORES') and value.isascii() and value.isdigit():
            return match.group(0)
        return '"' + name + '=<value omitted>"'
    return re.sub(r'"([A-Z][A-Z0-9_]*)=([^"\n]*)"', entry, text)

def update(row, status, detail):
    text = REPORT.read_text()
    pattern = r'^\| ' + re.escape(row) + r' \| [^|]+ \| [^\n]+ \|$'
    replacement = '| ' + row + ' | ' + status + ' | ' + PROVENANCE[row] + ' |'
    text, count = re.subn(pattern, lambda m: replacement, text, flags=re.M)
    assert count == 1, (row, count)
    block = '\n### ' + row + '\n\n' + detail.strip() + '\n'
    old = r'\n### ' + row + r'\n.*?(?=\n### |\n## |\Z)'
    if re.search(old, text, re.S):
        text = re.sub(old, lambda m: block, text, flags=re.S)
    else:
        text += block
    REPORT.write_text(text)

def run(row, args, *, cwd=ROOT, env=None, expected=0, marker=None, filename=None):
    print(row + ': running ' + (shlex.join(args) if isinstance(args, list) else args), flush=True)
    result = subprocess.run(args, cwd=cwd, env=env, shell=isinstance(args, str),
                            capture_output=True, text=True)
    ok = result.returncode == expected
    if marker:
        ok = ok and marker in result.stdout + result.stderr
    name = filename or row + '.txt'
    command = shlex.join(args) if isinstance(args, list) else args
    log = ('command: ' + command + '\nexit: ' + str(result.returncode)
           + '\nstdout:\n' + result.stdout + '\nstderr:\n' + result.stderr)
    (OUT / name).write_text(log)
    update(row, 'pass' if ok else 'fail',
           'Inputs: `repository_path` = `' + str(ROOT) + '`.\n\n'
           'Command: `' + command.replace('`', '\\`') + '`. Exit ' + str(result.returncode)
           + '. ' + ('Expected observable confirmed.' if ok else 'Expected observable was not reached.')
           + '\n\nEvidence: [command and output](evidence/qa-report-2026-11-11/' + name + ').')
    print(row + ': ' + ('PASS' if ok else 'FAIL') + ', exit ' + str(result.returncode), flush=True)
    return result, ok

def tasks():
    commands = []
    for i in range(1, 5):
        task = (SPEC / f'task_{i:02}.md').read_text()
        section = task.split('## Verification\n', 1)[1].split('\n## ', 1)[0]
        for number, command in enumerate(re.findall(r'^- `(.*)` — expected:', section, re.M), 1):
            commands.append((f'task_{i:02}-{number}', command))
    assert len(commands) == 9, len(commands)
    for name, command in commands:
        print(name + ': running authored Verification', flush=True)
        result = subprocess.run(command, shell=True, cwd=ROOT, capture_output=True, text=True)
        (OUT / (name + '.txt')).write_text(sanitize('command: ' + command + '\nexit: ' + str(result.returncode)
                                          + '\nstdout:\n' + result.stdout + '\nstderr:\n' + result.stderr))
        print(name + ': exit ' + str(result.returncode), flush=True)
