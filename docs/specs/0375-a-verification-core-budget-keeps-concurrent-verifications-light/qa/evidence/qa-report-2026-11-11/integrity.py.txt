from pathlib import Path
import collections
import re
import subprocess

root = Path.cwd()
spec = root / 'docs/specs/0375-a-verification-core-budget-keeps-concurrent-verifications-light'
report = spec / 'qa/qa-report-2026-11-11.md'
out = spec / 'qa/evidence/qa-report-2026-11-11'
text = report.read_text()
table = text.split('## Results\n\n', 1)[1].split('\n\n', 1)[0]
rows = [[v.strip() for v in line.strip('|').split('|')] for line in table.splitlines()[2:]]
assert len(rows) == 43, len(rows)
assert all(len(row) == 3 for row in rows)
assert len({row[0] for row in rows}) == 43
counts = collections.Counter('pass' if row[1] == 'pass' else 'fail' if row[1] == 'fail'
                            else 'environment' if row[1].startswith('blocked (environment:')
                            else 'declared' if row[1].startswith('blocked (declared:')
                            else row[1] for row in rows)
assert counts == {'pass': 40, 'fail': 1, 'environment': 1, 'declared': 1}, counts
for row in rows:
    body = text.split('\n### ' + row[0] + '\n', 1)[1].split('\n### ', 1)[0].split('\n## ', 1)[0]
    assert '```yaml\ninputs:\n' in body and 'kind: repository_path' in body, row[0]
    if row[0] == 'S01':
        assert 'kind: commit_range' in body
assert 'evidence_snapshots:' not in text
frontmatter = text.split('---\n', 2)[1]
for expected in ('verdict: fail', 'status: closed', 'rows_blocked_environment: 1',
                 'rows_blocked_finding: 0', 'rows_blocked_declared: 1',
                 'auditing_binary: "0.138.0 (996769bb, built 2026-11-11 03:43:50 -0300)"',
                 'auditor_staleness: "current: commit ancestry: build commit does not predate the delivery base"'):
    assert expected in frontmatter, expected
assert text.count('\n---\n') == 1
links = re.findall(r'\]\(([^)]+)\)', text)
for target in links:
    path = report.parent / target.split('#', 1)[0]
    assert path.is_file(), target
    ignored = subprocess.run(['git', 'check-ignore', '-q', str(path.relative_to(root))], cwd=root)
    assert ignored.returncode == 1, target
patterns_source = (root / 'internal/secretscan/patterns.go').read_text()
patterns = [(name, re.compile(expression)) for name, expression in
            re.findall(r'\{"([^"\n]+)", regexp.MustCompile\(`([^`]+)`\)\}', patterns_source)]
assert len(patterns) == 10
files = [file for file in sorted((spec / 'qa').rglob('*')) if file.is_file()]
matches = []
for file in files:
    content = file.read_bytes()
    assert b'\x00' not in content, file
    content = content.decode('utf-8')
    ignored = subprocess.run(['git', 'check-ignore', '-q', str(file.relative_to(root))], cwd=root)
    assert ignored.returncode == 1, file
    for line_number, line in enumerate(content.splitlines(), 1):
        for name, pattern in patterns:
            found = list(pattern.finditer(line))
            if name == 'credential-assignment':
                found = [match for match in found if re.search('[A-Za-z]', match.group(1)) and re.search('[0-9]', match.group(1))]
            if found:
                matches.append((str(file.relative_to(root)), line_number, name))
assert not matches, matches  # Only locations and kinds, never matched values.
tracked = subprocess.run(['git', 'diff', '--name-only', 'HEAD'], capture_output=True, text=True, check=True).stdout
assert not tracked.strip(), tracked
for i in range(1, 15):
    mutation = (out / f'M{i:02}.txt').read_text()
    assert 'BASELINE exit: 0' in mutation and 'MUTATION exit: 1' in mutation
    assert '--- PASS:' in mutation and '--- FAIL:' in mutation
for task, count in ((1, 2), (2, 4), (3, 2), (4, 1)):
    for number in range(1, count + 1):
        assert '\nexit: 0\n' in (out / f'task_{task:02}-{number}.txt').read_text()
summary = ('Report integrity: PASS\n'
           '43 contiguous Results rows: 40 pass, 1 fail, 1 environment-blocked, 1 declared-blocked.\n'
           'Every row has typed repository_path inputs; the scope audit has commit_range.\n'
           'One frontmatter block, closed/fail, correct blocked counts; auditor fields preserved.\n'
           + str(len(links)) + ' local evidence links resolve and are unignored.\n'
           + str(len(files)) + ' QA files are UTF-8 text and unignored.\n'
           'No secret-shaped match against the ten expressions read from internal/secretscan/patterns.go '
           '(Agent integrity check; Daemon secret scan remains separate).\n'
           'All 14 mutation pairs and nine authored Task Verification command exits confirmed.\n'
           'No tracked implementation or Task changes; no evidence_snapshots field.\n')
(out / 'report-integrity.txt').write_text(summary)
print(summary)
