import json
import os
from pathlib import Path
import subprocess
import sys
import tempfile

ROOT = Path.cwd()
SPEC = '0375-a-verification-core-budget-keeps-concurrent-verifications-light'
OUT = ROOT / 'docs/specs' / SPEC / 'qa/evidence/qa-report-2026-11-11'
CASES = {
    'version': ['--version'],
    'precondition': ['spec', 'check', SPEC, '--strict'],
    'doctor-help': ['doctor', '--help'],
    'archive-plan': ['archive', SPEC, '--plan', '--offline'],
}

def main():
    case = sys.argv[1]
    child_env = os.environ.copy()
    names = {'GH_TOKEN', 'GITHUB_TOKEN', 'TYPESAFE_API_KEY',
             'ROUNDFIX_OPENROUTER_JUDGE_API_KEY', 'ROUNDFIX_OPENROUTER_API_KEY',
             'ROUNDFIX_TYPESAFE_API_KEY'}
    for transport in json.loads((ROOT / 'internal/judge/questions.json').read_text())['transports']:
        names.add(transport['key_variable'])
    for name in names:
        child_env.pop(name, None)
    caches = subprocess.run(['go', 'env', 'GOMODCACHE', 'GOCACHE'],
                            capture_output=True, text=True, check=True).stdout.splitlines()
    with tempfile.TemporaryDirectory(prefix='qa0375-home-') as home:
        with tempfile.TemporaryDirectory(prefix='qa0375-tmp-') as tmp:
            child_env.update(HOME=home, TMPDIR=tmp, GOMODCACHE=caches[0], GOCACHE=caches[1])
            result = subprocess.run(['./bin/roundfix'] + CASES[case], cwd=ROOT,
                                    env=child_env, capture_output=True, text=True)
    record = ('command: ./bin/roundfix ' + ' '.join(CASES[case]) + '\n'
              'HOME and TMPDIR: fresh temporary directories; removed after exit\n'
              'GOMODCACHE and GOCACHE: retained at go env values (values omitted)\n'
              'unset variables: ' + ', '.join(sorted(names)) + '\n'
              'exit: ' + str(result.returncode) + '\nstdout:\n' + result.stdout
              + '\nstderr:\n' + result.stderr)
    (OUT / (case + '.txt')).write_text(record)
    print(record)
    return result.returncode

if __name__ == '__main__':
    sys.exit(main())
