import importlib.machinery
import importlib.util
import io
import os
from pathlib import Path
import shlex
import subprocess
import sys
import tarfile
import tempfile

HERE = Path(__file__).resolve().parent
sys.dont_write_bytecode = True
loader = importlib.machinery.SourceFileLoader('gate', str(HERE / 'gate-runner.py.txt'))
spec = importlib.util.spec_from_loader(loader.name, loader)
gate = importlib.util.module_from_spec(spec)
loader.exec_module(gate)

CORE = 'internal/corebudget/corebudget.go'
MUTATIONS = [
 (CORE, 'digits := explicit != ""', 'digits := false', 2, 'positive_whole_number'),
 (CORE, "if explicit[i] < '0' || explicit[i] > '9' {", "if (explicit[i] < '0' || explicit[i] > '9') && !(i == 0 && explicit[i] == '+') {", 2, 'signed_is_ignored'),
 (CORE, 'if in.MaxVerifying > 0 {', 'if false && in.MaxVerifying > 0 {', 1, 'two_at_once'),
 (CORE, 'budget.Share = max(1, budget.LogicalCPUs/2)', 'budget.Share = max(1, budget.LogicalCPUs)', 1, 'default_on_twelve'),
 (CORE, 'if cores > 0 && (name == GoVariable || name == Variable) {', 'if cores > 0 && name == Variable {', 3, 'budget_four'),
 ('internal/daemon/daemon.go', 'cmd.Env = corebudget.Environment(cmd.Env, req.Cores)', 'cmd.Env = corebudget.Environment(cmd.Env, 0)', 4, 'budget_four'),
 ('internal/daemon/engine.go', 'Cores:      engine.deps.VerificationCores,', 'Cores:      0,', 5, 'budget_five'),
 ('internal/daemon/task_engine.go', 'probeCommandsForRun(ctx, engine.deps.Verifier, engine.deps.VerificationCores,', 'probeCommandsForRun(ctx, engine.deps.Verifier, 0,', 5, 'budget_five'),
 ('internal/daemon/agent_session_owner.go', 'req.Cores = engine.deps.VerificationCores', 'req.Cores = 0', 5, 'budget_five'),
 ('internal/agent/acpx_runner.go', 'return corebudget.Environment(nil, cores)', 'return corebudget.Environment(nil, 0)', 6, 'budget_three'),
 ('internal/cli/implement.go', 'cores:        verificationCoreBudget(ctx, loadedConfig.Config).Cores,', 'cores:        0,', 7, ''),
 ('internal/cli/deliver_workflow.go', 'Cores:      cores,', 'Cores:      0,', 8, ''),
 ('internal/cli/deliver_linux_lane.go', 'if request.Cores > 0 {', 'if false && request.Cores > 0 {', 9, ''),
 ('internal/cli/doctor_verification_cores.go', 'if total > int64(budget.LogicalCPUs) {', 'if false && total > int64(budget.LogicalCPUs) {', 10, 'undivided_with_three_Runs'),
]

def main():
    repository_env = os.environ.copy()
    repository_env.update(GIT_CONFIG_GLOBAL=os.devnull, GIT_CONFIG_NOSYSTEM='1')
    # No user Git hooks/configuration or provider credentials enter the temporary repositories.
    for name in ('GH_TOKEN', 'GITHUB_TOKEN', 'TYPESAFE_API_KEY', 'ROUNDFIX_OPENROUTER_API_KEY',
                 'ROUNDFIX_OPENROUTER_JUDGE_API_KEY', 'ROUNDFIX_TYPESAFE_API_KEY'):
        repository_env.pop(name, None)
    archive = subprocess.run(['git', 'archive', gate.HEAD], cwd=gate.ROOT,
                             capture_output=True, check=True).stdout
    for i, (filename, old, new, named_index, subtest) in enumerate(MUTATIONS, 1):
        row = f'M{i:02}'
        package, test = gate.NAMES[named_index - 1]
        selector = '^' + test + '$' + ('/^' + subtest + '$' if subtest else '')
        args = ['go', 'test', '-count=1', '-timeout', '20m', '-v', '-run', selector, './' + package]
        print(row + ': archived baseline', flush=True)
        with tempfile.TemporaryDirectory(prefix='qa0375-mutation-') as temp:
            root = Path(temp)
            with tarfile.open(fileobj=io.BytesIO(archive)) as contents:
                contents.extractall(root, filter='data')
            preparation = []
            for command in [
                ['git', 'init', '-q'],
                ['git', 'add', '.'],
                ['git', '-c', 'core.hooksPath=/dev/null', '-c', 'commit.gpgsign=false',
                 '-c', 'user.name=QA', '-c', 'user.email=qa@localhost', 'commit', '-qm', 'Audited head baseline'],
            ]:
                done = subprocess.run(command, cwd=root, env=repository_env, capture_output=True, text=True)
                preparation.append(shlex.join(command) + ': exit ' + str(done.returncode))
                if done.returncode:
                    raise RuntimeError(done.stderr)
            baseline = subprocess.run(args, cwd=root, env=repository_env, capture_output=True, text=True)
            source = root / filename
            text = source.read_text()
            assert text.count(old) == 1, (row, filename, text.count(old))
            source.write_text(text.replace(old, new, 1))
            mutation = subprocess.run(args, cwd=root, env=repository_env, capture_output=True, text=True)
            failed_test = test + ('/' + subtest if subtest else '')
            ok = (baseline.returncode == 0 and '--- PASS: ' + test in baseline.stdout
                  and mutation.returncode == 1 and '--- FAIL: ' + failed_test in mutation.stdout
                  and '[build failed]' not in mutation.stdout + mutation.stderr
                  and 'operation not permitted' not in mutation.stdout + mutation.stderr)
            log = ('source: git archive ' + gate.HEAD + '\n' + '\n'.join(preparation)
                   + '\nEach mutation used a distinct temporary repository, removed after both tests.\n'
                   + 'mutation file: ' + filename + '\nbefore: ' + old + '\nafter: ' + new
                   + '\ncommand: ' + shlex.join(args) + '\nBASELINE exit: ' + str(baseline.returncode)
                   + '\nstdout:\n' + baseline.stdout + '\nstderr:\n' + baseline.stderr
                   + '\nMUTATION exit: ' + str(mutation.returncode)
                   + '\nstdout:\n' + mutation.stdout + '\nstderr:\n' + mutation.stderr)
        (HERE / (row + '.txt')).write_text(gate.sanitize(log))
        gate.update(row, 'pass' if ok else 'fail',
                    'Inputs: `repository_path` = `' + str(gate.ROOT) + '`; `commit_range` = `'
                    + gate.RANGE + '` identifies the integrated Task scope; '
                    'the archive itself is exactly `' + gate.HEAD + '`.\n\n'
                    'TechSpec Testing Approach sabotage ' + str(i) + '. One line in `' + filename
                    + '` changed in a distinct temporary Git repository with a baseline commit before mutation. '
                    'Only `' + selector + '` ran, with `-timeout 20m`. Baseline exit '
                    + str(baseline.returncode) + '; mutant exit ' + str(mutation.returncode) + '. '
                    + ('The named test passed before mutation and rejected the mutant through the expected assertion.' if ok
                       else 'The required baseline-pass / named-assertion-failure pair was not established.')
                    + '\n\nEvidence: [mutation, commands and both outputs](evidence/qa-report-2026-11-11/' + row + '.txt).')
        print(row + ': ' + ('PASS' if ok else 'FAIL') + ', baseline ' + str(baseline.returncode)
              + ', mutant ' + str(mutation.returncode), flush=True)

if __name__ == '__main__':
    main()
