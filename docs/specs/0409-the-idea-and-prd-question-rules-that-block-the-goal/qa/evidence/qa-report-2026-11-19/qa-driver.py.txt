def run(cmd,cwd=ROOT,env=None):
    p=subprocess.run(cmd,cwd=cwd,env=env,stdin=subprocess.DEVNULL,stdout=subprocess.PIPE,stderr=subprocess.PIPE,text=True)
    log="$ "+" ".join(cmd)+"\nstdout:\n"+p.stdout+"\nstderr:\n"+p.stderr+"\nexit: "+str(p.returncode)+"\n"
    return p,log
