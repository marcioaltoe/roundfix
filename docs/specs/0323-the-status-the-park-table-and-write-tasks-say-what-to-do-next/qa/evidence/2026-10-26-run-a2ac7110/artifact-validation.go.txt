package spec

import (
 "io/fs"
 "os"
 "path/filepath"
 "testing"
 "roundfix/internal/secretscan"
)

func TestQA0323FinalArtifact(t *testing.T) {
 t.Parallel()
 dir := filepath.Join("..", "..", "docs", "specs", "0323-the-status-the-park-table-and-write-tasks-say-what-to-do-next")
 report, err := ReadQAReport(dir)
 if err != nil { t.Fatal(err) }
 if report.Verdict != VerdictPartial || report.RowsBlockedEnvironment != 2 || report.RowsBlockedPrePullRequest != 1 || report.RowsBlockedNetworkProhibited != 1 || report.RowsBlockedDeclared != 1 || report.RowsBlockedFinding != 0 || report.RowsSkipped != 0 || report.Hollow { t.Fatalf("unexpected report metadata: %+v", report) }
 if err := QAReportEligibility(dir, report); err != nil { t.Fatal(err) }
 t.Log("production parser: qualifying partial; one pre-PR row, one prohibited-network row, one declared acceptance")
 files := 0
 err = filepath.WalkDir(filepath.Join(dir, "qa"), func(path string, entry fs.DirEntry, err error) error {
  if err != nil { return err }; if entry.IsDir() { return nil }
  data, err := os.ReadFile(path); if err != nil { return err }
  for _, match := range secretscan.ScanContent(path, data) { t.Error(match.String()) }
  files++; return nil
 })
 if err != nil { t.Fatal(err) }
 t.Logf("production secret scan: %d QA text files checked", files)
}
