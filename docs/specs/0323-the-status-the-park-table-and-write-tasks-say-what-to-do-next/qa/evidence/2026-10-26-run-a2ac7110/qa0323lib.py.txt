import os, pathlib, subprocess, tempfile, json, re, shlex, hashlib, tarfile, io, traceback
S=json.loads(pathlib.Path('/tmp/qa0323-state.json').read_text()); ROOT=pathlib.Path(S['root']); TMP=pathlib.Path(S['tmp']); EV=pathlib.Path(S['evidence']); REPORT=pathlib.Path(S['report']); SD=ROOT/'docs/specs'/S['spec']
ENV=os.environ.copy()
for key in tuple(ENV):
 if key.endswith(('API_KEY','TOKEN','SECRET','PASSWORD')): ENV.pop(key)
ENV.update(GOPROXY='off',GOSUMDB='off',GOTOOLCHAIN='local',GOMODCACHE=S['modcache'],GIT_CONFIG_NOSYSTEM='1',GIT_CONFIG_GLOBAL='/dev/null',GIT_TERMINAL_PROMPT='0')
CACHE=subprocess.run(['go','env','GOCACHE'],env=ENV,text=True,capture_output=True,check=True).stdout.strip()
def isolated():
 e=ENV.copy(); e.update(HOME=tempfile.mkdtemp(dir=TMP,prefix='home-'),TMPDIR=tempfile.mkdtemp(dir=TMP,prefix='tmp-'),PATH=str(TMP/'gh-bin')+os.pathsep+ENV['PATH'],GOCACHE=CACHE); return e

def run(cmd,cwd=ROOT,e=None):
 p=subprocess.run(cmd,cwd=cwd,env=e or isolated(),text=True,capture_output=True)
 return p,'$ '+shlex.join(cmd)+'\nstdout:\n'+p.stdout+'\nstderr:\n'+p.stderr+'\nexit: '+str(p.returncode)+'\n'

def finish(rid,status,observation,log):
 (EV/(rid+'.txt')).write_text(log)
 s=REPORT.read_text(); s=s.replace('| '+rid+' | pending |','| '+rid+' | '+status+' |',1)
 start=s.index('### '+rid+'\n'); end=s.find('\n### ',start+1)
 if end<0:end=len(s)
 block=s[start:end].replace('Observation: pending.','Observation: '+observation+'\n\nEvidence: [transcript](evidence/2026-10-26-run-a2ac7110/'+rid+'.txt).',1)
 REPORT.write_text(s[:start]+block+s[end:]); print(rid,status,observation,flush=True)

def tests(rid,commands):
 log=''; ok=True
 for cmd in commands:
  p,out=run(cmd); log+=out; ok=ok and p.returncode==0
  if '-v' in cmd:
   print('\n'.join(x for x in p.stdout.splitlines() if '--- PASS' in x or '--- FAIL' in x),flush=True)
 finish(rid,'pass' if ok else 'fail','Every command exited 0.' if ok else 'A command failed; inspect transcript.',log)
