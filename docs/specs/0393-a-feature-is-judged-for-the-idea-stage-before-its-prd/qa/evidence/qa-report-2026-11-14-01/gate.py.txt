def test(q,args,names=()):
    print(q,' '.join(args),flush=True)
    # All suite output remains outside the checkout until the foreground process exits.
    with tempfile.TemporaryFile(mode='w+') as f:
        p=sp.run(args,cwd=ROOT,stdout=f,stderr=sp.STDOUT,env={**os.environ,'GOPROXY':'off','GOSUMDB':'off'})
        f.seek(0); out=f.read()
    result=sp.CompletedProcess(args,p.returncode,out,'')
    link=evidence(q,args,result)
    passed=re.findall(r'(?m)^\s*--- PASS: (\S+)',out)
    missing=[n for n in names if n not in passed]
    good=p.returncode==0 and not missing and not re.search(r'(?m)^\s*--- SKIP:',out)
    summary=f'Exit {p.returncode}; {len(passed)} named test/subtest passes; no skips. '
    if missing:summary+='Missing required passes: '+', '.join(missing)+'. '
    row(q,'pass' if good else 'fail',summary+f'[Full command and every test/subtest]({link}).')
    print(q,'pass' if good else 'FAIL',len(passed),'passes',flush=True)
    return good
