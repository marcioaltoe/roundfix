def collect_keys(x):
    if isinstance(x,dict):
        for k,v in x.items():
            if 'key' in k and isinstance(v,str) and re.fullmatch('[A-Z][A-Z0-9_]+',v): KEYS.add(v)
            collect_keys(v)
    elif isinstance(x,list):
        for v in x: collect_keys(v)

def record(name,args,p,cwd=ROOT,note=''):
    content='command: '+shlex.join([str(a) for a in args])+'\ncwd: '+str(cwd)+'\nexit: '+str(p.returncode)+'\nstdout:\n'+p.stdout+'\nstderr:\n'+p.stderr+'\n'+note+'\n'
    (EVID/(name+'.txt')).write_text(content)
    print(name, 'exit',p.returncode, note,flush=True)

def command(name,args,cwd=ROOT,env=None):

def test_row(row,name,pkg,names=None,pattern=None,tags=None):
    args=['go','test','-count=1','-timeout','20m' if names else '30m','-v']
    if names: args+=['-run','^('+'|'.join(names)+')$']
    elif pattern: args+=['-run',pattern]
    if tags: args+=['-tags',tags]
    args += [pkg]
    p=command(name,args)
    missing=[n for n in names or [] if not re.search(r'--- PASS: '+n+r' ',p.stdout)]
    good=p.returncode==0 and not missing
    passes=re.findall(r'^\s*--- PASS: (\S+)',p.stdout,re.M)
    skips=re.findall(r'^\s*--- SKIP: (\S+)',p.stdout,re.M)
    (EVID/(name+'-names.txt')).write_text('Passed tests and subtests:\n'+'\n'.join(passes)+'\nSkipped:\n'+'\n'.join(skips)+'\n')
    update(row,'pass' if good else 'fail',f'Command `{shlex.join(args)}` exited {p.returncode}; {len(passes)} passing test/subtest records. Every named test is present: {not missing}.     return good
