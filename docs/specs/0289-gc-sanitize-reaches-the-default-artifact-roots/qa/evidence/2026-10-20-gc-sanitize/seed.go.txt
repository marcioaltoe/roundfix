package store
import("context";"os";"path/filepath";"testing";"time";"encoding/json")
func TestQASeed(t *testing.T) {
 t.Parallel()
 home,repo:=os.Getenv("QA_FIXTURE_HOME"),os.Getenv("QA_FIXTURE_REPO")
 s,err:=Open(context.Background(),home);if err!=nil{t.Fatal(err)}
 values:=[]string{""};if os.Getenv("QA_FIXTURE_MODE")=="parity"{values=append(values,"relative-artifacts")}
 meta:=[]map[string]string{}
 for i:=0;i<len(values);i++ {
  value:=values[i]
  r,err:=s.CreateRun(context.Background(),CreateRunRequest{Kind:KindImplement,GitRoot:repo,ArtifactDir:value,LocalBranch:"fix/qa-seed",HeadSHA:"abc123",SpecSlug:"qa-seed"});if err!=nil{t.Fatal(err)}
  if _,err=s.CompleteRun(context.Background(),r.ID,StateClean);err!=nil{t.Fatal(err)}
  stamp:=time.Now().Add(-400*time.Hour).UTC().Format(time.RFC3339Nano)
  if _,err=s.db.Exec("UPDATE runs SET created_at=?,updated_at=?,completed_at=? WHERE id=?",stamp,stamp,stamp,r.ID);err!=nil{t.Fatal(err)}
  root,err:=ResolveRunArtifactRoot(ArtifactRootRun{ID:r.ID,GitRoot:repo,Repository:r.RepositoryRoot,ArtifactDir:value},home);if err!=nil{t.Fatal(err)}
  path:=filepath.Join(root,"runs",r.ID,"agent","batch-001.log");if err=os.MkdirAll(filepath.Dir(path),0755);err!=nil{t.Fatal(err)}
  if err=os.WriteFile(path,[]byte("batch log"),0644);err!=nil{t.Fatal(err)}
  meta=append(meta,map[string]string{"id":r.ID,"root":root,"path":path,"recorded":value})
  if i==0&&os.Getenv("QA_FIXTURE_MODE")=="parity"{values=append(values,root)}
 }
 if err=s.Close();err!=nil{t.Fatal(err)}
 b,err:=json.MarshalIndent(meta,"","  ");if err!=nil{t.Fatal(err)}
 if err=os.WriteFile(os.Getenv("QA_FIXTURE_META"),b,0644);err!=nil{t.Fatal(err)}
}
