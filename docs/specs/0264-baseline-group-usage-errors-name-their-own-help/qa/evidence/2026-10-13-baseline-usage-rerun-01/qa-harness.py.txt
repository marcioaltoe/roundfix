def paths(helptext):
 out=[]
 def add(words):
  for i in range(1,len(words)+1):
   p=' '.join(words[:i])
   if p not in out: out.append(p)
 for line in helptext.splitlines():
  if not line.startswith('  roundfix '): continue
  words=[]
  for token in line[len('  roundfix '):].split():
   if token.startswith(('-','[','(','<')):
    if words and token.startswith('<') and token.endswith('>') and '|' in token:
     for alternative in token[1:-1].split('|'): add(words+[alternative])
     words=[]
    break
   words.append(token)
  add(words)
 return out
