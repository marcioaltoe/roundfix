def save(name, text):
    path = EV / name
    assert subprocess.run(['git', 'check-ignore', '-q', str(path)], cwd=ROOT).returncode == 1
    path.write_text(text)

def row(number, status):
    s = REPORT.read_text()
    s, n = re.subn(r'(?m)^(\| Q%02d \| )[^|]+(\|)' % number, lambda m: m[1] + status + ' ' + m[2], s)
    assert n == 1 or number in (6, 8, 9, 12)
    REPORT.write_text(s)

def run(name, command, cwd=ROOT, environ=None, expected=0):
    print('RUN', name, flush=True)
    p = subprocess.run(command, cwd=cwd, env=environ, text=True, capture_output=True)
    save(name + '.txt', 'command: ' + (' '.join(command)) + '\nexit: ' + str(p.returncode) + '\nstdout:\n' + p.stdout + '\nstderr:\n' + p.stderr)
    print('EXIT', name, p.returncode, flush=True)
    if expected is not None:
        assert p.returncode == expected, name + ': ' + p.stderr[-2000:] + p.stdout[-2000:]
    return p

@contextlib.contextmanager
def cli_env():
    # Read only the cache paths needed to preserve Go caches; never log values.
    e = dict(os.environ)
    for key in KEYS:
        e.pop(key, None)
    for key in ['GOCACHE', 'GOMODCACHE']:
        e[key] = subprocess.check_output(['go', 'env', key], text=True).strip()
    with tempfile.TemporaryDirectory() as h, tempfile.TemporaryDirectory() as t:
        e.update(HOME=h, TMPDIR=t)
        yield e

def fixture():
    tech = (SPEC / '_techspec.md').read_text().split('## Testing Approach\n')[1]
    escaped = re.search(r'```text\n(.*?)\n```', tech, re.S)[1]
    content = ast.literal_eval('"' + escaped.replace('"', '\\"') + '"')
    declaration = '## Unreachable Acceptance\n\n- criterion: the probe reads the live service\n  reason: needs a deploy\n  satisfied-by: the operator after release\n'
    save('fixture-report.txt', content)
    save('fixture-prd.txt', declaration)
    with tempfile.TemporaryDirectory() as d:
        work = Path(d)
        assert subprocess.run(['git', 'rev-parse', '--is-inside-work-tree'], cwd=work, capture_output=True).returncode != 0
        path = work / 'docs/specs/9001-fixture/qa/qa-report-2026-11-10.md'
        path.parent.mkdir(parents=True)
        (path.parent.parent / '_prd.md').write_text(declaration)
        path.write_text(content)
        with cli_env() as e:
            p = run('cli-broken-report', [str(ROOT/'bin/roundfix'), 'qa-report', 'accept', str(path)], cwd=work, environ=e, expected=1)
        want = f'roundfix: QA Report eligibility refused: QA Report {path}: Results row at line 16 is outside the Results table that starts at line 12; remove the blank line at {path}:15 so the row continues the table\n'
        assert p.stdout == '' and p.stderr == want

        assert lines[14] == '\n'

        with cli_env() as e:
            p = run('cli-repaired-report', [str(ROOT/'bin/roundfix'), 'qa-report', 'accept', str(path)], cwd=work, environ=e)
        assert p.stdout == '' and p.stderr == ''
        # A fresh process independently rereads the persisted repair.
        with cli_env() as e:
            p = run('cli-repaired-report-restart', [str(ROOT/'bin/roundfix'), 'qa-report', 'accept', str(path)], cwd=work, environ=e)

         'if readErr == nil && report.ResultsBreak != nil {', 'if false && readErr == nil && report.ResultsBreak != nil {'),
        ('no-seeded-prompt', 'internal/daemon/task_engine.go', './internal/daemon', 'TestQAPromptTellsTheGateToContinueTheSeededTable', None, None),
    ]
    summaries = []
    survivors = []
    for number, (name, file, package, test, before, after) in enumerate(cases, 1):
        with tempfile.TemporaryDirectory() as d:
            copy = Path(d)
            archive = copy / 'head.tar'
            with archive.open('wb') as out:
                subprocess.run(['git', 'archive', HEAD], cwd=ROOT, stdout=out, check=True)
            with tarfile.open(archive) as tf:
                tf.extractall(copy, filter='data')
            archive.unlink()
            # The repository's suite guard inventories even an archive copy via Git.
            subprocess.run(['git', 'init', '-q', str(copy)], check=True, capture_output=True)
            path = copy / file
            source = path.read_text()
            if before is None:
                line = next(line for line in source.splitlines(keepends=True) if 'prompt += "Add each row of the matrix to the seeded Results table' in line)
                before, after = line, ''
            assert source.count(before) == 1, name
            path.write_text(source.replace(before, after, 1))
            p = run('sabotage-%02d-%s' % (number, name), ['go', 'test', '-count=1', '-timeout', '20m', '-v', '-run', '^'+test+'$', package], cwd=copy, expected=None)
            assert '[build failed]' not in p.stdout and 'setup failed' not in p.stdout, name
            if p.returncode == 1 and '--- FAIL: '+test in p.stdout:
                result='named test failed as required'
            else:
                survivors.append(name)
                result='mutation survived: expected named failure, observed exit '+str(p.returncode)
            summaries.append(f'{number}. {name}: {result}; test {test}; source mutation confined to {file}; fresh git archive of {HEAD}; copy removed.')
    save('sabotage-summary.txt', '\n'.join(summaries)+'\n')
    row(4, 'fail' if survivors else 'pass')

def scope():
    def git(*args):
        return subprocess.check_output(['git',*args],cwd=ROOT,text=True)
    commits = [('task_01','71b2ec2872ff4a1969f068ecb4f5921a524000e0'),('task_03','4d1ed9ebe9781958abafcee3c475cb6ef2b3fac9'),('task_02',HEAD)]
    assert git('rev-parse','HEAD').strip()==HEAD
    audit=[]
    all_paths=set()
    for task, commit in commits:
        content=git('show',commit+':docs/specs/'+SLUG+'/'+task+'.md')
        declared=set(re.findall(r'^- (?:instruction|interface|creates|deletes): `([^`]+)`',content,re.M))
        if '## Recorded paths' in content:
            declared.update(re.findall(r'`([^`]+)`',content.split('## Recorded paths')[1].split('\n## ')[0]))
        declared.add('docs/specs/'+SLUG+'/'+task+'.md')
        paths=git('diff-tree','--no-commit-id','--name-only','-r',commit).splitlines()
        assert set(paths)<=declared,(task,set(paths)-declared)
        all_paths.update(paths)
        audit.append(task+' '+commit+'\n'+ '\n'.join('  declared: '+p for p in paths))
    auth=(SPEC/'_authorization.md').read_text()
    grant=set(re.findall(r'^  - (.+)$',auth.split('paths:\n')[1].split('operations:')[0],re.M))
    governed_source=(ROOT/'internal/speccheck/governed.go').read_text()
    patterns=re.findall(r'regexp.MustCompile\(\s*`([^`]+)`',governed_source)
    exact_section=governed_source.split('paths: exactGovernedPaths(')[1].split('\n\t\t),')[0]
    exact=set(re.findall(r'"([^"]+)"',exact_section))
    governed={p for p in all_paths if p in exact or any(re.search(pattern,p) for pattern in patterns)}
    assert governed<=grant,(governed-grant)
    assert len(governed)==6,governed
    for task,commit in commits:
        if any(p in governed for p in git('diff-tree','--no-commit-id','--name-only','-r',commit).splitlines()):
            parent_auth=git('show',commit+'^:docs/specs/'+SLUG+'/_authorization.md')
            base_auth=git('show',BASE+':docs/specs/'+SLUG+'/_authorization.md')
            assert parent_auth==base_auth==auth
    prohibited={'Makefile','go.mod','go.sum','.roundfixrc.yml','skills-lock.json','CHANGELOG.md','internal/app/version.go','dist/npm/roundfix/package.json','internal/speccheck/report.go'}
    forbidden_prefixes=['.github/workflows/','internal/agent/','internal/baseline/','docs/agents/','docs/adr/']
    for path in all_paths:
        assert path not in prohibited and not any(path.startswith(x) for x in forbidden_prefixes),path
        assert not (path.startswith('internal/cli/') and not path.endswith('_test.go')),path
        if path.startswith(('.agents/skills/','skills/')) and path!='skills/testdata/owned-skill-versions.json':
            assert any(path.startswith(prefix+skill+'/') for prefix in ['.agents/skills/','skills/'] for skill in ['roundfix','qa-gate']),path
        if path.endswith('.md') and path.startswith(('.agents/skills/','skills/')):
            before=git('show',BASE+':'+path)
            after=git('show',HEAD+':'+path)
            def settlement(s):
                m=re.search(r'(?ms)^### QA settlement\n.*?(?=^#{1,3} |\Z)',s)
                return m[0] if m else None
            assert settlement(before)==settlement(after),path
    assert set(git('diff','--name-only',BASE+'..'+HEAD).splitlines())==all_paths
    audit.append('Governed paths, all expressly authorized before task_03:\n'+'\n'.join(sorted(governed)))
    audit.append('No prohibited path changed; all QA settlement sections byte-identical; no authorization was folded into a Task commit.')
    save('scope-audit.txt','commit_range: '+BASE+'..'+HEAD+'\n\n'+'\n\n'.join(audit)+'\n')
    row(10,'pass')
    before=git('show',BASE+':CONTEXT.md'); after=(ROOT/'CONTEXT.md').read_text()
    for file in ['_prd.md','_techspec.md']:
        assert '- changes: **QA Report**' in (SPEC/file).read_text()
