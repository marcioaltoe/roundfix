def command(args, label, cwd=ROOT, expected=0, child_env=None):
    p = subprocess.run(args, cwd=cwd, env=child_env, text=True, capture_output=True)
    text = 'Command: ' + shlex.join(args) + '\nExit: ' + str(p.returncode) + '\nstdout:\n' + p.stdout + 'stderr:\n' + p.stderr
    (EVIDENCE / (label + '.txt')).write_text(text)
    print(label + ': exit ' + str(p.returncode), flush=True)
    if expected is not None:
        assert p.returncode == expected, label + ' unexpected exit; see retained output'
    return p

def mark(row, status, detail):
    text = REPORT.read_text()
    text, count = re.subn(r'(?m)^\| ' + row + r' \| pending \|', '| ' + row + ' | ' + status + ' |', text)
    assert count == 1, row
    start = text.index('### ' + row + '\n')
    end = text.find('\n### ', start + 5)
    if end < 0:
        end = text.find('\n## ', start + 5)
    if end < 0:
        end = len(text)
    text = text[:end] + '\nObserved: ' + detail + '\n' + text[end:]
    REPORT.write_text(text)

def shell_verification(task):
    text = (SPEC / (task + '.md')).read_text()
    committed = subprocess.check_output(['git', 'show', HEAD + ':' + str((SPEC / (task + '.md')).relative_to(ROOT))], text=True)
    assert text == committed, 'Verification source changed since audited head'
    section = text.split('## Verification\n', 1)[1].split('\n## ', 1)[0]
    commands = re.findall(r'^- `(.+?)` — expected:', section, re.M)
    assert commands, task
    return commands

Evidence: [named tests](evidence/qa-report-2026-11-10-01/named-tests.txt), [pass inventory](evidence/qa-report-2026-11-10-01/named-pass-summary.txt), [whole packages](evidence/qa-report-2026-11-10-01/whole-spec-speccheck-skills.txt), [documentation contracts](evidence/qa-report-2026-11-10-01/whole-docscontract.txt).')

def replace_once(root, path, old, new):
    file = root / path
    text = file.read_text()
    assert text.count(old) == 1, path + ' mutation not unique: ' + str(text.count(old))
    file.write_text(text.replace(old, new, 1))

def audit():
    lines = ['Audited head: ' + HEAD, 'Task range: ' + BASE + '..' + HEAD]
    commits = [('task_01', 'd41268d30f0dfe1ce64d5b1e8a02f0bf6de10b1d'),
               ('task_03', '327b8d807cc8a4c6e5d7cd4ae9764ef057b30de3'),
               ('task_02', HEAD)]
    grant_path = str((SPEC / '_authorization.md').relative_to(ROOT))
    grant = (SPEC / '_authorization.md').read_text()
    allowed_governed = set(re.findall(r'^  - (.+)$', grant.split('paths:\n', 1)[1].split('operations:', 1)[0], re.M))
    governed_source = (ROOT / 'internal/speccheck/governed.go').read_text()
    governed_patterns = [re.compile(x) for x in re.findall(r'regexp.MustCompile\(\s*`([^`]+)`', governed_source)]
    historical = set(re.findall(r'"([^"]+)"', governed_source.split('paths: exactGovernedPaths(', 1)[1].split('),', 1)[0]))
    all_paths = set()
    for task, commit in commits:
        paths = subprocess.check_output(['git', 'diff-tree', '--no-commit-id', '--name-only', '-r', commit], text=True).splitlines()
        task_path = str((SPEC / (task + '.md')).relative_to(ROOT))
        text = subprocess.check_output(['git', 'show', commit + ':' + task_path], text=True)
        declared = set(re.findall(r'^- (?:interface|creates|instruction|deletes): `([^`]+)`', text, re.M)) | {task_path}
        if '## Recorded paths' in text:
            declared.update(re.findall(r'`([^`]+)`', text.split('## Recorded paths', 1)[1].split('\n## ', 1)[0]))
        assert set(paths) <= declared, (task, set(paths) - declared)
        lines += ['', 'Command: git diff-tree --no-commit-id --name-only -r ' + commit, task + ': all changed paths declared'] + paths
        all_paths.update(paths)
    entire = set(subprocess.check_output(['git', 'diff', '--name-only', BASE + '..' + HEAD], text=True).splitlines())
    assert all_paths == entire, 'Task union differs from range'
    governed = {p for p in entire if p in historical or any(pattern.search(p) for pattern in governed_patterns)}
    assert governed == allowed_governed, (governed - allowed_governed, allowed_governed - governed)
    forbidden_exact = {'Makefile', 'go.mod', 'go.sum', '.roundfixrc.yml', 'skills-lock.json', 'CHANGELOG.md', 'internal/app/version.go', 'dist/npm/roundfix/package.json', 'internal/speccheck/report.go'}
    for path in entire:
        assert path not in forbidden_exact, path
        assert not path.startswith(('.github/workflows/', 'internal/agent/', 'internal/baseline/', 'docs/agents/', 'docs/adr/')), path
        assert not (path.startswith('internal/cli/') and not path.endswith('_test.go')), path
        if path.startswith(('.agents/skills/', 'skills/')):
            assert path.startswith(('.agents/skills/roundfix/', '.agents/skills/qa-gate/', 'skills/roundfix/', 'skills/qa-gate/')) or path == 'skills/testdata/owned-skill-versions.json', path
    def section(text):
        return re.search(r'(?ms)^### QA settlement\n.*?(?=^#{1,3} |\Z)', text).group(0)
    for path in ['.agents/skills/qa-gate/SKILL.md', 'skills/qa-gate/SKILL.md', '.agents/skills/roundfix/SKILL.md', 'skills/roundfix/SKILL.md']:
        old = subprocess.check_output(['git', 'show', BASE + ':' + path], text=True)
        assert section(old) == section((ROOT / path).read_text()), 'settlement changed: ' + path
    for revision in [BASE, '327b8d807cc8a4c6e5d7cd4ae9764ef057b30de3^', HEAD]:
        assert subprocess.check_output(['git', 'show', revision + ':' + grant_path], text=True) == grant
    subprocess.run(['git', 'merge-base', '--is-ancestor', '1febafab745e13e3e0b5403911293be113755733', '327b8d807cc8a4c6e5d7cd4ae9764ef057b30de3^'], check=True)
    lines += ['', 'Governed classification from all regexp.MustCompile patterns and exact historical set in internal/speccheck/governed.go:', *sorted(governed), 'All six listed in approved authorization, unchanged at base, Task 03 parent and head.', 'Grant revision 1febafab precedes Task 03. No grant folded into a Task.', 'No forbidden path changed; all four QA settlement sections byte-identical.']
    (EVIDENCE / 'scope-audit.txt').write_text('\n'.join(lines) + '\n')
    mark('Q10', 'pass', 'Per-commit diff-tree paths exactly fit each Task declaration plus its own Task file. Their union equals the Task range. All six changed Governed Paths are authorized; the grant predates Task 03 and is unchanged. No forbidden path, other skill or QA settlement section changed. Evidence: [scope and ancestry audit](evidence/qa-report-2026-11-10-01/scope-audit.txt).')

def archive_and_controls():

    with tempfile.TemporaryDirectory(prefix='qa-home-') as home, tempfile.TemporaryDirectory(prefix='qa-tmp-') as tmp:
        child = os.environ.copy()
        child.update(HOME=home, TMPDIR=tmp, **caches)
