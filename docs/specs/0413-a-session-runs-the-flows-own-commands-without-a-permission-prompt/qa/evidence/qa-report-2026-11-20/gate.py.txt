def host_env():

    for k in KEYS: e.pop(k,None)

def run(args, cwd=ROOT, env=None):
    return subprocess.run([str(x) for x in args],cwd=cwd,env=env or host_env(),stdin=subprocess.DEVNULL,stdout=subprocess.PIPE,stderr=subprocess.PIPE,text=True)

def transcript(args,p):
    words=[str(x) for x in args]
    if words and words[0]==str(BINARY): words[0]='./bin/roundfix'
    return '$ '+shlex.join(words)+'\nstdout:\n'+p.stdout+'\nstderr:\n'+p.stderr+'\nexit: '+str(p.returncode)+'\n'

def save(name,content):
    p=EVIDENCE/name
    ck=run(['git','check-ignore','--',str(p.relative_to(ROOT))])
    assert ck.returncode==1, 'ignored evidence '+name
    p.write_text(content)
    return p

def record(row,status,detail):
    s=REPORT.read_text()
    s,n=re.subn(r'(?m)^\| '+re.escape(row)+r' \| [^|]+ \|',lambda m:'| '+row+' | '+status+' |',s)
    assert n==1,(row,n)
    tag='### '+row+' evidence\n'
    start=s.index(tag)+len(tag)
    end=s.find('\n### ', start)
    if end<0: end=s.find('\n## ',start)
    if end<0: end=len(s)
    part=s[start:end]
    if '\nObservation: pending.' in part: part=part.replace('\nObservation: pending.','\n'+detail)
    else: part+='\n'+detail+'\n'
    REPORT.write_text(s[:start]+part+s[end:])
    print(row+': '+status,flush=True)

@contextlib.contextmanager
def product_env():
    global TOOLCHAINS
    if not TOOLCHAINS:
        for k in ('GOMODCACHE','GOCACHE'):
            p=run(['go','env',k]); assert p.returncode==0; TOOLCHAINS[k]=p.stdout.strip()
        TOOLCHAINS['RUSTUP_HOME']=os.environ.get('RUSTUP_HOME',os.path.join(os.environ['HOME'],'.rustup'))
        TOOLCHAINS['CARGO_HOME']=os.environ.get('CARGO_HOME',os.path.join(os.environ['HOME'],'.cargo'))
    with tempfile.TemporaryDirectory(prefix='qa0413-',dir='/private/tmp') as td:
        t=pathlib.Path(td); h=t/'home'; tmp=t/'tmp'; fake=t/'fakebin'
        for d in (h,tmp,fake): d.mkdir()
        gh=fake/'gh'; gh.write_text('#!/bin/sh\nexit 1\n'); gh.chmod(0o755)
        e=host_env(); e.update(TOOLCHAINS); e['HOME']=str(h); e['TMPDIR']=str(tmp); e['PATH']=str(fake)+os.pathsep+e['PATH']; e['GIT_CONFIG_NOSYSTEM']='1'; e['GIT_CONFIG_GLOBAL']=os.devnull
        try: yield t,e
        finally:
            for p in t.rglob('*'):
                if not p.is_symlink():
                    try: p.chmod(p.stat().st_mode | 0o700)
                    except FileNotFoundError: pass

    return path

def test_commands(row,commands):
    lines=[]; failed=[]
    for args in commands:
        p=run(args); lines.append(transcript(args,p)); print(row,shlex.join(args),'exit',p.returncode,flush=True)
        if p.returncode: failed.append(shlex.join(args))
    save(row+'-tests.txt','\n'.join(lines))
    record(row,'fail' if failed else 'pass','Observed: '+('failed commands: '+', '.join(failed) if failed else 'every listed command exited 0.')+' Full stdout, stderr, exits and named subtests: ['+row+' tests](evidence/qa-report-2026-11-20/'+row+'-tests.txt).')
    return not failed

def flow(row,fn):
    logs=[]; artifacts={}
    try:
        with product_env() as (t,e):
            # A separate temporary home contains user-owned files as a non-mutation canary.
            home=pathlib.Path(e['HOME'])
            (home/'.claude').mkdir(); (home/'.claude/settings.json').write_text('{"model":"home-only"}\n')
            (home/'.codex/rules').mkdir(parents=True); (home/'.codex/rules/default.rules').write_text('# home-only\n')
            initial={str(p.relative_to(home)):p.read_bytes() for p in home.rglob('*') if p.is_file()}
            fn(t,e,logs,artifacts)
            final={str(p.relative_to(home)):p.read_bytes() for p in home.rglob('*') if p.is_file()}
            assert final==initial,'temporary home was changed'
            logs.append('Independent temporary-home snapshot unchanged.\n')
        status='pass'; observation='all expected outputs, independent file/state checks and exits matched.'
    except Exception as ex:
        logs.append(traceback.format_exc()); status='fail'; observation=str(ex)
    for name,content in artifacts.items():
        save(name,content.decode() if isinstance(content,bytes) else content)
    save(row+'-flow.txt','\n'.join(logs))
    record(row,status,'Observed: '+observation+' [Full flow transcript](evidence/qa-report-2026-11-20/'+row+'-flow.txt).')

def call(args,repo,e,logs,expected=0):

def plan(h,repo,e,logs,remove=False):
    args=['permissions','--harness',h,'--format','json']+(['--remove'] if remove else [])
    p=call(args,repo,e,logs)
    j=json.loads(p.stdout); assert j['schemaVersion']=='roundfix/flow-permissions/v1'
    return j['harnesses'][0],p.stdout

def match_transcript(pattern,actual):
    regex=''
    for line in pattern.splitlines():
        if line=='...':regex+='(?:[^\n]*\n)*'
        else:
            parts=re.split(r'(<[^>]+>)',line)
            regex+=''.join('[^\n]+' if x.startswith('<') and x.endswith('>') else re.escape(x) for x in parts)+'\n'
    assert re.fullmatch(regex,actual), 'transcript mismatch\nexpected:\n'+pattern+'\nactual:\n'+actual

def surface(n):
    def work(t,e,logs,artifacts):
        repo=make_repo(t/'repo',e)
        blocks=re.findall(r'```transcript\n(.*?)```',(SPEC/'_techspec.md').read_text(),re.S)
        block=blocks[n-1]
        command,rest=block.split('\n   stdout:\n') if '\n   stdout:\n' in block else block.split('\nstdout:\n')
        # Markdown indentation belongs to the fence, not the command's output.
        lines=block.splitlines(); normalized='\n'.join(x[3:] if x.startswith('   ') else x for x in lines)+'\n'
        command,rest=normalized.split('\nstdout:\n',1)
        expected_out,rest=rest.split('stderr:\n',1)
        expected_err,exitstr=rest.rsplit('exit: ',1)
        args=shlex.split(command.removeprefix('$ '))[1:]
        p=call(args,repo,e,logs,int(exitstr.strip()))
        match_transcript(expected_out,p.stdout); match_transcript(expected_err,p.stderr)
        assert not (repo/'.claude').exists() and not (repo/'.codex').exists()
        logs.append('Surface Transcript '+str(n)+': stdout and stderr independently matched, exact exit. Only version/digest placeholders and explicit ellipses varied.\n')
    flow('Q04'+str(n),work)

def claude_flow(t,e,logs,artifacts):
    repo=make_repo(t/'repo',e); target=repo/'.claude/settings.json';target.parent.mkdir()
    before=b'{\n\t"permissions": {\n\t\t"allow": [\n\t\t\t"Bash(make test)"\n\t\t],\n\t\t"ask": [\n\t\t\t"Bash(rm *)"\n\t\t]\n\t},\n\t"model": "qa-model"\n}'
    target.write_bytes(before);(t/'before.json').write_bytes(before);artifacts['claude-before.json']=before
    j,raw=plan('claude',repo,e,logs);artifacts['claude-preview.json']=raw
    assert j['state']=='not_applied' and re.fullmatch('[a-f0-9]{12}',j['digest']) and j['digest'] in j['next']
    assert target.read_bytes()==before
    n,m=len(j['add']),len(j['addGuards']);logs.append('Preview measured add='+str(n)+', addGuards='+str(m)+'; no counts fixed in the probe.\n')
    p=call(['permissions','--harness','claude','--apply',j['digest']],repo,e,logs)
    assert 'Wrote %d entries and %d Guard Entries to .claude/settings.json.'%(n,m) in p.stdout
    applied=target.read_bytes();artifacts['claude-applied.json']=applied;obj=json.loads(applied)
    assert obj['permissions']['allow']==['Bash(make test)']+j['add']
    assert obj['permissions']['ask']==['Bash(rm *)']+j['addGuards']
    assert obj['model']=='qa-model' and not applied.endswith(b'\n')
    for line in applied.splitlines():
        if line.strip().startswith(b'"') or line.strip() in (b'],',b']',b'},'):
            assert line.startswith(b'\t'),'tab indentation lost'
    again,_=plan('claude',repo,e,logs);assert again['state']=='current' and again['digest']=='' and again['next']==''
    p=call(['permissions','--harness','claude'],repo,e,logs);assert 'Nothing to write: the file holds the installed list.' in p.stdout
    p=call(['permissions','--harness','claude','--remove'],repo,e,logs)
    remove,_=plan('claude',repo,e,logs,True)
    assert all(x in p.stdout for x in j['add']+j['addGuards'])
    assert remove['digest'] in p.stdout and len(remove['remove'])==n+m
    call(['permissions','--harness','claude','--remove','--apply',remove['digest']],repo,e,logs)
    assert target.read_bytes()==(t/'before.json').read_bytes(),'removal did not restore first bytes'
    artifacts['claude-restored.json']=target.read_bytes()
    stale,_=plan('claude',repo,e,logs)
    changed=before.replace(b'qa-model',b'qb-model');target.write_bytes(changed)
    p=call(['permissions','--harness','claude','--apply',stale['digest']],repo,e,logs,3)
    assert not p.stdout and p.stderr=='The preview changed since digest '+stale['digest']+' was printed. Nothing was written.\nRun the preview again: roundfix permissions --harness claude\n'
    assert target.read_bytes()==changed
    for args in (['permissions','--apply','x'],['permissions','--harness','opencode','--apply','x'],['permissions','--harness','vim'],['permissions','--format','yaml']):call(args,repo,e,logs,2)
    outside=t/'not-git';outside.mkdir();call(['permissions','--harness','claude'],outside,e,logs,2)
    logs.append('Strict JSON and exact lists confirmed; tabs and no final newline preserved; apply/removal restored the byte snapshot; stale/invalid commands preserved the changed file.\n')
