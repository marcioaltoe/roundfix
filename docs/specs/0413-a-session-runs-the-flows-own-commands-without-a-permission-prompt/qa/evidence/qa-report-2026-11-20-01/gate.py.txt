def match_transcript(index,p):
 text=(SPEC/'_techspec.md').read_text()
 blocks=re.findall(r'```transcript\n(.*?)\n   ```',text,re.S)
 block='\n'.join(x[3:] if x.startswith('   ') else x for x in blocks[index].splitlines())
 out=block.split('stdout:\n')[1].split('stderr:')[0].rstrip('\n')
 err=block.split('stderr:')[1].split('exit:')[0].strip('\n')
 def match(want,got):
  lines=want.splitlines();chunks=[]
  for line in lines:
   if line=='...':chunks.append(r'(?:[^\n]*\n)*');continue
   escaped=re.escape(line)
   escaped=re.sub(r'<[^<>\n]+>',r'[^\n]+',escaped)
   chunks.append(escaped+'\n')
  assert re.fullmatch(''.join(chunks),got if got.endswith('\n') or not got else got+'\n'),(want,got)
 match(out,p.stdout);match(err,p.stderr)
 assert p.returncode==int(block.split('exit:')[1].strip())
 note('Transcript matching: separate stdout/stderr; exit exact; ... and <placeholder> rules satisfied.')
