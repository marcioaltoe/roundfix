def run(args,cwd=ROOT,env=None):
    print('RUN', ' '.join(map(str,args)),flush=True)
    p=subprocess.run(args,cwd=cwd,env=env or ENV,stdout=subprocess.PIPE,stderr=subprocess.PIPE,text=True)
    print('EXIT',p.returncode,flush=True)
    return p

def keep(name,p,args):
    text='command: '+' '.join(map(str,args))+'\nstdout:\n'+p.stdout+'\nstderr:\n'+p.stderr+'\nexit: '+str(p.returncode)+'\n'
    # Host paths are not needed to reproduce fixture results.
    for key in ['GOCACHE','GOMODCACHE','TMPDIR']:
        value=os.environ.get(key)
        if value: text=text.replace(value,'<session '+key+'>')
    (E/name).write_text(text)

def note(row,text,status=None):
    s=REPORT.read_text(); marker='### '+row+'\n'
    start=s.index(marker)+len(marker); nxt=s.find('\n### ',start)
    if nxt<0: nxt=len(s)
    s=s[:nxt]+'\n'+text+'\n'+s[nxt:]
    if status:
        s=re.sub(r'^\| '+re.escape(row)+r' \| pending \|',lambda _: '| '+row+' | '+status+' |',s,flags=re.M)
    REPORT.write_text(s)

def git(*args,cwd=ROOT):
    p=run(['git',*args],cwd); assert p.returncode==0,p.stderr
    return p.stdout

def unpack(rev,dir):
    b=subprocess.check_output(['git','archive',rev],cwd=ROOT)
    with tarfile.open(fileobj=io.BytesIO(b)) as tar: tar.extractall(dir,filter='data')

def init_fixture(d):
    git('init','-q',cwd=d); git('add','-A',cwd=d)

def sabotage():
    original=(ROOT/'Makefile').read_text()
    first='[ -d "$$cache" ] && probe=$$(mktemp "$$cache/.write-test.XXXXXX" 2>/dev/null) && rm -f "$$probe" 2>/dev/null'
    mutations=[('a','[ -d "$$cache" ]','mkdir -p "$$cache" 2>/dev/null',['set_and_not_mounted']),('b','printf 2048)','printf 10240)',CASES[1:4]),('c',"printf 'gocache: warning:"+original.split("printf 'gocache: warning:",1)[1].split('; \\\n',1)[0],':',CASES[1:4]),('d','GOCACHE := $(GOCACHE_FROM_GO_ENV)\nGOCACHE_BOUND_MB ?= 10240','GOCACHE := $(GOCACHE_FROM_GO_ENV)\nGOCACHE_BOUND_MB ?= 0',['set_and_writable']),('e',first,'[ -d "$$cache" ]',['set_and_read-only']),('f','ifeq ($(origin GOCACHE),undefined)','ifneq ($(origin GOCACHE),command line)',['environment_wins'])]
    for letter,old,new,expected in mutations:
        row='Q06'+letter
        assert old in original,(letter,old)
        with tempfile.TemporaryDirectory(prefix='roundfix-qa-sabotage-'+letter+'-') as tmp:
            d=pathlib.Path(tmp);unpack(HEAD,d);init_fixture(d)
            (d/'Makefile').write_text(original.replace(old,new,1))
            changed=git('diff','--name-only',cwd=d).splitlines();assert changed==['Makefile'],changed
            patch=git('diff','--','Makefile',cwd=d)
            args=['go','test','-count=1','-timeout','20m','-v','-run','^'+TEST+'$','./internal/verifyselect']
            p=run(args,cwd=d);keep('sabotage-'+letter+'.txt',p,args)
            (E/('sabotage-'+letter+'.patch.txt')).write_text(patch)
            out=p.stdout+p.stderr
            ok=p.returncode!=0 and 'panic:' not in out and all('--- FAIL: '+TEST+'/'+case+' ' in out for case in expected)
            status='pass' if ok else 'fail'
            note(row,'Observed: mutant test exit '+str(p.returncode)+'; required failures: '+', '.join(expected)+'. Only Makefile changed in this independent archive. Transcript inspected for the expected assertion failures and absence of panic. [Transcript](evidence/qa-report-2026-11-13/sabotage-'+letter+'.txt), [mutation](evidence/qa-report-2026-11-13/sabotage-'+letter+'.patch.txt).',status)
            print(row,status,flush=True)
