writer='internal/baseline/owned_skills_lock.go'

remove='''\t\tif ref, ok := values["ref"]; ok {
\t\t\tedits = append(edits, replacement{offset + ref.removeStart, offset + ref.removeEnd, nil})
\t\t\tupdate.Unpinned = append(update.Unpinned, name)
\t\t}
'''
def replace_once(s,a,b):
 assert s.count(a)==1,(a,s.count(a))
 return s.replace(a,b,1)
def mutate(root,n):
 p=root/writer;s=p.read_text()
 if n==1:s=replace_once(s,remove,'')
 if n==2:s=replace_once(s,'first.removeEnd = keyStart','first.removeEnd = first.end')
 if n==3:
  start=s.index('\t\tentry := entries[name]')
  end=s.index('\t\tencoded, err := json.Marshal(digest)',start)
  block=s[start:end];s=s[:start]+s[end:]
  anchor='\t\tfiles, exists, err := inspectRestoreTarget(root, ".agents/skills/"+name)'
  assert s.count(anchor)==2
  s=s.replace(anchor,block+anchor,1)
 if n==4:s=replace_once(s,'if !pinned && !hash {','if !pinned {')
 if n==5:
  at=s.index('func StaleOwnedLockEntries(')
  prefix=s[:at];tail=s[at:]
  tail=replace_once(tail,'sorted := append([]string(nil), names...)','''var all struct { Skills map[string]json.RawMessage `json:"skills"` }
 _ = json.Unmarshal(document.before, &all)
 sorted := make([]string, 0, len(all.Skills))
 for name := range all.Skills { sorted = append(sorted, name) }''')
  s=prefix+tail
 if n<=5:p.write_text(s)
 if n in (6,7):
  p=root/doctor;s=p.read_text()
  if n==6:s=replace_once(s,'result.Status = CheckStatusWarn','result.Status = CheckStatusOK')
  if n==7:
   s=replace_once(s,'if !slices.Contains(strings.Split(result.NextAction, " && "), ownedSkillsNextAction) {','if true {')
   s=replace_once(s,'\t"slices"\n','')
  p.write_text(s)
 if n==8:
  for path in ['.agents/skills/roundfix/references/setup.md','skills/roundfix/references/setup.md']:
   p=root/path;s=p.read_text();start=s.index('With a Setup Manifest, Doctor appends\n');end=s.index(' A bundle\n',start)
   p.write_text(s[:start]+s[end+1:])
archive=subprocess.check_output(['git','archive',HEAD],cwd=ROOT)
cases=[(1,'ref_first','./internal/baseline','TestRecordInstalledOwnedSkillsRemovesTheRefOfAnInstalledOwnedSkill'),(2,'ref_first','./internal/baseline','TestRecordInstalledOwnedSkillsRemovesTheRefOfAnInstalledOwnedSkill'),(3,'skills_tool_format','./internal/baseline','TestRecordInstalledOwnedSkillsRemovesTheRefOfAnInstalledOwnedSkill'),(4,'pinned_after_a_bundle_install','./internal/cli','TestDoctorNamesAStaleOwnedSkillLockEntry'),(5,'unpinned_with_the_installed_hash','./internal/cli','TestDoctorNamesAStaleOwnedSkillLockEntry'),(6,'pinned_after_a_bundle_install','./internal/cli','TestDoctorNamesAStaleOwnedSkillLockEntry'),(7,'','./internal/cli','TestDoctorKeepsAFailedSkillsLineWithAStaleOwnedSkillLockEntry'),(8,'','./skills','TestTheGuidesStateTheUnpinnedOwnedLockEntry')]
for n,case,pkg,test in cases:
 if len(sys.argv)>1 and n!=int(sys.argv[1]):continue
 row=f'S{n:02}';print('START',row,flush=True)
 with tempfile.TemporaryDirectory(prefix=row+'-',dir=BASE) as td:
  root=pathlib.Path(td)/'repo';root.mkdir()
  with tarfile.open(fileobj=io.BytesIO(archive)) as tar:tar.extractall(root,filter='data')
  subprocess.run(['git','-c','init.defaultBranch=main','init','--quiet'],cwd=root,check=True)
  subprocess.run(['git','-c','core.fsmonitor=false','add','--all'],cwd=root,check=True)
  mutate(root,n)
  diff=subprocess.check_output(['git','-c','core.fsmonitor=false','diff','--no-ext-diff','--'],cwd=root,text=True)
  pattern='^'+test+'$'+('/^'+case+'$' if case else '')
  cmd=['go','test','-count=1','-timeout','20m','-v','-run',pattern,pkg]
  with tempfile.TemporaryDirectory(prefix='h-',dir=BASE) as h,tempfile.TemporaryDirectory(prefix='t-',dir=BASE) as t:
   p=subprocess.run(cmd,cwd=root,env=cleanenv(h,t),text=True,capture_output=True)
  want='--- FAIL: '+test+('/'+case if case else '')
  killed=p.returncode!=0 and want in p.stdout and '[build failed]' not in p.stdout and '[no tests to run]' not in p.stdout
  out='source: git archive '+HEAD+'\nmutation:\n'+diff+'\ncommand: '+shlex.join(cmd)+'\nstdout:\n'+p.stdout+'stderr:\n'+p.stderr+f'exit: {p.returncode}\nexpected failure: '+want+'\nsemantic kill: '+str(killed)+'\n'
  link=save(row+'-sabotage',out)
  mark(row,'pass' if killed else 'fail','Own temporary `git archive '+HEAD+'` copy, initialized and staged before mutation. `'+shlex.join(cmd)+'` exited '+str(p.returncode)+'.   print('DONE',row,'KILLED' if killed else 'SURVIVED',p.returncode,p.stdout[-1700:],p.stderr[-800:],flush=True)
