def run(args, cwd=ROOT, environ=None):
    return subprocess.run(args, cwd=cwd, env=environ, text=True, capture_output=True)


def record_command(args, cwd=ROOT, environ=None):
    r = run(args, cwd, environ)
    return r, "$ " + shlex.join(args) + "\nexit: " + str(r.returncode) + "\nstdout:\n" + r.stdout + "\nstderr:\n" + r.stderr + "\n"


def finish(row, name, log, passed, summary):
    # Only called after every child has exited, including test suites.
    (EV / name).write_text(log)
    s = REPORT.read_text().replace("| " + row + " | pending |", "| " + row + " | " + ("pass" if passed else "fail") + " |")
    start = s.index("### " + row + "\n")
    end = s.find("\n### ", start + 1)
    if end < 0:
        end = len(s)
    part = s[start:end].replace("Observed: pending.", "Observed: " + summary + " Evidence: [" + name + "](evidence/qa-report-2026-11-09/" + name + ").")
    REPORT.write_text(s[:start] + part + s[end:])
    print(row, "PASS" if passed else "FAIL", summary, flush=True)


def isolated(args, cwd=ROOT):
    caches = {k: run(["go", "env", k]).stdout.strip() for k in ("GOCACHE", "GOMODCACHE")}
    with tempfile.TemporaryDirectory() as h, tempfile.TemporaryDirectory() as t:
        e = os.environ.copy()
        e.update(caches, HOME=h, TMPDIR=t)
        for k in list(e):
            if "API_KEY" in k:
                del e[k]
        return record_command(args, cwd, e)

def named():
    logs, ok = [], True
    for pkg, names in [("spec", SPEC_TESTS), ("cli", CLI_TESTS), ("daemon", DAEMON_TESTS)]:
        r, log = record_command(["go", "test", "-count=1", "-timeout", "20m", "-v", "-run", "^(" + "|".join(names) + ")$", "./internal/" + pkg])
        absent = [n for n in names if not re.search(r"--- PASS: " + n + r" ", r.stdout)]
        ok &= r.returncode == 0 and not absent
        logs.append(log + "Missing PASS lines: " + repr(absent) + "\n")
        print(pkg, "exit", r.returncode, "missing", absent, flush=True)
    finish("Q02", "named-tests.txt", "\n".join(logs), ok, "All 16 required named tests passed." if ok else "Named regressions failed; inspect the retained output.")


def suites():
    commands = [["go", "test", "-count=1", "-timeout", "20m", "-v", "./internal/spec"], ["go", "test", "-count=1", "-timeout", "20m", "-v", "./internal/daemon"], ["go", "test", "-count=1", "-timeout", "20m", "-v", "./skills"], ["go", "test", "-count=1", "-timeout", "20m", "-v", "-tags", "docscontract", "./internal/docscontract"], ["go", "test", "-count=1", "-timeout", "20m", "-v", "-run", "QAReport|Settle|QAPartial|PartialShape", "./internal/cli"]]

        logs.append(log)
        ok &= r.returncode == 0

    finish("Q03", "package-suites.txt", "\n".join(logs), ok, "Five required regression runs exited 0." if ok else "A required suite failed; inspect the retained output.")
