def run(cmd,cwd=W,env=None):
    return subprocess.run(cmd,cwd=cwd,env=env or BASEENV,stdin=subprocess.DEVNULL,capture_output=True,text=True)

def record(name,cmd,r):
    (E/(name+'.txt')).write_text('$ '+shlex.join(cmd)+'\nstdout:\n'+r.stdout+'\nstderr:\n'+r.stderr+'\nexit: '+str(r.returncode)+'\n')
    print(name,'exit',r.returncode,flush=True)

def note(row,status,detail,files):
    text=REPORT.read_text()
    begin=text.index('## Results\n'); finish=text.index('\n## ',begin+5)
    table=text[begin:finish]
    table=re.sub(r'(?m)^\| '+row+r' \| [^|]+ \|', '| '+row+' | '+status+' |',table)
    text=text[:begin]+table+text[finish:]
    start=text.index('### '+row+'\n'); end=text.find('\n### ',start+5)
    if end<0:end=len(text)
    block=text[start:end]
    block=block.replace('Evidence: pending.','Evidence: '+detail+' '+', '.join('['+f+'](evidence/qa-report-2026-11-17-01/'+f+')' for f in files)+'.')
    REPORT.write_text(text[:start]+block+text[end:])

def safe_env(h,t):
    v=BASEENV.copy();v.update(HOME=str(h),TMPDIR=str(t)); return v

def checked(name,cmd,cwd=W,env=None):
    r=run(cmd,cwd,env);record(name,cmd,r)
    if r.returncode:raise RuntimeError(name+' failed; inspect '+name+'.txt')
    return r

def snapshot(root):
    out=[]
    for p in sorted(root.rglob('*')):
        if p.is_symlink():
            b=os.readlink(p).encode();out.append((str(p.relative_to(root)),len(b),hashlib.sha256(b).hexdigest(),'symlink'))
        elif p.is_file():
            b=p.read_bytes();out.append((str(p.relative_to(root)),len(b),hashlib.sha256(b).hexdigest(),'file'))
    return out
