def manifest(repo):
 rows=[]
 for p in sorted(repo.rglob('*')):
  if p.is_symlink():
   data=os.readlink(p).encode();rows.append((str(p.relative_to(repo)),'symlink',len(data),hashlib.sha256(data).hexdigest()))
  elif p.is_file():
   data=p.read_bytes();rows.append((str(p.relative_to(repo)),'file',len(data),hashlib.sha256(data).hexdigest()))
 return rows
