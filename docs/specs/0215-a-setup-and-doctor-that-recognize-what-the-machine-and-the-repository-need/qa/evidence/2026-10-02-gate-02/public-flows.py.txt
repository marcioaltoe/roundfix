def run(label, args, cwd=repo, changes=None):
    ee=dict(env); ee.update(changes or {})
    def fingerprint():
        digest=hashlib.sha256()
        for p in sorted(repo.rglob('*')):
            if '.git' in p.relative_to(repo).parts or p.is_symlink() or not p.is_file():continue
            digest.update(p.relative_to(repo).as_posix().encode());digest.update(p.read_bytes())
        return digest.hexdigest()
    before=fingerprint()
    p=subprocess.run(['./bin/roundfix']+args, cwd=cwd, env=ee, text=True, capture_output=True, timeout=60)
    result={'command':'./bin/roundfix '+' '.join(args), 'cwd':str(cwd), 'HOME':str(home), 'mode':ee.get('QA_MODE'), 'exit':p.returncode, 'stdout':p.stdout, 'stderr':p.stderr,'repository_digest_before':before,'repository_digest_after':fingerprint()}
    (evidence/(label+'.json')).write_text(json.dumps(result,indent=2))
    print(label,'exit',p.returncode);print(p.stdout);print(p.stderr)
    return result
