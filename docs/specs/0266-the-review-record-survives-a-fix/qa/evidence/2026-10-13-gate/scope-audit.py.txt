import json,re,subprocess,tempfile
from pathlib import Path
out=Path(__file__).parent.resolve();root=Path.cwd().resolve();slug='0266-the-review-record-survives-a-fix';specroot='docs/specs/'+slug
log=[]
def git(*args):
 r=subprocess.run(['git',*args],text=True,capture_output=True)
 log.append('$ git '+' '.join(args)+'\n'+r.stdout+r.stderr+'EXIT: '+str(r.returncode))
 assert r.returncode==0,(args,r.stderr)
 return r.stdout.strip()
head=git('rev-parse','HEAD');base=git('merge-base','HEAD','main')
commits=git('rev-list','--reverse',base+'..'+head).splitlines();tasks=[];audit=[]
for sha in commits:
 body=git('show','-s','--format=%B',sha)
 paths=git('diff-tree','--no-commit-id','--name-only','-r',sha).splitlines()
 m=re.search(r'^Roundfix-Task: (task_\d+)$',body,re.M)
 entry={'commit':sha,'task':m.group(1) if m else None,'paths':paths};audit.append(entry)
 if m:
  file=specroot+'/'+m.group(1)+'.md';content=git('show',sha+':'+file)
  tasks.append({'Key':sha,'File':file,'Content':content})
allpaths=git('diff','--name-only',base,head).splitlines()
request={'Tasks':tasks,'Paths':allpaths}
(out/'scope-input.json').write_text(json.dumps(request,indent=2)+'\n')
with tempfile.TemporaryDirectory(prefix='rf266-scope-') as tmp:
 overlay=Path(tmp)/'overlay.json';overlay.write_text(json.dumps({'Replace':{str(root/'cmd/verify-select/main.go'):str(out/'scope-probe.go.txt')}}))
 cmd=['go','run','-overlay',str(overlay),'./cmd/verify-select']
 r=subprocess.run(cmd,input=json.dumps(request),text=True,capture_output=True)
 log.append('$ '+' '.join(cmd)+' < scope-input.json\n'+r.stdout+r.stderr+'EXIT: '+str(r.returncode))
 assert r.returncode==0,r.stderr
 parsed=json.loads(r.stdout)
issues=[]
authpath=specroot+'/_authorization.md';auth=git('show',base+':'+authpath)
allowed=re.findall(r'^  - (\S+)$',auth.split('paths:\n',1)[1].split('operations:',1)[0],re.M)
for entry in audit:
 if entry['task']:
  declared=parsed['declared'][entry['commit']];file=specroot+'/'+entry['task']+'.md'
  outside=[p for p in entry['paths'] if p not in declared and p!=file]
  entry['declarations']=declared;entry['undeclared']=outside
  if outside:issues.append({'task':entry['task'],'undeclared':outside})
  git('merge-base','--is-ancestor',base,entry['commit'])
 if not entry['task']:
  if entry['paths']!=[specroot+'/task_03.md']:issues.append({'unexpected_non_task_commit':entry})
  entry['classification']='pre-Task planning declaration adjustment'
if len([e for e in audit if e['task']])!=4:issues.append('expected four Task commits')
ungranted=[p for p in parsed['governed'] if p not in allowed]
if ungranted:issues.append({'ungranted':ungranted})
for p in allpaths:
 if p.startswith(('internal/delivery/','internal/config/','internal/baseline/assets/modules/','.github/workflows/')) or p in ['Makefile','go.mod','go.sum']:
  issues.append({'forbidden_path':p})
 if p.startswith(('.agents/skills/','skills/')) and not p.startswith(('.agents/skills/roundfix/','skills/roundfix/')) and p!='skills/testdata/owned-skill-versions.json':issues.append({'other_skill':p})
 if p.endswith('.md'):
  old=git('show',base+':'+p) if subprocess.run(['git','cat-file','-e',base+':'+p],capture_output=True).returncode==0 else ''
  new=git('show',head+':'+p)
  section=r'^### QA settlement\n.*?(?=^#{1,3} |\Z)'
  if re.findall(section,old,re.M|re.S)!=re.findall(section,new,re.M|re.S):issues.append({'settlement_changed':p})
status=git('status','--porcelain','--untracked-files=all')
delta=git('diff','--name-only').splitlines()+git('diff','--cached','--name-only').splitlines()
if delta:issues.append({'unexpected_tracked_delta':delta})
for line in status.splitlines():
 if not line[3:].startswith(specroot+'/qa/'):issues.append({'unexpected_worktree_path':line})
result={'head':head,'base':base,'tasks':audit,'all_changed_paths':allpaths,'governed_changed_paths':parsed['governed'],'bounded_paths':allowed,'issues':issues,'worktree_delta':status}
(out/'R7.json').write_text(json.dumps(result,indent=2)+'\n')
(out/'R7.txt').write_text('\n\n'.join(log)+'\n\nSUMMARY:\n'+json.dumps(result,indent=2)+'\n')
print(json.dumps(result,indent=2));raise SystemExit(0 if not issues else 1)
