package main
import("encoding/json";"fmt";"os";"roundfix/internal/spec";"roundfix/internal/speccheck")
type TaskInput struct {Key, File, Content string}
type Request struct {Tasks []TaskInput; Paths []string}
func main(){var req Request;if err:=json.NewDecoder(os.Stdin).Decode(&req);err!=nil{panic(err)}
 declared:=map[string][]string{};governed:=[]string{}
 for _,t:=range req.Tasks{paths,err:=spec.DeclaredTaskPaths([]byte(t.Content),t.File);if err!=nil{panic(fmt.Errorf("%s: %w",t.Key,err))};declared[t.Key]=paths}
 for _,p:=range req.Paths{if speccheck.GovernedPath(p){governed=append(governed,p)}}
 if err:=json.NewEncoder(os.Stdout).Encode(map[string]any{"declared":declared,"governed":governed});err!=nil{panic(err)}}
