import hashlib,json,os,subprocess,tempfile
from pathlib import Path
out=Path(__file__).parent.resolve(); binary=Path('bin/roundfix').resolve()
log=[]; results=[]
with tempfile.TemporaryDirectory(prefix='rf266-public-') as tmp:
 root=Path(tmp).resolve(); home=root/'home'; home.mkdir()
 env={'HOME':str(home),'PATH':os.environ['PATH'],'LANG':'en_US.UTF-8','TMPDIR':str(root)+'/', 'GIT_CONFIG_NOSYSTEM':'1','GIT_CONFIG_GLOBAL':'/dev/null','XDG_CONFIG_HOME':str(home/'config')}
 def run(args,cwd=None,expected=0):
  r=subprocess.run(args,cwd=cwd,env=env,text=True,capture_output=True)
  log.append('cwd: '+str(cwd)+'\n$ '+' '.join(args)+'\nSTDOUT:\n'+r.stdout+'\nSTDERR:\n'+r.stderr+'\nEXIT: '+str(r.returncode))
  assert r.returncode==expected,(args,r.returncode,r.stderr)
  return r
 main=root/'main'; run(['git','init','-b','main',str(main)])
 run(['git','config','user.name','QA Fixture'],main);run(['git','config','user.email','qa@example.invalid'],main);run(['git','config','core.hooksPath','/dev/null'],main)
 (main/'review.txt').write_text('base\n');run(['git','add','review.txt'],main);run(['git','commit','-m','test: create disposable QA base'],main)
 base=run(['git','rev-parse','HEAD'],main).stdout.strip()
 (main/'review.txt').write_text('candidate\n');run(['git','add','review.txt'],main);run(['git','commit','-m','test: create disposable QA candidate'],main)
 real=root/'real'; real.mkdir();link=root/'link';link.symlink_to(real,target_is_directory=True)
 item=link/'repo-slug'/'item';run(['git','worktree','add','-b','test/qa-linked',str(item)],main)
 resolved=item.resolve();version=run(['git','--version']).stdout.strip()
 listing=run(['git','worktree','list','--porcelain'],main).stdout
 assert 'worktree '+str(resolved) in listing and 'worktree '+str(item) not in listing
 for spelling in [item,resolved]:assert run(['git','rev-parse','--show-toplevel'],spelling).stdout.strip()==str(resolved)
 (item/'bin').mkdir();(item/'bin'/'roundfix').symlink_to(binary)
 key=hashlib.sha256(str(resolved).encode()).hexdigest()[:16]
 for mode in ['explicit','default']:
  artifacts=root/'artifacts' if mode=='explicit' else None
  (item/'.roundfixrc.yml').write_text('defaults:\n  artifact_dir: '+json.dumps(str(artifacts) if artifacts else '')+'\npre_pr_review:\n  provider: none\n')
  observations=[]
  for spelling in [item,resolved]:
   review=run(['./bin/roundfix','review','--base',base],spelling)
   record=json.loads(review.stdout);assert record['outcome']=='omitted' and record['repository']==str(resolved)
   files=list((artifacts if artifacts else home/'.roundfix').rglob('pre-pr-review.json'))
   assert len(files)==1,files
   path=files[0];assert path.parent.name==key
   assert json.loads(path.read_text())==record
   assert not (path.parent/'heads').exists()
   prior=path.read_bytes()
   disposition=run(['./bin/roundfix','review','dispose','F1','--fixed-by','HEAD'],spelling,2)
   assert 'pre-pr review record outcome is "omitted", not findings' in disposition.stderr
   assert path.read_bytes()==prior
   observations.append({'spelling':str(spelling),'record_path':str(path),'review_exit':0,'dispose_exit':2,'dispose_reason':disposition.stderr.strip()})
  assert observations[0]['record_path']==observations[1]['record_path']
  results.append({'mode':mode,'observations':observations})
 log.append('Independent Git measurement: '+version+'; both spellings report '+str(resolved))
 log.append('CLI uses only audited binary target '+str(binary)+'; provider none; fresh processes and persisted record re-read; temporary fixtures removed on exit.')
(out/'R5.txt').write_text('\n\n'.join(log)+'\n')
(out/'R5.json').write_text(json.dumps(results,indent=2)+'\n')
