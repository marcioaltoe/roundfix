import re,subprocess
from pathlib import Path
ROOT=Path.cwd(); SLUG='0344-the-spec-skills-announce-the-spec-events-the-user-defines'
SPEC=ROOT/'docs/specs'/SLUG; QA=SPEC/'qa'; E=QA/'evidence/2026-11-03-gate'; REPORT=QA/'qa-report-2026-11-03.md'
BASE='ff394176cfd721e978e4a9186dba536dfe43a94b'; HEAD='1c96d7c31560da31bcc5006574b71cc0a87ed261'
def git(*args):
    p=subprocess.run(['git',*args],text=True,capture_output=True,check=True)
    return p.stdout
commits=git('rev-list','--reverse',BASE+'..'+HEAD).splitlines()
assert len(commits)==4,commits
allowed_skills={'archive-spec','roundfix','write-prd','write-tasks','write-techspec'}
record=(SPEC/'_authorization.md').read_text().split('paths:\n',1)[1].split('operations:',1)[0]
granted=set(re.findall(r'^  - (.+)$',record,re.M))
lines=['Task commit range: '+BASE+'..'+HEAD,'The seeded mechanical audit already lists task_01 to task_03 and their operative grants. task_04 changes only its Task file and the new Go contract test, neither a Governed Path; there is no unlisted tooling commit requiring another authorization chronology audit.']
allpaths=set()
for number,commit in enumerate(commits,1):
    task=SPEC/f'task_{number:02}.md'; text=task.read_text()
    declarations=set(re.findall(r'^- (?:interface|creates|deletes|instruction): `([^`]+)`',text,re.M))
    if '## Recorded paths' in text:
        declarations.update(re.findall(r'`([^`]+)`',text.split('## Recorded paths',1)[1].split('\n## ',1)[0]))
    declarations.add(str(task.relative_to(ROOT)))
    command=['diff-tree','--no-commit-id','--name-only','-r',commit]
    changed=set(git(*command).splitlines()); allpaths.update(changed)
    outside=changed-declarations
    assert not outside,(commit,outside)
    governed={path for path in changed if path.startswith('.agents/skills/') or (path.startswith('skills/') and path.endswith('/SKILL.md'))}
    assert not governed-granted,(commit,governed-granted)
    lines.append('\nTask '+str(number)+' '+commit+'\ncommand: git '+' '.join(command))
    for path in sorted(changed):
        lines.append('  '+path+' — declared'+('; governed, grant lists exact path' if path in governed else ''))
    assert 'status: completed' in text
# An explicit allowed set verifies all exclusions in Requirement 9, including unedited skills and references.
allowed={f'docs/specs/{SLUG}/task_{n:02}.md' for n in range(1,5)}
allowed.update(['CONTEXT.md','skills/testdata/owned-skill-versions.json','docs/user-guide/context-driven-development.md','docs/references/behavior-surfaces.json','skills/spec_event_announcement_test.go'])
for skill in allowed_skills:
    for tree in ['.agents/skills','skills']: allowed.add(f'{tree}/{skill}/SKILL.md')
for tree in ['.agents/skills','skills']: allowed.add(f'{tree}/roundfix/references/implement.md')
assert not allpaths-allowed,allpaths-allowed
lines.append('\nRequirement 9 exclusions: no Makefile, CI, Go module, project config, lockfile, changelog, app version, dist, skill-coverage record, other Go code, Baseline module, docs/agents guide, Archive Record, modes/follow-up reference, roundfix_modes_test.go or other skill file changed.')
patch=git('diff','--no-ext-diff','--unified=0',BASE+'..'+HEAD,'--','CONTEXT.md','.agents/skills','skills','docs/references/behavior-surfaces.json','docs/user-guide/context-driven-development.md')
(E/'task-range.patch.txt').write_text(patch)
for chunk in patch.split('diff --git ')[1:]:
    path=re.search(r'^a/(.+?) b/',chunk).group(1)
    additions=[line[1:] for line in chunk.splitlines() if line.startswith('+') and not line.startswith('+++')]
    removals=[line[1:] for line in chunk.splitlines() if line.startswith('-') and not line.startswith('---')]
    if path=='CONTEXT.md':
        assert not removals,removals
        assert len(additions)==8,len(additions)
        assert [line for line in additions if line.startswith('**')]==['**Spec Event**:','**Spec Event Announcement**:']
        lines.append('CONTEXT range patch: eight added lines for the two declared terms, no deletions and no other glossary-entry changes; current definitions at CONTEXT.md:260 and CONTEXT.md:264.')
    if path.startswith('.agents/skills/') or (path.startswith('skills/') and not path.endswith('.go') and '/testdata/' not in path):
        assert all(re.match(r'^\s*version: \d+\.',line) for line in removals),(path,removals)
        content=[line for line in additions if not re.match(r'^\s*version: \d+\.',line)]
        if any(line.strip() for line in content):
            assert content[0]=='## Spec Event Announcement',(path,content[:2])
            assert all(not line.startswith('## ') or line=='## Spec Event Announcement' for line in content),path
            assert all('### QA settlement' not in line for line in content),path
        lines.append(path+': only announcement insertion and/or version fields changed; existing QA settlement and other sections untouched.')
status=git('status','--porcelain')
assert all(line[3:].startswith(f'docs/specs/{SLUG}/qa/') for line in status.splitlines()),status
lines.append('\nWorktree delta: only QA directory is untracked; product and Task files have no delta. No checkout-to-commit byte comparison performed. Task range patches were read to establish authored scope and preservation.')
(E/'history-scope.txt').write_text('\n'.join(lines)+'\n')
s=REPORT.read_text().replace('| Q04 | pending |','| Q04 | pass |')
a=s.index('### Q04\n'); b=s.find('\n### ',a+1)
s=s[:a]+s[a:b].replace('Observation: pending.','Observation: All four Task commits stay within their Task declarations plus the assigned Task file; every changed Governed Path is granted. The explicit excluded-path set is untouched. The range adds only the two glossary entries, announcement sections and skill version fields; no QA settlement section changes. [Audit and exact path lists](evidence/2026-11-03-gate/history-scope.txt); [range patch](evidence/2026-11-03-gate/task-range.patch.txt). The seeded mechanical chronology audit was retained, not repeated.')+s[b:]
REPORT.write_text(s)
print('Four Task commits audited; no undeclared paths or prohibited scope changes; QA settlement preserved.')
print('\n'.join(lines))
