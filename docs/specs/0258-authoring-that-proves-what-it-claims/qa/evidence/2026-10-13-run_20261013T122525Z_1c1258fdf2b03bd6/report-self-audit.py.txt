from pathlib import Path
import re,json,subprocess
E=Path(__file__).resolve().parent;P=E.parents[1]/'qa-report-2026-10-13.md';S=P.parents[1].relative_to(Path.cwd());s=P.read_text();seed=(E/'seeded-report.txt').read_text();checks={}
checks['one_frontmatter']=s.startswith('---\n') and len(re.findall(r'^---$',s,re.M))==2
front=s.split('---\n')[1];checks['closed_partial']='status: closed' in front and 'verdict: partial' in front
for key in ['auditing_binary','auditor_staleness']:
    checks[key+'_preserved']=re.search(r'^'+key+':.*$',seed,re.M).group()==re.search(r'^'+key+':.*$',s,re.M).group()
def section(t,title):return re.search(r'^## '+re.escape(title)+r'\n.*?(?=^## |\Z)',t,re.M|re.S).group().rstrip('\n')
# Boundary newlines separate appended Markdown sections; every mechanical byte within the section is preserved.
for title in ['Performed repairs','Assigned repair failures','Authorization audit inputs','Mechanical findings','Mechanical skips']:
    checks['preserved_'+title]=section(seed,title)==section(s,title)
rows=re.findall(r'^\| (\d+) \| ([^|]+) \| ([^|]+) \|$',section(s,'Results'),re.M)
checks['nine_formal_rows']=len(rows)==9;checks['no_pending_rows']=all(status.strip()!='pending' for rid,status,prov in rows)
counts={'pass':0,'environment':0,'finding':0,'declared':0,'fail':0,'skipped':0}
allowed={'task_05 Requirement 1','task_05 Requirement 10','repository Verification','outside-evidence row','PRD Unreachable Acceptance','Pull Request row'}
for rid,status,prov in rows:
    status=status.strip();counts['environment' if status.startswith('blocked (environment:') else 'finding' if status.startswith('blocked (finding:') else 'declared' if status.startswith('blocked (declared:') else status]+=1
    block=re.search(r'^### '+rid+r'\n.*?(?=^### |^## |\Z)',s,re.M|re.S).group();checks['inputs_'+rid]='inputs:' in block and 'kind: repository_path' in block
    if rid=='10':checks['scope_commit_input']='kind: commit_range' in block
    checks['provenance_'+rid]=all(any(part.strip().startswith(a) for a in allowed) for part in prov.strip().split(';'))
checks['exact_block_counts']=all(f'rows_blocked_{k}: {counts[k]}' in front for k in ['environment','finding','declared'])
checks['no_evidence_snapshots']='evidence_snapshots' not in s
checks['all_execution_observations_complete']='Observation: pending.' not in s and '\nPending.' not in s and 'Pending sweep.' not in s
missing=[];links=0
for target in re.findall(r'\]\(([^)]+)\)',s):
    if '://' in target or target.startswith('#'):continue
    links+=1
    if not (P.parent/target.split('#')[0]).exists():missing.append(target)
checks['evidence_links_resolve']=not missing
stages=[]
for path in ['.agents/skills/roundfix/references/spec.md','docs/user-guide/commands/spec.md']:
    text=Path(path).read_text()
    for code,stage in [('SC-ABSENCE-UNPROVEN','TechSpec'),('SC-ADR-ALTERNATIVES-UNRELATED','TechSpec'),('SC-COVERAGE-UNREAD','TechSpec'),('SC-CONSUMERS-UNPROVEN','Tasks')]:
        start=text.index('`'+code+'`');paragraph=text[start:].split('\n\n')[0];stages.append({'path':path,'code':code,'matched':stage in paragraph})
checks['documented_stages']=all(x['matched'] for x in stages)
checks['source_clean']=not subprocess.run(['git','diff','--name-only'],capture_output=True,text=True,check=True).stdout
status=subprocess.run(['git','status','--porcelain'],capture_output=True,text=True,check=True).stdout
checks['only_qa_untracked']=all(line[3:].startswith(str(S/'qa')) for line in status.splitlines())
result={'checks':checks,'counts':counts,'missing_links':missing,'evidence_links_checked':links,'document_stages':stages,'status':status,'verdict':'pass' if all(checks.values()) else 'fail'}
(E/'report-self-audit.json').write_text(json.dumps(result,indent=2));print(json.dumps({'verdict':result['verdict'],'failures':[k for k,v in checks.items() if not v],'counts':counts,'links':links},indent=2))
raise SystemExit(0 if all(checks.values()) else 1)
