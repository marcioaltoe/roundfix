import subprocess,json,os,tempfile,shutil
from pathlib import Path
E=Path(__file__).resolve().parent
cache=Path('/tmp/roundfix-go-build-'+str(os.getuid()));cache.mkdir(exist_ok=True)
env=dict(os.environ,GOCACHE=str(cache),GOPROXY='off')
commands=[
['go','test','-count=1','-json','./internal/speccheck'],
['go','test','-count=1','-json','-run','^(TestEveryOwnedSkillVersionIsRecorded|TestAuthorialSkillSync|TestAuthoringSkillsStateTheProofRules|TestWriteTasksSkillStatesTheDeclaredPathRules|TestSettlementGuidanceIsOneTable)$','./skills'],
['go','test','-count=1','-json','-run','^(TestRunSpecCheckCleanText|TestCheckCitationCoverageErrorLocations|TestSpecCheckRunVerification|TestSpecCheckClassifiesTheGateBoundary|TestRunSpecAuditPreservesSpecCheckBehavior|TestDeliverPlan.*)$','./internal/cli'],
['go','test','-count=1','-json','-run','SpecConsistency','./internal/daemon'],
['go','test','-count=1','-json','-run','^TestArchiveLayoutCharacterizationPinsCorpusGoldenAfterSpec0095$','./internal/spec'],
['go','test','-count=1','-json','-tags','docscontract','-run','^(TestCheckCorpusGolden|TestCheckActiveCorpusHasNoErrors|TestTheSkillCoverageMapIsCurrent)$','./internal/docscontract']]
results=[]
for i,cmd in enumerate(commands):
 log=Path(tempfile.mkdtemp(prefix='0258-qa-tests-'))/f'tests-{i+1:02}.jsonl'
 with log.open('w') as f:p=subprocess.run(cmd,env=env,stdout=f,stderr=subprocess.STDOUT)
 shutil.copyfile(log,E/log.name)
 events=[]
 for line in log.read_text().splitlines():
  try:events.append(json.loads(line))
  except ValueError:pass
 passes=[v['Test'] for v in events if v.get('Action')=='pass' and 'Test' in v]
 fails=[v.get('Test',v.get('Package')) for v in events if v.get('Action')=='fail']
 result={'command':cmd,'exit':p.returncode,'evidence':log.name,'passed_tests':passes,'failed_tests':fails};results.append(result)
 (E/'test-results.json').write_text(json.dumps(results,indent=2))
 print(json.dumps({'package':cmd[-1],'exit':p.returncode,'passes':len(passes),'failures':fails}),flush=True)
