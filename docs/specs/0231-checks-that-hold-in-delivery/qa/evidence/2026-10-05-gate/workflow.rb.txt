require 'yaml'
require 'tmpdir'
require 'open3'
f='.github/workflows/ci-verify.yml'
w=YAML.load_file(f); steps=w.fetch('jobs').fetch('verify').fetch('steps'); step=steps.find{|s|s['name']=='Merge the current base branch'}; script=step.fetch('run')
old=YAML.safe_load(IO.popen(['git','show','f2022a59:'+f], &:read)); olds=old.fetch('jobs').fetch('verify').fetch('steps')
['Verify','Verify docs'].each{|name|raise name unless steps.find{|s|s['name']==name}==olds.find{|s|s['name']==name}}
checkout=steps.find{|s|s['uses'].to_s.start_with?('actions/checkout@')}.fetch('with')
raise 'checkout' unless checkout['fetch-depth']==0 && checkout['persist-credentials']==false && checkout['ref'].include?("github.event_name == 'pull_request'") && checkout['ref'].include?("|| ''")
raise 'base env' unless step['env']['BASE_REF']=='${{ github.base_ref }}' && !steps.find{|s|s['name']=='Verify changed'}.fetch('env').key?('VERIFY_BASE')
ENV['GIT_CONFIG_GLOBAL']='/dev/null'; ENV['GIT_CONFIG_NOSYSTEM']='1'
def git(dir,*a)
 o,e,s=Open3.capture3('git','-C',dir,*a);raise e unless s.success?;o.strip
end
[false,true].each do |conflict|
 Dir.mktmpdir('0231-ci-') do |d|
  o=d+'/origin';Dir.mkdir(o);git(o,'init','-q','-b','main')
  git(o,'config','user.name','qa');git(o,'config','user.email','qa@example.com')
  File.write(o+'/shared','base\n');git(o,'add','.');git(o,'commit','-qm','base')
  git(o,'checkout','-qb','feature');File.write(o+(conflict ? '/shared' : '/feature'),'head\n');git(o,'add','.');git(o,'commit','-qm','head');head=git(o,'rev-parse','HEAD')
  git(o,'checkout','-q','main');File.write(o+(conflict ? '/shared' : '/fix'),'moved\n');git(o,'add','.');git(o,'commit','-qm','moved');tip=git(o,'rev-parse','HEAD')
  git(d,'clone','-q',o,d+'/ci');ci=d+'/ci';git(ci,'checkout','-q','--detach',head)
  out,err,status=Open3.capture3({'BASE_REF'=>'main','GITHUB_ENV'=>d+'/env'},'bash','-e','-c',script,chdir:ci)
  puts "conflict=#{conflict} head=#{head} moved_tip=#{tip} exit=#{status.exitstatus}\n#{out}#{err}"
  if conflict
   raise 'conflict accepted' if status.success? || File.exist?(d+'/env') || out.include?('::notice')
   raise 'head moved' unless git(ci,'rev-parse','HEAD')==head
  else
   raise 'merge' unless status.success? && git(ci,'rev-parse','HEAD^1')==head && git(ci,'rev-parse','HEAD^2')==tip
   raise 'export' unless File.read(d+'/env')=="VERIFY_BASE=#{tip}\n"
   raise 'notice' unless out.lines.count{|l|l.strip=="::notice title=tested-base::#{tip}"}==1
   raise 'fix absent' unless File.read(ci+'/fix')=='moved\n'
   puts "parents independently verified; VERIFY_BASE=#{tip}; committer=#{git(ci,'show','-s','--format=%cn <%ce>')}"
  end
 end
end
puts 'PASS: workflow success/conflict; push Verify and Verify docs unchanged'
