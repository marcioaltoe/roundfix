def run(args, label, cwd=ROOT, env=None, expect=0, timeout=1800):
    p = subprocess.run([str(x) for x in args], cwd=cwd, env=env or BASE_ENV,
                       text=True, stdout=subprocess.PIPE, stderr=subprocess.PIPE, timeout=timeout)
    (E / (label + '.txt')).write_text('command: '+shlex.join([str(x) for x in args])+'\ncwd: '+str(cwd)+'\nexit: '+str(p.returncode)+'\nstdout:\n'+p.stdout+'\nstderr:\n'+p.stderr)
    print(label+': exit '+str(p.returncode), flush=True)
    if expect is not None:
        assert p.returncode == expect, label + ': unexpected exit; see evidence'
    return p

def transcript_match(number, p):
    blocks = re.findall(r'```transcript\n(.*?)```', (SPEC/'_techspec.md').read_text(), re.S)
    block = '\n'.join(line.removeprefix('   ') for line in blocks[number-1].splitlines()).rstrip()+'\n'
    _, expected = block.split('stdout:\n',1)
    out, expected = expected.split('stderr:\n',1)
    err, code = expected.rsplit('exit: ',1)
    def matches(template, actual):
        parts=[]
        for line in template.splitlines():
            line=line.removeprefix('   ')
            if line == '...':
                parts.append('(?:[^\\n]*\\n)*')
            else:
                parts.append(re.sub(r'<[^>]+>', r'[^\\n]+', re.escape(line))+'\\n')
        return re.fullmatch(''.join(parts),actual) is not None
    assert int(code.strip()) == p.returncode
    assert matches(out,p.stdout), 'stdout transcript '+str(number)
    assert matches(err,p.stderr), 'stderr transcript '+str(number)

    commits=['dc5cb42b671ca137d09a3c5db7340bfef162d951','2d20362385da7025f24acce81debef29610ace3b',HEAD]
    authorization=git(['show',base+':'+str((SPEC/'_authorization.md').relative_to(ROOT))],ROOT)
    granted=re.search(r'\npaths:\n(.*?)\noperations:',authorization,re.S).group(1)
    granted=set(re.findall(r'^  - (.+)$',granted,re.M))
    source=(ROOT/'internal/speccheck/governed.go').read_text()
    patterns=[re.compile(x) for x in re.findall(r'regexp.MustCompile\(\s*`([^`]+)`',source)]
    exact=set(re.findall(r'"([^"\n]+)"',source.split('paths: exactGovernedPaths(',1)[1].split('),',1)[0]))
    def governed(p): return p in exact or any(r.search(p) for r in patterns)
    output=['repository_path: '+str(ROOT),'commit_range: '+base+'..'+HEAD,'Authorization recorded at Delivery Base before all Task commits.']
    allchanged=set()
    for n,commit in enumerate(commits,1):
        task=SPEC/('task_0'+str(n)+'.md')
        body=git(['show',commit+':'+str(task.relative_to(ROOT))],ROOT)
        declared=set(re.findall(r'^- (?:interface|creates|instruction|deletes): `([^`]+)`',body,re.M))
        if '## Recorded paths' in body: declared.update(re.findall(r'`([^`]+)`',body.split('## Recorded paths',1)[1].split('\n## ',1)[0]))
        declared.add(str(task.relative_to(ROOT)))
        changed=git(['diff-tree','--no-commit-id','--name-only','-r',commit],ROOT).splitlines()
        output.append('task_0'+str(n)+' '+commit)
        for p in changed:
            assert p in declared, 'undeclared Task path: '+p
            assert not governed(p) or p in granted,'ungranted Governed Path: '+p
            output.append('  '+p+(' [governed; granted]' if governed(p) else ' [ordinary; declared]'))
        allchanged.update(changed)
    denied={'Makefile','go.mod','go.sum','.roundfixrc.yml','skills-lock.json','CHANGELOG.md','internal/app/version.go','dist/npm/roundfix/package.json','docs/references/coverage-record.json'}
    for p in allchanged:
        assert p not in denied
        assert not p.startswith(('.github/workflows/','internal/judge/','internal/baseline/assets/source-baselines/','.claude/','.codex/','.opencode/'))
        assert not p.startswith('internal/baseline/assets/modules/') or p=='internal/baseline/assets/modules/core.json'
        assert not p.startswith('.agents/skills/') or p.startswith('.agents/skills/roundfix/')
        assert not (p.startswith('skills/') and '/SKILL.md' in p) or p.startswith('skills/roundfix/')
        if '/skills/' in p or p.startswith('skills/'):
            before=git(['show',base+':'+p],ROOT)
            after=(ROOT/p).read_text()
            def settlement(text):
                if '### QA settlement\n' not in text: return ''
                return re.split(r'\n#{1,3} ',text.split('### QA settlement\n',1)[1],maxsplit=1)[0]
            assert settlement(before)==settlement(after),'QA settlement changed: '+p
    assert git(['show',base+':internal/cli/follow_up_status.go'],ROOT)==(ROOT/'internal/cli/follow_up_status.go').read_text().strip()
    output += ['All changed files are Task-declared or recorded; every governed change is authorized.', 'No prohibited path or QA settlement section changed; follow_up_status.go equals Delivery Base.', 'Project Constraints: no new identifier schema, auth or HTTP endpoint; applicable accepted ADRs and owned guidance are present.']
    (E/'H01-history.txt').write_text('\n'.join(output)+'\n');setrow('H01','pass')
