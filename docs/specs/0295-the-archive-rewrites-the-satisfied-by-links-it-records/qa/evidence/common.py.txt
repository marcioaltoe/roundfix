import os,pathlib,subprocess,tempfile,json,hashlib,re,tarfile,io,shutil
STATE=pathlib.Path(pathlib.Path('/private/tmp/0295-state-path').read_text()); S=json.loads(STATE.read_text()); ROOT=pathlib.Path(S['root']); TMP=pathlib.Path(S['tmp']); SPEC=ROOT/'docs/specs/0295-the-archive-rewrites-the-satisfied-by-links-it-records'; EV=SPEC/'qa/evidence'; REPORT=SPEC/'qa/qa-report-2026-10-21.md'
E={k:v for k,v in os.environ.items() if k in ('PATH','TMPDIR','LANG','LC_ALL','TERM')}; E.update(HOME=str(TMP/'home'),GIT_CONFIG_GLOBAL=str(TMP/'gitconfig'),GIT_CONFIG_NOSYSTEM='1',GOPROXY='off',GOTOOLCHAIN='local',GOSUMDB='off')
# Existing local module/build caches permit strictly offline Go invocations.
for k in ('GOPATH','GOCACHE'):
 E[k]=subprocess.check_output(['go','env',k],text=True).strip()

def run(args,cwd=ROOT,log=None,expected=None,env=None):
 p=subprocess.run(args,cwd=cwd,env=env or E,text=True,capture_output=True)
 if log is not None: log.append('$ '+' '.join(map(str,args))+'\nstdout:\n'+p.stdout+'stderr:\n'+p.stderr+'exit: '+str(p.returncode)+'\n')
 if expected is not None and p.returncode!=expected: raise RuntimeError('Unexpected exit '+str(p.returncode)+' for '+str(args)+'\n'+p.stdout+p.stderr)
 return p

def mark(row,status,body):
 t=REPORT.read_text(); t=re.sub(r'(?m)^\| '+re.escape(row)+r' \| [^|]* \|', '| '+row+' | '+status+' |',t)
 t=t.replace('## '+row+'\n\ninputs:', '## '+row+'\n\ninputs:')
 start=t.index('## '+row+'\n'); end=t.find('\n## ',start+4); end=len(t) if end<0 else end
 section=t[start:end]; section=section.replace('Pending execution.',body); t=t[:start]+section+t[end:]; REPORT.write_text(t)

def archive_copy(rev,name):
 target=TMP/name; target.mkdir(exist_ok=True); raw=subprocess.check_output(['git','archive',rev],cwd=ROOT)
 with tarfile.open(fileobj=io.BytesIO(raw)) as a:a.extractall(target,filter='data')
 return target
