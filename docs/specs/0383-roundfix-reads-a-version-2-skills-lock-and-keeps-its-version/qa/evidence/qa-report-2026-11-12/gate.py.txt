def run(args, cwd=ROOT, env=None, expected=0):
    p = subprocess.run(args, cwd=cwd, env=env or BASEENV, text=True, stdout=subprocess.PIPE, stderr=subprocess.STDOUT)
    out = "$ " + " ".join(args) + "\n" + p.stdout + "\nexit: " + str(p.returncode) + "\n"
    if expected is not None and p.returncode != expected:
        print(out[-5000:], flush=True)
        raise RuntimeError("unexpected exit: " + str(p.returncode))
    return p, out

def save(name, content):
    (E/name).write_text(content)

def status(row, value, detail):
    s=REPORT.read_text()
    s=re.sub(r"^\| "+row+r" \| [^|]+ \|", "| "+row+" | "+value+" |",s,flags=re.M)
    marker="### "+row+" evidence\n"
    s=s.replace(marker,marker+"\n"+detail+"\n",1)
    REPORT.write_text(s)
    print(row+": "+value, flush=True)
