from pathlib import Path
import json,subprocess,re,os
s=json.loads(Path('/tmp/qa0291-state.json').read_text()); tmp=Path(s['tmp']); copy=tmp/'clean'; results={}
env=os.environ.copy();env.update(GOPROXY='off',GOSUMDB='off')
# Explicit named tests and regressions, without executing unrelated package tests.
names=[]
for p in Path(s['spec']).glob('task_0[1-4].md'):
 txt=p.read_text();v=txt.split('## Verification')[1].split('## References')[0];names.extend(re.findall(r'Test[A-Za-z0-9]+',v))
for p in ['internal/delivery/run_slot_wait_test.go','internal/config/run_slot_wait_test.go','internal/cli/doctor_test.go','internal/cli/doctor_characterization_test.go']:
 names.extend(re.findall(r'func (Test\w+)\(', (copy/p).read_text()))
regex='^('+'|'.join(sorted(set(names)))+')$'
commands=[('named-tests',['go','test','-count=1','-v','-timeout=180s','-run',regex,'./internal/cli','./internal/config','./internal/delivery','./internal/store']),('delivery-config-regressions',['go','test','-count=1','-v','-timeout=180s','./internal/config','./internal/delivery']),('skills',['go','test','-count=1','./skills']),('docscontract',['go','test','-count=1','-tags','docscontract','./internal/docscontract']),('skill-mirror',['make','skills-sync-check'])]
for name,cmd in commands:
 with (tmp/(name+'.txt')).open('w') as f:
  f.write('command: '+' '.join(cmd)+'\n');f.flush();r=subprocess.run(cmd,cwd=copy,env=env,stdout=f,stderr=subprocess.STDOUT);f.write('\nexit: '+str(r.returncode)+'\n')
 results[name]=r.returncode;print(name,r.returncode,flush=True)
 (tmp/'test-results.json').write_text(json.dumps(results))
# All authored Verification commands on the archived head, with offline dependency resolution.
for p in sorted(Path(s['spec']).glob('task_0[1-4].md')):

  f.write('command: '+cmd+'\n');f.flush();r=subprocess.run(['sh','-c',cmd],cwd=copy,env=env,stdout=f,stderr=subprocess.STDOUT);f.write('\nexit: '+str(r.returncode)+'\n')
