#!/bin/bash
set -u
meta="$1"; count="$2"; shift 2
burner_pids=()
cleanup() {
  for pid in "${burner_pids[@]}"; do kill -TERM "$pid" 2>/dev/null || true; done
  for pid in "${burner_pids[@]}"; do wait "$pid" 2>/dev/null || true; done
  printf 'cleanup_complete=true\n' >> "$meta"
}
trap cleanup EXIT
trap 'exit 130' INT
trap 'exit 143' TERM
printf 'start_brt=%s\n' "$(TZ=America/Sao_Paulo date '+%Y-%m-%dT%H:%M:%S%z')" >> "$meta"
printf 'start_load=%s\n' "$(sysctl -n vm.loadavg)" >> "$meta"
for ((i=0; i<count; i++)); do
  nice -n 19 yes >/dev/null &
  pid=$!; burner_pids+=("$pid")
  printf 'burner_pid=%s\n' "$pid" >> "$meta"
done
"$@"
result=$?
printf 'end_brt=%s\n' "$(TZ=America/Sao_Paulo date '+%Y-%m-%dT%H:%M:%S%z')" >> "$meta"
printf 'end_load=%s\nexit_status=%s\n' "$(sysctl -n vm.loadavg)" "$result" >> "$meta"
exit "$result"
