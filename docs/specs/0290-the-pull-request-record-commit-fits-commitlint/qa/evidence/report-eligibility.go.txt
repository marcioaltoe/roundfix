package spec
import "testing"
func TestQA0290ReportEligibility(t *testing.T) {
 t.Parallel()
 dir := "../../docs/specs/0290-the-pull-request-record-commit-fits-commitlint"
 report, err := ReadQAReport(dir)
 if err != nil { t.Fatal(err) }
 if err := QAReportEligibility(dir, report); err != nil { t.Fatal(err) }
 if report.Verdict != VerdictPartial || report.RowsBlockedEnvironment != 2 || report.RowsBlockedFinding != 0 || report.RowsBlockedDeclared != 1 || report.RowsBlockedPrePullRequest != 1 || report.RowsBlockedNetworkProhibited != 1 || report.RowsSkipped != 0 { t.Fatalf("unexpected report metadata: %+v", report) }
 t.Logf("qualifying partial: environment=%d (pre-PR=%d, network-prohibited=%d), finding=%d, declared=%d, skipped=%d", report.RowsBlockedEnvironment, report.RowsBlockedPrePullRequest, report.RowsBlockedNetworkProhibited, report.RowsBlockedFinding, report.RowsBlockedDeclared, report.RowsSkipped)
}
