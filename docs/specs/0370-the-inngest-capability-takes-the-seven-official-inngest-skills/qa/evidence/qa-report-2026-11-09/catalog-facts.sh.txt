#!/usr/bin/env bash
set -eu
ref=e45466d2b9e15480eb286a4c19b513494a867bce
module=internal/baseline/assets/modules/backend.json
golden=internal/baseline/assets/formatter-fixtures/standard-typescript-monorepo/golden/docs/agents/skill-dispatch.md
jq -en --slurpfile m "$module" '
  {
    "inngest-cli": "Installing or running the Inngest CLI and its local Dev Server.",
    "inngest-durable-functions": "Building or changing Inngest functions, their triggers, retries, and schedules.",
    "inngest-events": "Designing Inngest events, fan-out, and idempotent event handling.",
    "inngest-flow-control": "Configuring Inngest concurrency, throttling, rate limits, debounce, or batching.",
    "inngest-middleware": "Adding Inngest middleware for logging, tracing, error tracking, or dependency injection.",
    "inngest-setup": "Setting up Inngest in a project: the SDK, client configuration, environment variables, and the serve endpoint.",
    "inngest-steps": "Building or changing the steps of Inngest functions: step runs, sleeps, waits for events, and invocations."
  } as $want
  | ($want|keys) as $names
  | ($m[0].requiredSkills | map(select(startswith("inngest"))) == $names)
    and ([$m[0].skillDispatch[] | select(.skill|startswith("inngest")) | .skill] == $names)
    and ([$m[0].skillDispatch[] | select(.skill|startswith("inngest"))
      | .skill as $skill | .triggers == [{id:("trigger.backend."+$skill),when:$want[$skill]}]] | all)
'
printf '%s\n' 'Module: exactly seven required and dispatched, each with exactly the prescribed trigger.'
jq -r '.skillDispatch[] | select(.skill|startswith("inngest-")) | .triggers[] | "  - `\(.id)`: \(.when)"' "$module" |
while IFS= read -r line; do
  grep -Fx -- "$line" "$golden"
done
if grep -F -- 'trigger.backend.inngest`:' "$golden"; then exit 1; fi
if grep -F -- 'trigger.backend.inngest:' "$golden"; then exit 1; fi
for name in go rust typescript python go-cli-typescript-bun; do
  path="internal/baseline/assets/setups/$name.json"
  jq -e --arg ref "$ref" '[..|objects|select(.type? == "github")|.ref==$ref]|length>0 and all' "$path"
  if grep -F -- c8b2e8511a7ff89dbe9e87a56cc3c29a6acbcb84 "$path"; then exit 1; fi
  printf '%s\n' "$name: every github source at the expected catalog revision, old revision absent."
done
jq -en --slurpfile a internal/baseline/assets/setups/typescript.json --slurpfile b internal/baseline/assets/setups/go-cli-typescript-bun.json '
  ["inngest-cli","inngest-durable-functions","inngest-events","inngest-flow-control","inngest-middleware","inngest-setup","inngest-steps"] as $names
  | def selected($s): [$s.skills[]|select(.name|startswith("inngest"))|{name,path,sourcePath:.source.path,treeDigest}]|sort_by(.name);
  selected($a[0]) as $x | selected($b[0]) as $y
  | $x == $y and ([$x[].name] == $names)
    and ([$x[]|.treeDigest != "" and .treeDigest != null and .path == ("skills/05-implementation-loop/"+.name) and .sourcePath == .path]|all)
'
printf '%s\n' 'Both TypeScript snapshots: seven catalog paths, equal nonempty tree digests, no old inngest.'
jq -e --arg ref "$ref" '[.skills[]|select(.source=="marcioaltoe/skills")|.ref==$ref]|length>0 and all' skills-lock.json
jq '[.skills[]|select(.source=="marcioaltoe/skills")]|length' skills-lock.json
printf '%s\n' 'All catalog lock entries at expected revision.'
