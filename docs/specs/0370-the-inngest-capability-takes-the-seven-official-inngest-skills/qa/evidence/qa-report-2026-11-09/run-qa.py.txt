@contextlib.contextmanager
def isolated():
    with tempfile.TemporaryDirectory(prefix='qa-0370-home-') as h, tempfile.TemporaryDirectory(prefix='qa-0370-tmp-', dir=ROOT / 'bin') as t:
        env = os.environ.copy()
        env.update(CACHES)
        env.update(HOME=h, TMPDIR=t, GIT_ALLOW_PROTOCOL='file')
        for key in KEYS:
            env.pop(key, None)
        yield env

def run(name, args, cwd=ROOT, env=None, expected=0):
    observed_executables = set()
    if args[:2] == ['make', 'skills-third-party-bump']:
        p = subprocess.Popen(args, cwd=cwd, env=env, text=True, stdout=subprocess.PIPE, stderr=subprocess.STDOUT)
        while True:
            for path in Path(env['TMPDIR']).glob('rf-skills.*/roundfix'):
                observed_executables.add(str(path.relative_to(ROOT)))
            try:
                output, _ = p.communicate(timeout=1)
                break
            except subprocess.TimeoutExpired:
                pass
        report = REPORT.read_text()
        for path in sorted(observed_executables):
            report = report.replace('user_flow_executables:\n', 'user_flow_executables:\n  - ' + path + '\n', 1)
        REPORT.write_text(report)
    else:
        p = subprocess.run(args, cwd=cwd, env=env, text=True, stdout=subprocess.PIPE, stderr=subprocess.STDOUT)
        output = p.stdout
    # Retain names of isolated environment variables, never their values.
    if env is not None:
        for key in ['HOME', 'TMPDIR', 'GOMODCACHE', 'GOCACHE']:
            output = output.replace(env[key], '<' + key + '>')
    (EVID / (name + '.txt')).write_text('command: ' + shlex.join(args) + '\nrepository: ' + str(cwd) + '\n' + output + '\n' + '\n'.join('built executable inside Run: ' + path for path in sorted(observed_executables)) + '\nexit: ' + str(p.returncode) + '\n')
    print(name + ': exit ' + str(p.returncode), flush=True)
    if expected is not None and p.returncode != expected:
        raise RuntimeError(name + ' unexpected exit; see evidence')
    return p.returncode, output

def done(row, status, note):
    data = REPORT.read_text()
    data, count = re.subn(r'(?m)^\| ' + str(row) + r' \| pending \|', '| ' + str(row) + ' | ' + status + ' |', data)
    assert count == 1, (row, count)
    data += '\n### Row ' + str(row) + '\n\n' + note + '\n'
    REPORT.write_text(data)

@contextlib.contextmanager
def snapshot(revision, label):
    # Scratch executables remain inside this Run Worktree, never another Run.
    with tempfile.TemporaryDirectory(prefix='qa-0370-' + label + '-', dir=ROOT / 'bin') as path:
        target = Path(path)
        raw = subprocess.check_output(['git', 'archive', revision], cwd=ROOT)
        with tarfile.open(fileobj=io.BytesIO(raw)) as archive:
            archive.extractall(target, filter='data')
        yield target

def tests(name, args, cwd=ROOT, expected=0):
