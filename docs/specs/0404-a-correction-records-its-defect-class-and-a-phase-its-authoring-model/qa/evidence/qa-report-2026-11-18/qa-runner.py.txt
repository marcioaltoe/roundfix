def mark(row,status,detail):
 p=REPORT.read_text(); p=re.sub(r'^\| '+re.escape(row)+r' \| .*? \| (.*?) \|$',lambda m:'| '+row+' | '+status+' | '+m[1]+' |',p,flags=re.M)
 pattern=r'(### '+re.escape(row)+r'\n.*?)(?=\n### |\Z)'
 p=re.sub(pattern,lambda m:m[1]+'\n'+detail+'\n',p,flags=re.S)
 REPORT.write_text(p); print(row,status,flush=True)

def run(args,cwd=None,en=None):
 p=subprocess.run([str(x) for x in args],cwd=cwd or ROOT,env=en or ENV,stdin=subprocess.DEVNULL,capture_output=True,text=True)
 return p

def checked(args,cwd=None,en=None):
 p=run(args,cwd,en)
 if p.returncode: raise RuntimeError(shlex.join([str(x) for x in args])+': '+p.stdout+p.stderr)
 return p.stdout.strip()

def git(repo,*args): return checked(['git','-C',repo,*args])

def isolated(h,t):
 en=dict(ENV,HOME=str(h),TMPDIR=str(t),RUSTUP_HOME=RUST,CARGO_HOME=CARGO)
 for k in ['GIT_AUTHOR_NAME','GIT_AUTHOR_EMAIL','GIT_COMMITTER_NAME','GIT_COMMITTER_EMAIL']: en.pop(k,None)
 return en

def capture(label,p,args,cwd,extra=''):
 text='command: '+shlex.join([str(a) for a in args])+'\ncwd: '+str(cwd)+'\nstdout:\n'+p.stdout+'stderr:\n'+p.stderr+f'exit: {p.returncode}\n'+extra
 (E/(label+'.txt')).write_text(text)
 return text

def link(label): return '[evidence](evidence/qa-report-2026-11-18/'+label+'.txt)'

def init(repo):

def commit(repo,msg):
 git(repo,'add','-A'); git(repo,'commit','--allow-empty','-m',msg); return git(repo,'rev-parse','HEAD')

def snap(*roots):
 result={}
 for r in roots:
  for p in r.rglob('*'):
   if p.is_file():
    st=p.stat(); result[str(p)]=(st.st_size,st.st_mtime_ns,hashlib.sha256(p.read_bytes()).hexdigest())
 return result
