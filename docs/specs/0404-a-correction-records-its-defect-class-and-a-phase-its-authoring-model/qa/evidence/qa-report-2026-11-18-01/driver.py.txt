def transcript_match(actual,expected):
 a=actual.splitlines();b=expected.splitlines()
 def match(i,j):
  if j==len(b):return i==len(a)
  if b[j]=='...':return any(match(k,j+1) for k in range(i,len(a)+1))
  pattern=''.join('.+' if part.startswith('<') else re.escape(part) for part in re.split(r'(<[^>]+>)',b[j]))
  return i<len(a) and re.fullmatch(pattern,a[i]) is not None and match(i+1,j+1)
 assert match(0,0),('transcript differs',actual,expected)
