def versions(text):
 vals=re.findall(r'^\s*(?:version):\s*([^\s]+)',text.split('---',2)[1],flags=re.M)

def corrected(data):
 section=data.split('## ADR Alternatives',1)[1]
 comment=section[section.index('<!--'):section.index('-->')+3]
 norm=' '.join(comment.split())
 expected='rejected. Begin the relation with `not applicable`, or with `não se aplica` in a Spec written in Portuguese, for an ADR the Spec lists only to record why it does not apply.'
 assert expected in norm,norm
 assert norm.count('Begin the relation with')==1,norm
 assert norm.index('not applicable')<norm.index('não se aplica')
 return norm
