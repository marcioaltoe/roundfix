def run(name,args,cwd=ROOT,extra=None,expected=0):
    env=ENV.copy(); env.update(extra or {})
    p=subprocess.run(args,cwd=cwd,env=env,stdout=subprocess.PIPE,stderr=subprocess.PIPE,text=True)
    (E/(name+'.txt')).write_text('$ '+shlex.join(args)+'\nworking_tree: '+('audited head' if cwd==ROOT else 'disposable copy')+'\nstdout:\n'+p.stdout+'\nstderr:\n'+p.stderr+'\nexit: '+str(p.returncode)+'\n')
    print(name+': exit '+str(p.returncode),flush=True)
    if p.returncode!=expected: raise RuntimeError(name+': unexpected exit; see evidence')
    return p.stdout+p.stderr
