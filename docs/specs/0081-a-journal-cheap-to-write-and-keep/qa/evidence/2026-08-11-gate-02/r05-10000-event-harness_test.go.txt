package tui

import (
	"testing"

	"roundfix/internal/spec"
)

func TestQACockpitRefreshCostAtTenThousandEvents(t *testing.T) {
	type refreshCost struct {
		headerRows   int
		fullRows     int
		payloadBytes int
	}
	costs := map[int]refreshCost{}
	for _, backlog := range []int{20, 10_000} {
		source := newForwardCursorSource(t, backlog)
		model := newForwardCursorCockpit(t, source)
		if source.headerRows != backlog+2 {
			t.Fatalf("backlog %d: opening fold read %d headers, want %d", backlog, source.headerRows, backlog+2)
		}
		source.reset()

		source.addLine("agent line after backlog\n")
		source.addTaskEvent(t, "task_02", "settled", spec.StatusCompleted, 2)
		model.refreshTaskJournalEvents()
		costs[backlog] = refreshCost{
			headerRows:   source.headerRows,
			fullRows:     source.fullRows,
			payloadBytes: source.payloadBytes,
		}
	}

	if costs[20] != costs[10_000] {
		t.Fatalf("refresh cost follows backlog size: 20=%+v 10000=%+v", costs[20], costs[10_000])
	}
	if costs[10_000].headerRows != 2 || costs[10_000].fullRows != 1 {
		t.Fatalf("10,000-event refresh read %+v, want 2 headers and 1 full event", costs[10_000])
	}
	t.Logf("QA_COCKPIT backlog_20=%+v backlog_10000=%+v", costs[20], costs[10_000])
}
