def isolated(home,tmp):

def command(args,cwd=ROOT,env=None):
 p=subprocess.run(args,cwd=cwd,env=env,capture_output=True,text=True)
 return p,'command: '+(' '.join(args))+'\nexit: '+str(p.returncode)+'\nstdout:\n'+p.stdout+'stderr:\n'+p.stderr

def testrow(rid,args,passes=()):
 with tempfile.TemporaryDirectory() as h,tempfile.TemporaryDirectory() as t:
  p,body=command(['go','test','-count=1','-timeout','20m',*args],env=isolated(h,t))
  ok=p.returncode==0 and all('--- PASS: '+n+' ' in p.stdout for n in passes)
  result(rid,body,ok)
  print('\n'.join((p.stdout+p.stderr).splitlines()[-8:]),flush=True)
