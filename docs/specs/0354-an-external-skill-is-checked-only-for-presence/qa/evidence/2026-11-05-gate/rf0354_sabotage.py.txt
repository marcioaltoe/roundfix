def replace_function(s,name,body):
 start=s.index('func '+name+'(');br=s.index('{',start);depth=1;i=br+1
 while depth:
  depth+=(s[i]=='{')-(s[i]=='}');i+=1
 return s[:br+1]+'\n'+body+'\n'+s[i-1:]
