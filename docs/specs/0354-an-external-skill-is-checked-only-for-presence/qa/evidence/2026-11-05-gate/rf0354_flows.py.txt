def snap(root):
 return {str(p.relative_to(root)):hashlib.sha256(p.read_bytes()).hexdigest() for p in root.rglob('*') if p.is_file() and not p.is_symlink() and '.git' not in p.relative_to(root).parts and 'bin' not in p.relative_to(root).parts}

def skillsline(p): return next((v for v in p.stdout.splitlines() if v.startswith('skills: ')),'')
