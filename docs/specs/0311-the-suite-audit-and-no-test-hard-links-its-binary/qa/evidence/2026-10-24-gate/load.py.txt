 binary=work/(label+'.test'); cmd=['go','test','-c','-o',str(binary),'./internal/agent']

  out.write('revision: '+sha+'\ncommand: '+' '.join(cmd)+'\n'); out.flush(); result=subprocess.run(cmd,cwd=root,env=e,stdout=out,stderr=subprocess.STDOUT); out.write('exit: '+str(result.returncode)+'\n')
 assert result.returncode==0,(label,'compile failed'); print(label,'compiled',flush=True)
for label,sha in [('head',plan['head']),('base',plan['base'])]:

 for round_no in range(1,9):
  start=now(); children=[]; handles=[]; rounddir=work/(label+'-'+str(round_no)); rounddir.mkdir()

   out=(rounddir/('run-'+str(index)+'.txt')).open('w'); handles.append(out)
