 "os"
 "path/filepath"
 "testing"
 "unicode/utf8"
 "roundfix/internal/secretscan"
)
func TestQA0397FinalReportAndEvidence(t *testing.T) {
 t.Parallel()
 report := "/Volumes/Brain-SSD/cache/.roundfix/worktrees/roundfix-deliver-0397-each-phase-of-a-spec-is-recorded-and-starts-in-a-fresh-session-b9133b5c8c843a67-d32224fe-81b16323/run_20261116T113044Z_115c9d691f2c8277/docs/specs/0397-each-phase-of-a-spec-is-recorded-and-starts-in-a-fresh-session/qa/qa-report-2026-11-16.md"
 dir := "/Volumes/Brain-SSD/cache/.roundfix/worktrees/roundfix-deliver-0397-each-phase-of-a-spec-is-recorded-and-starts-in-a-fresh-session-b9133b5c8c843a67-d32224fe-81b16323/run_20261116T113044Z_115c9d691f2c8277/docs/specs/0397-each-phase-of-a-spec-is-recorded-and-starts-in-a-fresh-session"
 q, err := ReadQAReportFile(report)
 if err != nil { t.Fatal(err) }
 if err := QAReportEligibility(dir, q); err != nil { t.Fatal(err) }
 if q.Verdict != VerdictPartial || q.RowsBlockedDeclared != 6 || q.RowsBlockedEnvironment != 1 || q.RowsBlockedFinding != 0 || q.RowsSkipped != 0 { t.Fatalf("unexpected verdict/counts: %+v",q) }
 if err := RequireResolvedEvidenceLinks(t.Context(),report); err != nil { t.Fatal(err) }
 if err := RequireOwnRunEvidence(t.Context(),report,BorrowedEvidenceOptions{RunBranchPrefix:"roundfix/run-",AuditedHead:"b0c9bb8eabf854efcc831b92dbec0fe4d531f423"}); err != nil { t.Fatal(err) }
 files := 0
 err = filepath.Walk(filepath.Join(dir,"qa"),func(path string,info os.FileInfo,err error) error {
  if err != nil { return err }; if info.IsDir(){return nil}
  data, err := os.ReadFile(path); if err != nil{return err}
  if !utf8.Valid(data) { t.Errorf("non-text evidence: %s",path) }
  for _,match := range secretscan.ScanContent(path,data) { t.Error(match.String()) }
  files++; return nil
 })
 if err != nil{t.Fatal(err)}
 t.Logf("eligible qualifying partial; counts 1 environment, 0 finding, 6 declared; links resolved; evidence belongs to this Run; %d text files scanned with no secret matches",files)
}
