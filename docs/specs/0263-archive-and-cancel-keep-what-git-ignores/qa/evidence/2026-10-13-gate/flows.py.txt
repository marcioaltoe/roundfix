def git(repo,*args):return run(repo,['git',*args],True)['stdout']
def sha(repo):
 out={}
 for p in sorted(repo.rglob('*')):
  if p.is_symlink():out[str(p.relative_to(repo))]=hashlib.sha256(('symlink:'+os.readlink(p)).encode()).hexdigest()
  elif p.is_file():out[str(p.relative_to(repo))]=hashlib.sha256(p.read_bytes()).hexdigest()
 return out
def tree(repo):return {p:d for p,d in sha(repo).items() if not p.startswith('.git/')}
