def clean(s):
    for a,b in sorted(REDACT.items(),key=lambda v:-len(v[0])): s=s.replace(a,b)
    return s

def run(args,cwd=W,env=None,expect=0):
    p=subprocess.run(args,cwd=cwd,env=env or E,stdin=subprocess.DEVNULL,text=True,stdout=subprocess.PIPE,stderr=subprocess.PIPE)
    label=' '.join(map(str,args)).replace(str(W)+'/','./')
    log='$ '+label+'\nstdout:\n'+p.stdout+'\nstderr:\n'+p.stderr+'\nexit: '+str(p.returncode)+'\n'
    if expect is not None and p.returncode!=expect: raise RuntimeError(clean(log))
    return p,clean(log)
def save(name,txt):
