def snapshot(repo):
 result=[]
 for p in sorted(repo.rglob('*')):
  if '.git' in p.relative_to(repo).parts: continue
  if p.is_symlink(): result.append((str(p.relative_to(repo)),'link',os.readlink(p)))
  elif p.is_file(): result.append((str(p.relative_to(repo)),p.stat().st_size,hashlib.sha256(p.read_bytes()).hexdigest()))
 return result
