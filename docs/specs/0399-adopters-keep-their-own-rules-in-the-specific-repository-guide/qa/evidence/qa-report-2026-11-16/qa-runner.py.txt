def git(*args,cwd=ROOT):
 p=subprocess.run(['git',*args],cwd=cwd,text=True,capture_output=True,stdin=subprocess.DEVNULL)
 if p.returncode: raise RuntimeError(p.stderr)
 return p.stdout

@contextlib.contextmanager
def isolated():
 m=subprocess.check_output(['go','env','GOMODCACHE'],text=True).strip()
 c=subprocess.check_output(['go','env','GOCACHE'],text=True).strip()
 with tempfile.TemporaryDirectory(prefix='qa399-home-') as h,tempfile.TemporaryDirectory(prefix='qa399-tmp-') as t:
  en=os.environ.copy()
  for k in list(en):
   if k.endswith(('API_KEY','TOKEN','SECRET','PASSWORD')) or k=='CLAUDE_CODE_SESSION_ID': en.pop(k,None)
  en.update(HOME=h,TMPDIR=t,GOMODCACHE=m,GOCACHE=c,GOPROXY='off',GOSUMDB='off',GOTOOLCHAIN='local',GIT_CONFIG_NOSYSTEM='1',GIT_CONFIG_GLOBAL='/dev/null',GIT_TERMINAL_PROMPT='0')
  yield en

def command(cmd,label,cwd=ROOT,en=None):
 if en is None:
  with isolated() as ie: return command(cmd,label,cwd,ie)
 p=subprocess.run([str(a) for a in cmd],cwd=cwd,env=en,text=True,capture_output=True,stdin=subprocess.DEVNULL)
 (E/(label+'.txt')).write_text('$ '+subprocess.list2cmdline([str(a) for a in cmd])+'\ncwd: '+str(cwd)+'\nstdout:\n'+p.stdout+'\nstderr:\n'+p.stderr+'\nexit: '+str(p.returncode)+'\n')
 print(label, 'exit',p.returncode,flush=True)
 return p

def row(q,status,detail,links=()):
 text=REPORT.read_text()
 text=re.sub(r'(?m)^\| '+re.escape(q)+r' \| [^|]+ \|',f'| {q} | {status} |',text)
 pattern=r'(### '+q+r'\n.*?)(?=\n### |\Z)'
 def update(m):
  b=m.group(1)
  b=b.replace('Status: pending.',f'Status: {status}.')
  return b+'\n\n'+detail+'\n'+''.join(f'\nEvidence: [{l}](evidence/qa-report-2026-11-16/{l}).\n' for l in links)
 text=re.sub(pattern,update,text,flags=re.S)
 REPORT.write_text(text)
 print(q,status,detail[:150],flush=True)

def test(label,args,required=()):

 missing=[n for n in required if not re.search(r'--- PASS: '+re.escape(n)+r' ',p.stdout)]
 assert p.returncode==0 and not missing, f'{label}: exit {p.returncode}; missing {missing}; {p.stderr[-500:]}'
 return p
