        def emit():
            nonlocal outside

            keep=[];i=0
            def cut():
                nonlocal keep
                while keep and not keep[0][1].strip():keep.pop(0)
                while keep and not keep[-1][1].strip():keep.pop()
                if keep:
                    first=keep[0][1].strip().encode()
                    if len(first)>120:
                        first=first[:120]
                        while True:
                            try:first=first.decode();break
                            except UnicodeDecodeError:first=first[:-1]
                        first+="..."
                    else:first=first.decode()
                    result.append({"path":path,"startLine":keep[0][0],"endLine":keep[-1][0],"firstLine":first,"moveTo":"docs/agents/specific-repository.md"})
                keep=[]
            while i<len(span):
                b=re.match(r"^\s*<!--\s*BEGIN:([^\s]+)\s*-->\s*$",span[i][1])
                found=None
                if b:
                    for j in range(i+1,len(span)):
                        e=re.match(r"^\s*<!--\s*END:([^\s]+)\s*-->\s*$",span[j][1])
                        if e and e.group(1)==b.group(1):found=j;break
                if found is not None:cut();i=found+1
                else:keep.append(span[i]);i+=1
            cut()
    return sorted(result,key=lambda x:(x["path"],x["startLine"]))
