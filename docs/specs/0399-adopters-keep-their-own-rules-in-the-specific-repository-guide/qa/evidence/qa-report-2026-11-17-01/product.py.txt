def snapshot(repo):
    rows=[]
    for p in sorted(repo.rglob("*")):
        if ".git" in p.relative_to(repo).parts:continue
        if p.is_symlink():rows.append((str(p.relative_to(repo)),"symlink",os.readlink(p)))
        elif p.is_file():
            b=p.read_bytes();rows.append((str(p.relative_to(repo)),len(b),hashlib.sha256(b).hexdigest()))
    return rows
