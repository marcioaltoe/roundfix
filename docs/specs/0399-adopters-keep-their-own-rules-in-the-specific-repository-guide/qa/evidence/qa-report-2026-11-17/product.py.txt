def snapshot(repo):
    out={}
    for p in sorted(repo.rglob('*')):
        if '.git' in p.relative_to(repo).parts: continue
        if p.is_symlink(): out[str(p.relative_to(repo))]={'link':os.readlink(p)}
        elif p.is_file():
            b=p.read_bytes();out[str(p.relative_to(repo))]={'size':len(b),'sha256':hashlib.sha256(b).hexdigest()}
    return out
