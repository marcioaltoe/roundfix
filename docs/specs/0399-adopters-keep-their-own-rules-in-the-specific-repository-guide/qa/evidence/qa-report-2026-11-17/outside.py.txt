    paths={a['path'] for a in manifest['managedArtifacts'] if a['path'].endswith('.md') and a['path']!='docs/agents/specific-repository.md'}
    alias=repo/'CLAUDE.md'
    if alias.is_file() and not alias.is_symlink(): paths.add('CLAUDE.md')
    results=[]
    for path in sorted(paths):
        p=repo/path
        if p.is_symlink(): continue
        lines=p.read_text().splitlines();keep=[True]*len(lines)
        a=0
        while a<len(lines):
            match=re.match(r'^\s*<!-- setup-context-driven:begin id=([^ ]+) .*-->\s*$',lines[a])
            if match:
                end=next((b for b in range(a+1,len(lines)) if re.match(r'^\s*<!-- setup-context-driven:end id='+re.escape(match[1])+r' -->\s*$',lines[b])),None)
                if end is not None:
                    for b in range(a,end+1): keep[b]=False
                    a=end+1;continue
            foreign=re.match(r'^\s*<!--\s*BEGIN:([^\s]+?)\s*-->\s*$',lines[a])
            if foreign:
                end=next((b for b in range(a+1,len(lines)) if re.match(r'^\s*<!--\s*END:'+re.escape(foreign[1])+r'\s*-->\s*$',lines[b])),None)
                if end is not None:
                    for b in range(a,end+1): keep[b]=False
                    a=end+1;continue
            a+=1
        a=0
        while a<len(lines):
            if not keep[a]: a+=1;continue
            end=a
            while end<len(lines) and keep[end]:end+=1
            left,right=a,end
            while left<right and not lines[left].strip():left+=1
            while right>left and not lines[right-1].strip():right-=1
            if left<right: results.append((path,left+1,right,lines[left].strip()))
            a=end
    return sorted(results)
