        elif letter=='b':
            after=before.replace('for start < end && len(bytes.TrimSpace(lines[start])) == 0 {','for false && start < end && len(bytes.TrimSpace(lines[start])) == 0 {',1).replace('for start < end && len(bytes.TrimSpace(lines[end-1])) == 0 {','for false && start < end && len(bytes.TrimSpace(lines[end-1])) == 0 {',1)

        p.write_text(after)
        diff=''.join(difflib.unified_diff(before.splitlines(True),after.splitlines(True),fromfile=file,tofile=file))

No regeneration, synchronization or mutation of this Run source.
