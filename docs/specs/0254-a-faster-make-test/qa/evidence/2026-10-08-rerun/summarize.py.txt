import json,re,gzip
from pathlib import Path
E=Path(__file__).resolve().parent
measurements=json.loads((E/'measurements.json').read_text())
comparisons=[]
for pair in [1,2]:
 summaries={}
 for label in ['start','head']:
  name=f'suite-{label}-{pair}'
  if name not in measurements: continue
  packages={}; phase={};paused=set();failures=[]
  transcript=E/(name+'.stdout.log.txt')
  raw=transcript.read_text() if transcript.exists() else gzip.open(str(transcript)+'.gz','rt').read()
  for line in raw.splitlines():
   try:v=json.loads(line)
   except ValueError:continue
   action=v.get('Action');test=v.get('Test')
   if action in ['pass','fail','skip'] and not test:packages[v['Package']]={'elapsed':v.get('Elapsed'),'status':action}
   if action=='fail':failures.append(v)
   if v.get('Package')=='roundfix/internal/cli' and test and '/' not in test:
    if action=='pause':paused.add(test)
    elif action in ['pass','fail','skip']:phase[test]=v.get('Elapsed',0)
  sequential={n:t for n,t in phase.items() if n not in paused}
  summary={'measurement':measurements[name],'packages':packages,'sequential_cli':{'count':len(sequential),'elapsed_sum':sum(sequential.values()),'tests':sequential},'failures':failures}
  (E/(name+'-summary.json')).write_text(json.dumps(summary,indent=2));summaries[label]=summary
 if len(summaries)!=2:continue
 a=summaries['start'];b=summaries['head'];wall=b['measurement']['wall']/a['measurement']['wall'];phase=b['sequential_cli']['elapsed_sum']/a['sequential_cli']['elapsed_sum']
 load_drift={k:abs(a['measurement'][k][0]-b['measurement'][k][0])/min(a['measurement'][k][0],b['measurement'][k][0]) for k in ['load_before','load_after']}
 result={'pair':pair,'wall_ratio':wall,'wall_ceiling':.65,'wall_meets':wall<=.65,'sequential_ratio':phase,'sequential_ceiling':.25,'sequential_meets':phase<=.25,'load_drift':load_drift,'load_repeat_trigger':max(load_drift.values())>.5,'both_exit_zero':a['measurement']['exit']==b['measurement']['exit']==0}
 comparisons.append(result)
(E/'suite-pairs.json').write_text(json.dumps(comparisons,indent=2));print(json.dumps(comparisons,indent=2))
