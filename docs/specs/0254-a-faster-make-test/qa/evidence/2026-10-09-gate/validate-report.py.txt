import re,json,subprocess
from pathlib import Path
E=Path(__file__).resolve().parent
R=E.parents[1]/"qa-report-2026-10-09.md"
s=R.read_text(); front=s.split("---",2)[1]
assert len(re.findall(r"^---$",s,re.M))==2
assert 'auditing_binary: "0.64.0 (93c4dac1, built 2026-10-09 06:39:22 -0300)"' in front
assert 'auditor_staleness: "current: commit ancestry: build commit does not predate the delivery base"' in front
assert "status: closed" in front and re.search(r"^verdict: (pass|fail|partial)$",front,re.M)
rows=re.findall(r"^\| (R\w+) \| ([^|]+) \| ([^|]+) \|$",s,re.M)
expected={"R1","R7a","R7b","R7c","R7d","R7e","R9","R10","R11"}
assert {r[0] for r in rows}==expected and len(rows)==9
counts={"environment":0,"finding":0,"declared":0}; passed=0
for ident,status,provenance in rows:
 status=status.strip();assert status!="pending" and status!="skipped"
 if status=="pass":passed+=1
 elif status.startswith("blocked ("):
  kind=status.split("(",1)[1].split(":",1)[0];counts[kind]+=1
 else:assert status=="fail"
 block=re.search(r"^### "+ident+r"\n(.*?)(?=^### |^## |\Z)",s,re.M|re.S)
 assert block and len(re.findall(r"```yaml\ninputs:",block[1]))==1
 assert "kind:" in block[1] and "ref:" in block[1]
 assert "Evidence:" in block[1] and "Independent confirmation:" in block[1]
for kind,n in counts.items():assert f"rows_blocked_{kind}: {n}" in front
assert "| consequent-fix commit order | consequent-fix declaration |" in s
assert "evidence_snapshots" not in s
assert all(x["status"]=="reaped" for x in json.loads((E/"started-pids.json").read_text()))
assert subprocess.run(["git","-c","core.fsmonitor=false","diff","--quiet"]).returncode==0
result={"rows":len(rows),"passed":passed,"blocked":counts,"pending":0,"mechanical_seed_preserved":True,"implementation_delta":False}
(E/"report-audit.json").write_text(json.dumps(result,indent=2));print(json.dumps(result,indent=2))
