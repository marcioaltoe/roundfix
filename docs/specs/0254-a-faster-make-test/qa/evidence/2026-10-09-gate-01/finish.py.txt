P=E.parent.parent/'qa-report-2026-10-09-01.md'

 if n!='sabotage-rule' and m['exit']!=0:failures.append(n+': exit '+str(m['exit']))
if not F['all_named_tests_pass'] or not F['all_profiles_pause']:failures.append('Named tests or six profile parallel events missing')

 if not a['wall_meets'] or not a['sequential_meets']:failures.append('Pair '+str(a['pair'])+' fails a numeric target')
if not R['race']['all_seven_pass']:failures.append('Seven-package race proof failed')
if not R['shuffle']['all_seven_pass']:failures.append('Seven-package shuffle proof failed')

if not R['contracts']['both_pass'] or not R['contracts']['head_faster']:failures.append('Full Contract Run metric failed')
verdict='fail' if failures else 'partial'
t=P.read_text()
t=re.sub(r"\n## Execution progress\n.*?(?=\n## |\Z)", "", t, flags=re.S)
t=t.replace('status: in-progress','status: closed',1).replace('verdict: pending','verdict: '+verdict,1)
# Replace the provisional performance section without disturbing other sections.
start=t.index('## Performance execution\n')
end=t.index('\n## ',start+3)
performance='## Performance execution\n\nRequirement 3 ran the exact `go test -count=1 -parallel 16 -json ./...` start/head pair and the authored repeat after pair 1 load drift. Each clone had its own warmed cache. for a in Pairs:
 for label in ['start','head']:
  n=f"suite-{label}-{a['pair']}";m=M[n];s=json.loads((E/(n+'-summary.json')).read_text())['sequential_cli']
  performance+=f"| {a['pair']} | {label} | {m['exit']} | {m['wall']:.2f} | {m['user']:.2f} | {m['system']:.2f} | {s['elapsed_sum']:.2f} | {s['count']} | {m['load_before'][0]:.2f}/{m['load_after'][0]:.2f} |\n"
performance+='\n'
for a in Pairs:
 performance+=f"Pair {a['pair']}: wall ratio {a['wall_ratio']*100:.2f}% (target ≤70%); CLI sequential ratio {a['sequential_ratio']*100:.2f}% (target ≤25%); boundary load drift requires repeat: {a['load_repeat_trigger']}.\n\n"
performance+='Both pairs are disclosed. Any continuing load drift limits a stable causal estimate; no favorable-run search or changed threshold occurred. Numeric acceptance is evaluated at the approved thresholds.\n'
t=t[:start]+performance+t[end:]
t+='\n## Focused behavior and negative proof\n\nRequirements 2 and 4: focused rule, selector, profile and archive checks exit 0. focused-observables.json confirms the named tests, unchanged Makefile wiring tests, all six profile pause events, seeded rule/collision cases and current ceiling counts. Full ordinary suites execute all created/named ordinary Task tests.\n\n'
t+='Current counts: '+ '; '.join(F['counts'])+'.\n\n'
t+='In the disposable head clone only, removing the first Parallel call from TestArchiveRefusesASpecAFileStillPins makes the rule fail at internal/cli/archive_active_spec_path_test.go:31. Original bytes are restored in finally; the restored rule passes. This deliberate negative result is expected evidence, not a product failure. Exact diagnostic and exits: sabotage-rule.stdout.log.txt, restored-rule.stdout.log.txt and focused-observables.json.\n\n'

t+=f"Independent Git archive inventory compares {archive['inventories'][0]['regular_files']} whole-archive files with {archive['inventories'][1]['regular_files']} narrowed files; both retain the same nonempty set of {len(archive['inventories'][0]['qa_reports'])} QA Reports. Both live archive-reader tests pass. See archive-equivalence.json.\n"
t+='\n## Safety execution\n\nRequirement 5 runs all seven packages once with `go test -count=1 -race -short -timeout 30m` and once with `go test -count=1 -shuffle=on -v -timeout 20m`. Short mode skips the two declared Store journal harnesses; shuffle runs them. Exact command lists are in measurements.json; shuffle seeds and every observed test result are retained in the transcript and shuffle-seeds.json.\n\n| Package | Race/short | Shuffle |\n| --- | --- | --- |\n'
for p in ['cli','daemon','spec','speccheck','worktree','store','baseline']:

 for mode in ['race','shuffle']:
  a=R[mode]['packages'].get(p,{});vals.append(a.get('status','missing')+' '+a.get('detail',''))
 t+=f"| internal/{p} | {vals[0]} | {vals[1]} |\n"

 t+='Mandatory execution failures: '+ '; '.join(failures)+'. Exact diagnostics remain in the named transcripts. These independent failures do not block unrelated coverage rows.\n'

 t+='No current execution failure found. Prior F4 forge-probe failure is not reproduced in the fresh required race run; the full ordinary suites and shuffled run also pass. (E/'execution-verdict.json').write_text(json.dumps({'verdict':verdict,'failures':failures},indent=2))
# Compress only large finished logs; retain byte integrity proof.
compressed={}
for p in E.glob('*.stdout.log.txt'):
 if p.stat().st_size>1000000:
  raw=p.read_bytes();gzip.open(str(p)+'.gz','wb').write(raw)
  assert gzip.open(str(p)+'.gz','rb').read()==raw
  compressed[p.name]={'sha256':hashlib.sha256(raw).hexdigest(),'bytes':len(raw),'gzip':p.name+'.gz'};p.unlink()
(E/'compression.json').write_text(json.dumps(compressed,indent=2))
print(json.dumps({'verdict':verdict,'failures':failures,'pairs':Pairs,'race':R['race']['all_seven_pass'],'shuffle':R['shuffle']['all_seven_pass'],'contracts':R['contracts']},indent=2))
