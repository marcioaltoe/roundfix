import subprocess,re,json,tarfile,io,hashlib
from pathlib import Path
R=Path.cwd(); S=R/'docs/specs/0254-a-faster-make-test'; E=S/'qa/evidence/2026-10-08-final'
def git(*args): return subprocess.check_output(['git','-c','core.fsmonitor=false',*args],text=True)
def section(text,name):
 m=re.search(r'^## '+re.escape(name)+r'\s*\n(.*?)(?=^## |\Z)',text,re.M|re.S)
 return m.group(1) if m else ''
tasks={n:(S/(n+'.md')).read_text() for n in ['task_01','task_02','task_03','task_04','task_06']}
context={n:set(re.findall(r'^- (?:instruction|interface|creates): `([^`]+)`',section(t,'Context'),re.M)) for n,t in tasks.items()}
recorded={n:set(re.findall(r'^- `([^`]+)`',section(t,'Recorded paths'),re.M)) for n,t in tasks.items()}
commits={}
for line in git('log','--format=%H','2ce5abe8..HEAD').splitlines():
 body=git('show','-s','--format=%B',line)
 match=re.search(r'Roundfix-Task: (task_\d+)',body)
 if match:commits[match.group(1)]=line
exceptions=context['task_03']|context['task_04']
seams={p for p in context['task_06'] if p.startswith('internal/cli/') and p.endswith('.go') and not p.endswith('_test.go')}
granted=set(re.findall(r'^  - (.+)',section((S/'_authorization.md').read_text(),'no such section'),re.M))
auth=(S/'_authorization.md').read_text();granted=set(re.findall(r'^  - (.+)',auth.split('paths:\n')[1].split('operations:')[0],re.M))
governed={'docs/agents/specific-repository.md','internal/baseline/derived_ownership_test.go','internal/cli/cli_test.go','internal/spec/coverage_test.go','internal/speccheck/mechanical_test.go'}
packages={'task_01':{'cli'},'task_02':{'daemon','spec','speccheck','worktree','store','baseline'},'task_06':{'cli'}}
def normalize(src):
 src=re.sub(r'\b\w+\.Parallel\(\)\s*;?', '',src)
 src=re.sub(r'// Sequential:[^\n]*','',src)
 return re.sub(r'\s+','',src)
audit={}
for n,c in sorted(commits.items()):
 paths=git('diff-tree','--no-commit-id','--name-only','-r',c).splitlines()
 (E/(n+'.patch.txt')).write_text(git('show','--format=fuller','--no-ext-diff',c))
 undeclared=[p for p in paths if p not in context[n]|recorded[n]|{str(S.relative_to(R)/(n+'.md'))}]
 invalid=[p for p in recorded[n] if not (p.endswith('_test.go') and len(p.split('/'))==3 and p.split('/')[1] in packages.get(n,set()) or n=='task_06' and p in seams)]
 prohibited=[p for p in paths if p in ['Makefile','go.mod','go.sum','.roundfixrc.yml'] or p.startswith('.github/workflows/') or p.endswith('.go') and not p.endswith('_test.go') and p!='internal/verifyselect/contracts.go' and not(n=='task_06' and p in seams)]
 changed_meaning=[]
 if n in ['task_01','task_02']:
  for p in paths:
   if not p.endswith('_test.go') or p in exceptions:continue
   try:old=git('show',c+'^:'+p)
   except subprocess.CalledProcessError:continue
   new=git('show',c+':'+p)
   if normalize(old)!=normalize(new):changed_meaning.append(p)
 audit[n]={'commit':c,'paths':paths,'undeclared':undeclared,'recorded_count':len(recorded[n]),'invalid_recorded':invalid,'prohibited':prohibited,'conversion_meaning_changes':changed_meaning,'governed_changed':sorted(set(paths)&governed),'governed_unbounded':sorted((set(paths)&governed)-granted)}
# Supplemental maintainer repair changes three timing-sensitive tests back to Sequential,
# lowers/raises the final ceiling to its authorized <=12 bound, and adds scope/coverage records.
head=git('rev-parse','HEAD').strip();repairpaths=git('diff-tree','--no-commit-id','--name-only','-r',head).splitlines()
(E/'maintainer-repair.patch.txt').write_text(git('show','--format=fuller','--no-ext-diff',head))
audit['maintainer_repair']={'commit':head,'paths':repairpaths,'declared_task06_paths':[p for p in repairpaths if p in context['task_06']],'description':'Explicit repair of QA F3/F4, outside the Task commits; inspect patch without reclassifying it as a Task commit.'}
(E/'scope-audit.json').write_text(json.dumps(audit,indent=2))
print(json.dumps({n:{k:v for k,v in a.items() if k not in ['paths']} for n,a in audit.items()},indent=2))
glossary=(R/'CONTEXT.md').read_text();(E/'glossary.json').write_text(json.dumps({term:bool(re.search(r'\*\*'+term+r'\*\*:\s*\n.+\n_Avoid_',glossary)) for term in ['Sequential Test','Parallel Test Package']},indent=2))
# Independently compare materialized inventories using the pinned public Git read path.
revision=re.search(r'PinnedHistoryRevision\s*=\s*"([^"]+)"',(R/'internal/gittest/pinned_history.go').read_text()).group(1)
inventories=[]
for pattern in ['docs/history/specs',':(glob)docs/history/specs/*/qa/qa-report-*.md']:
 raw=subprocess.check_output(['git','-c','core.fsmonitor=false','archive','--format=tar',revision,pattern])
 with tarfile.open(fileobj=io.BytesIO(raw)) as t:files=sorted(x.name for x in t.getmembers() if x.isfile())
 qa=[p for p in files if re.search(r'/qa/qa-report-[^/]+\.md$',p)]
 inventories.append({'pattern':pattern,'regular_files':len(files),'qa_reports':qa,'qa_reports_hash':hashlib.sha256('\n'.join(qa).encode()).hexdigest()})
(E/'archive-equivalence.json').write_text(json.dumps({'revision':revision,'same_nonempty_report_set':bool(inventories[0]['qa_reports']) and inventories[0]['qa_reports']==inventories[1]['qa_reports'],'inventories':inventories},indent=2))
