import os, subprocess, pathlib, tempfile, shutil, json
ROOT=pathlib.Path.cwd()
EV=ROOT/'docs/specs/0243-a-history-that-holds-only-records/qa/evidence/2026-10-07-terminal'
REPORT=EV.parent.parent/'qa-report-2026-10-07.md'
tmp=pathlib.Path(tempfile.mkdtemp(prefix='qa0243-ablation-',dir='/private/tmp'))
repo=tmp/'clone';home=tmp/'home';home.mkdir()
env={'PATH':os.environ['PATH'],'HOME':str(home),'TMPDIR':os.environ.get('TMPDIR','/tmp'),'GOMODCACHE':'/Users/marcio/go/pkg/mod','GOCACHE':'/private/tmp/qa0243-gocache','RUSTUP_HOME':'/Users/marcio/.rustup','CARGO_HOME':'/Users/marcio/.cargo','GIT_CONFIG_NOSYSTEM':'1','GIT_CONFIG_GLOBAL':'/dev/null'}
def run(args,name,expected=0,cwd=None):
    cwd=cwd or repo
    with (EV/(name+'.stdout.txt')).open('wb') as out,(EV/(name+'.stderr.txt')).open('wb') as err:
        r=subprocess.run(args,cwd=cwd,env=env,stdout=out,stderr=err)
    (EV/(name+'.command.txt')).write_text(json.dumps({'argv':args,'cwd':str(cwd),'exit':r.returncode},indent=2)+'\n')
    print(name,'exit',r.returncode,flush=True)
    if expected is not None and r.returncode!=expected:raise RuntimeError(name+' exited '+str(r.returncode))
    return (EV/(name+'.stdout.txt')).read_text()
def size():return sum(p.stat().st_size for p in (repo/'docs/history').rglob('*') if p.is_file())
try:
    run(['git','clone','--no-local',str(ROOT),str(repo)],'R06-clone',cwd=tmp)
    run(['git','config','user.name','QA disposable clone'],'R06-name')
    run(['git','config','user.email','qa@example.invalid'],'R06-email')
    run(['git','config','core.fsmonitor','false'],'R06-fsmonitor')
    run(['git','config','core.excludesFile',str(home/'ignore')],'R06-ignore')
    (home/'ignore').write_text('/bin/\n')
    (repo/'bin').mkdir(exist_ok=True);os.symlink(ROOT/'bin/roundfix',repo/'bin/roundfix')
    head=run(['git','rev-parse','HEAD'],'R06-head').strip()
    before=size()
    run(['git','tag','-a','history-full','-m','Full history in QA clone'],'R06-tag')
    out=run(['./bin/roundfix','history','sanitize','--apply','--batch','300'],'R06-apply')
    after=size()
    assert after<1000000 and head=='1048ca2f2b2bfdc6ef6bfecd93277bade741860e'
    record=next((repo/'docs/history/specs').glob('0003*.md'))
    data=record.read_text();assert 'disposition: no-qa' in data
    (EV/'R06-0003-no-qa-record.md').write_text(data)
    run(['go','test','-count=1','-v','-tags','docscontract','-run','^TestSecondbrainExportFollowsTheHistoryForm$','./internal/docscontract'],'R07-clone-before',expected=1)
    export=repo/'.secondbrain-export'
    export.write_text('\n'.join(x for x in export.read_text().splitlines() if not x.startswith('!docs/history'))+'\n')
    run(['go','test','-count=1','-v','-tags','docscontract','-run','^TestSecondbrainExportFollowsTheHistoryForm$','./internal/docscontract'],'R07-clone-after')
    run(['git','add','--all'],'R06-stage')
    run(['git','commit','-m','chore: sanitize history in disposable QA clone'],'R06-commit')
    summary={'source_head':head,'bytes_before':before,'bytes_after':after,'apply':out.strip(),'make_verify':None,'make_verify_docs':None,'clone_removed':False}
    (EV/'R06-summary.json').write_text(json.dumps(summary,indent=2)+'\n')
    run(['make','verify'],'R06-make-verify',expected=None)
    run(['make','verify-docs'],'R06-make-verify-docs',expected=None)
    summary['make_verify']=json.loads((EV/'R06-make-verify.command.txt').read_text())['exit']
    summary['make_verify_docs']=json.loads((EV/'R06-make-verify-docs.command.txt').read_text())['exit']
    (EV/'R06-summary.json').write_text(json.dumps(summary,indent=2)+'\n')
    # Gate builds may replace the symlink; persistence CLI uses the Run-worktree binary again.
    (repo/'bin/roundfix').unlink();os.symlink(ROOT/'bin/roundfix',repo/'bin/roundfix')
    # Reads after the gates independently confirm fully reduced state persists.
    run(['./bin/roundfix','history','sanitize'],'R06-final-plan')
    assert summary['make_verify']==0 and summary['make_verify_docs']==0,summary
    s=REPORT.read_text().replace('| R06 | pending |','| R06 | pass |')
    a=s.index('### R06\n');b=s.index('\n### R07',a)
    detail='Evidence: `evidence/2026-10-07-terminal/R06-*.txt` and `R06-summary.json`. Disposable no-local clone at audited HEAD: '+out.strip()+f' Bytes {before:,} → {after:,}; make verify and make verify-docs both exit 0. Export contract fails (exit 1) before exclusion removal and passes (exit 0) after. 0003-dogfood-polish has no-qa disposition. Final fresh CLI plan confirms zero pending; clone removed after evidence capture.'
    s=s[:a]+s[a:b].replace('Evidence: pending.',detail)+s[b:];REPORT.write_text(s)
finally:
    shutil.rmtree(tmp)
    p=EV/'R06-summary.json'
    if p.exists():
        data=json.loads(p.read_text());data['clone_removed']=not tmp.exists();p.write_text(json.dumps(data,indent=2)+'\n')
    print('Disposable clone removed',not tmp.exists(),flush=True)
