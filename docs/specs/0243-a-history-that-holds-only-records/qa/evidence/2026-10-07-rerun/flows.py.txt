def run(args, cwd, env, name, expected=0):
    r = subprocess.run(args, cwd=cwd, env=env, capture_output=True)
    (EV / (name+'.stdout.txt')).write_bytes(r.stdout)
    (EV / (name+'.stderr.txt')).write_bytes(r.stderr)
    (EV / (name+'.command.txt')).write_text(json.dumps({'argv':args, 'cwd':str(cwd), 'exit':r.returncode}, indent=2)+'\n')
    assert r.returncode == expected, (name, r.returncode, r.stderr.decode(errors='replace'))
    return r.stdout.decode(errors='replace')

def mark(row, detail, status='pass'):
    s = REPORT.read_text()
    s = s.replace('| '+row+' | pending |', '| '+row+' | '+status+' |')
    a=s.index('### '+row+'\n'); b=s.find('\n### ',a+1)
    if b<0:b=len(s)
    s=s[:a]+s[a:b].replace('Evidence: pending.', detail)+s[b:]
    REPORT.write_text(s)
    print(row, status, detail, flush=True)

def snapshot(repo):
    return {str(p.relative_to(repo)):hashlib.sha256(p.read_bytes()).hexdigest() for p in repo.rglob('*') if p.is_file() and '.git' not in p.relative_to(repo).parts and 'bin' not in p.relative_to(repo).parts}

def write(repo, name, data):
    p=repo/name;p.parent.mkdir(parents=True,exist_ok=True);p.write_text(data)
