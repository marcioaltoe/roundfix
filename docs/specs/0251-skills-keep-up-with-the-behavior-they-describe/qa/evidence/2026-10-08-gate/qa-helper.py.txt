def run(name,args,cwd=ROOT,expected=None,custom=None):
 e=env.copy();e.update(custom or {})
 t=time.monotonic();p=subprocess.run(args,cwd=cwd,env=e,text=True,capture_output=True)
 data={'command':shlex.join(args),'cwd':str(cwd),'exit':p.returncode,'wall_seconds':round(time.monotonic()-t,3),'stdout':p.stdout,'stderr':p.stderr}
 (EV/(name+'.json')).write_text(json.dumps(data,indent=2)+'\n')
 print(name,'exit',p.returncode,'wall',data['wall_seconds'],flush=True)
 if expected is not None: assert p.returncode==expected,(name,data)
 return p

def settle(row,status,detail):

 start=s.index('### '+row+'\n');pos=s.index('Pending execution.',start)
 s=s[:pos]+detail+s[pos+len('Pending execution.'):];REPORT.write_text(s)
def git(cwd,*args):
