def run(args, cwd=ROOT, extra=None):
    environ = ENV.copy()
    environ.update(extra or {})
    return subprocess.run(args, cwd=cwd, env=environ, capture_output=True, text=True)

def save(name, args, result, note=""):
    text = "$ " + shlex.join(args) + "\nexit: " + str(result.returncode) + "\nstdout:\n" + result.stdout + "\nstderr:\n" + result.stderr + "\n" + note
    (EVIDENCE / name).write_text(text)
    print(name + ": exit " + str(result.returncode), flush=True)

def mark(row, status):
    text = REPORT.read_text()
    text, count = re.subn(r"(?m)^\| " + re.escape(row) + r" \| [^|]+ \|", "| " + row + " | " + status + " |", text)
    assert count == 1, (row, count)
    REPORT.write_text(text)

def test(name, pattern, cwd=ROOT):
    args = ["go", "test", "-count=1", "-timeout", "20m", "-v", "-run", pattern, "./internal/cli"]
    result = run(args, cwd)
    save(name, args, result)
    return result

def archive(revision, destination):
    result = subprocess.run(["git", "archive", revision], cwd=ROOT, env=ENV, capture_output=True)
    assert result.returncode == 0
    with tarfile.open(fileobj=io.BytesIO(result.stdout)) as source:
        source.extractall(destination, filter="data")
    # git archive excludes .git; suiteguard fingerprints the disposable tree.
    # Initialize and index that tree without creating any commit.
    for args in [["git", "init", "--quiet"], ["git", "add", "--all"]]:
        initialized = run(args, destination)
        assert initialized.returncode == 0, initialized.stderr
