import sys,datetime
sys.path.insert(0,'/private/tmp/rf273-rerun-s1suy30p');from rf273_lib import *
ARGS=['review','dispose','F1','--dismiss','--evidence','not a defect']
def setup(name):
 p=repo(name);art=TMP/(name+'-artifacts');art.mkdir();(p/'.roundfixrc.yml').write_text('defaults:\n  artifact_dir: '+str(art)+'\n');(p/'review.txt').write_text('stable anchor\nneeds fix\n');reviewed=commit(p,'reviewed two findings')
 items=[{'id':'F1','text':'review.txt:1: Failure: claimed defect','anchor':{'path':'review.txt','startLine':1,'endLine':1},'validation':{'status':'stands'}},{'id':'F2','text':'review.txt:2: Failure: needs correction','anchor':{'path':'review.txt','startLine':2,'endLine':2},'validation':{'status':'stands'}}]
 record={'repository':str(p),'baseCommit':ok(['git','rev-parse','HEAD~1'],p),'headCommit':reviewed,'provider':'codex','source':'project','outcome':'findings','findings':'\n'.join(x['text'] for x in items),'findingItems':items}
 dest=art/'pre-pr-review'/hashlib.sha256(str(p).encode()).hexdigest()[:16]/'pre-pr-review.json';dest.parent.mkdir(parents=True);dest.write_text(json.dumps(record));return p,art,reviewed,record

def accepted():
 p,art,reviewed,rec=setup('dismiss-after-fix-v2');(p/'review.txt').write_text('stable anchor\ncorrected\n');head=commit(p,'fix second finding');log=[]
 r=logcli(log,p,['review','dispose','F2','--fixed-by',head]);assert r.returncode==0 and not r.stderr
 ledger=art/'pre-pr-review-dispositions.jsonl';before=ledger.read_bytes();r=logcli(log,p,ARGS);assert r.returncode==0 and not r.stderr;entry=json.loads(r.stdout);assert entry=={'repository':str(p),'headCommit':reviewed,'finding':'F1','text':rec['findingItems'][0]['text'],'disposition':'dismissed','evidence':'not a defect','recordedAt':entry['recordedAt'],'currentHead':head};assert ledger.read_bytes()==before+r.stdout.encode();assert list(entry)==['repository','headCommit','finding','text','disposition','evidence','recordedAt','currentHead']
 saved=ledger.read_bytes();r=logcli(log,p,ARGS);assert r.returncode==2 and 'already has a disposition' in r.stderr and ledger.read_bytes()==saved
 log.append('Independent duplicate-command read and ledger reopen preserve reviewed head, currentHead and exactly one appended F1; double submission adds nothing.')
 finish('06',log,'Surface Transcript 4 matches full JSON field order, empty stderr and exit 0. Fixed F2 then dismissed F1 after its fix; fresh CLI duplicate refusal and ledger reread preserve exactly one dismissal.')

def refused(name,body,row):

 r=logcli(log,p,ARGS);assert r.returncode==2 and r.stderr==reason and ledger.read_bytes()==before
 log.append('Repeated refusal and independent ledger reopen: bytes unchanged.')
 if row:finish(row,log,'Surface Transcript 5 matches exactly. Changed anchor refused twice without ledger mutation.')
 return log

def reviewed_base():

accepted();refused('dismiss-changed-anchor','changed anchor\ncorrected\n','07');reviewed_base()
