def run(args,cwd=ROOT,env=None,allowed=(0,)):
    p=subprocess.run(args,cwd=cwd,env=env,text=True,capture_output=True)
    LOG.append('command: '+shlex.join([str(a) for a in args])+'\ncwd: '+str(cwd)+'\nstdout:\n'+p.stdout+'stderr:\n'+p.stderr+'\nexit: '+str(p.returncode)+'\n')
    if p.returncode not in allowed:raise AssertionError('unexpected exit '+str(p.returncode)+' from '+shlex.join(args)+'\n'+p.stdout[-1800:]+p.stderr[-1800:])
    return p

def git(*args,cwd=ROOT):return run(['git',*args],cwd=cwd).stdout

def observe(message):LOG.append(message+'\n')

def archive(rev,dest):
    p=subprocess.run(['git','archive',rev],cwd=ROOT,capture_output=True,check=True)
    with tarfile.open(fileobj=io.BytesIO(p.stdout)) as tf:tf.extractall(dest,filter='data')
    observe('Fresh git archive '+rev+' extracted; no earlier Run artifact used.')
    git('init','-q',cwd=dest);git('add','-A',cwd=dest)

def historical(commit):
    c=git('rev-parse',commit).strip()
    deleted=git('diff','--name-only','--diff-filter=D',c+'^',c,'--','docs/backlog','docs/findings').splitlines()
    changes=git('diff','--name-status',c+'^',c).splitlines()
    found=[]
    for path in deleted:
        pairs=[x.split('\t') for x in changes if re.match(r'R\d+\t',x) and x.split('\t')[1]==path and '/references/' in x.split('\t')[2]]
        for pair in pairs:
            previous=git('show',c+'^:'+path);status=re.search(r'^status: .+$',previous,re.M).group()
            current=git('show',c+':'+pair[2]);now=re.search(r'^status: .+$',current,re.M).group()
            observe('Historical comparison: '+path+' is deleted in path-limited listing, '+pair[0]+' to '+pair[2]+' without path limit; parent '+status+'; adopted copy '+now+'.')
            assert status in ['status: open','status: pending']
            assert now in ['status: promoted','status: done','status: partial']
            found.append((path,pair[2],status,now))
    assert found, 'no claimed adopted record in '+commit
    return found
