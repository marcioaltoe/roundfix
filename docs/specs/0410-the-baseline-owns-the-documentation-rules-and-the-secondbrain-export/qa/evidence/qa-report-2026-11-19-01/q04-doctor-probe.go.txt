package cli
import ("os"; "path/filepath"; "strings"; "testing")
func TestQADocumentationHead(t *testing.T) {
 root, err := os.Getwd(); if err != nil { t.Fatal(err) }; root=filepath.Clean(filepath.Join(root,"../.."))
 results := doctorDocumentationResults(root)
 if len(results)!=2 { t.Fatalf("%+v",results) }
 for _,r := range results {
  t.Logf("%s: %s (%s)",r.Name,r.Status,r.Detail)
  if r.Status!=CheckStatusOK { t.Fatalf("%+v",r) }
  if r.Name==HealthCheckBaselineExport && !strings.HasPrefix(r.Detail,"the Standard Export List is current; ") {t.Fatal(r.Detail)}
  if r.Name==HealthCheckBaselineDocs && r.Detail!="the documentation follows the docs-layout guide" {t.Fatal(r.Detail)}
 }
}
