from pathlib import Path
import subprocess,os,re,json,shlex,sys
ROOT=Path(__file__).resolve().parents[6]
SPEC=ROOT/'docs/specs/0410-the-baseline-owns-the-documentation-rules-and-the-secondbrain-export'
EV=SPEC/'qa/evidence/qa-report-2026-11-19'
REPORT=SPEC/'qa/qa-report-2026-11-19.md'
HEAD='357ae20690b9438cfb48c8cd5f87cb4a85162741'
def run(name,args,cwd=ROOT,env=None,expected=0):
 p=subprocess.run(args,cwd=cwd,env=env,stdin=subprocess.DEVNULL,stdout=subprocess.PIPE,stderr=subprocess.STDOUT,text=True)
 text='$ '+shlex.join(map(str,args))+'\n'+p.stdout+'\nexit: '+str(p.returncode)+'\n'
 (EV/(name+'.txt')).write_text(text)
 print(name, 'exit',p.returncode, 'passes',len(re.findall(r'--- PASS:',p.stdout)),flush=True)
 return p

def note(row,text,status=None):
 s=REPORT.read_text();marker='### '+row+'\n'; start=s.index(marker);end=s.find('\n### ',start+len(marker));end=len(s) if end<0 else end
 s=s[:end]+'\n'+text+'\n'+s[end:]
 if status:s=re.sub(r'(?m)^\| '+re.escape(row)+r' \| [^|]+ \|', '| '+row+' | '+status+' |',s)
 REPORT.write_text(s)
def link(name):return '[%s](evidence/qa-report-2026-11-19/%s.txt)'%(name,name)

def tests():
 (ROOT/'bin/qa').mkdir(exist_ok=True)
 cmds=[('baseline-suite','./internal/baseline',[],'Q02'),('spec-suite','./internal/spec',[],'Q03'),('releaseplan-suite','./internal/releaseplan',[],'Q03'),('cli-suite','./internal/cli',['-run','Baseline|GuidanceComposition|SetupManifest|Doctor|UpdateCheck|ReleasePlan'],'Q03'),('skills-suite','./skills',[],'Q05'),('docscontract-suite','./internal/docscontract',['-tags','docscontract'],'Q06')]
 outcomes={}
 for name,pkg,extra,row in cmds:
  binary='bin/qa/'+name+'.test'
  p=run(name,['go','test','-count=1','-timeout','40m','-v','-o',str(ROOT/binary),*extra,pkg]);outcomes[name]=p.returncode
  s=REPORT.read_text().replace('  - bin/roundfix\n','  - bin/roundfix\n  - '+binary+'\n',1);REPORT.write_text(s)
  note(row,'Observed '+link(name)+'; exit '+str(p.returncode)+'.')
 for name,args in [('skills-sync-check',['make','skills-sync-check']),('verify-docs',['make','verify-docs'])]:
  p=run(name,args);outcomes[name]=p.returncode;note('Q05' if name=='skills-sync-check' else 'Q06','Observed '+link(name)+'; exit '+str(p.returncode)+'.')
 (EV/'test-outcomes.txt').write_text(json.dumps(outcomes,indent=2)+'\n')
if __name__=='__main__' and len(sys.argv)>1 and sys.argv[1]=='tests':tests()
