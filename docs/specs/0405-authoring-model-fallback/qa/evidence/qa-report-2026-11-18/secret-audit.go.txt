package main

import (
 "fmt"
 "os"
 "path/filepath"
 "roundfix/internal/secretscan"
)

func main() {
 count, matches := 0, 0
 for _, root := range os.Args[1:] {
  err := filepath.WalkDir(root, func(path string, entry os.DirEntry, err error) error {
   if err != nil { return err }
   if entry.IsDir() { return nil }
   data, err := os.ReadFile(path); if err != nil { return err }
   count++
   for _, m := range secretscan.ScanContent(path, data) { fmt.Println(m.String()); matches++ }
   return nil
  })
  if err != nil { fmt.Fprintln(os.Stderr, err); os.Exit(2) }
 }
 fmt.Printf("Scanned %d files; Secret Matches: %d\n", count, matches)
 if matches > 0 { os.Exit(1) }
}
