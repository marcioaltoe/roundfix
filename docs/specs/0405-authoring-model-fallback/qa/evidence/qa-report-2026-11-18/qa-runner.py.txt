def run(args,cwd=ROOT,env=None,code=None,protect=False):
    global CURRENT
    actual=list(map(str,args))
    if protect:
        profile=pathlib.Path(env['HOME'])/'qa.sb'
        profile.write_text('(version 1)\n(allow default)\n(deny network*)\n(deny file-write*)\n(allow file-write* (subpath '+json.dumps(str(cwd)) +') (subpath '+json.dumps(env['HOME'])+') (literal "/dev/null"))\n(deny file-read* (subpath "/Users/marcio"))\n')
        actual=['/usr/bin/sandbox-exec','-f',str(profile)]+actual
    p=subprocess.run(actual,cwd=cwd,env=env or ENV,text=True,capture_output=True)
    if CURRENT:
        with (E/(CURRENT+'.txt')).open('a') as log:
            log.write('cwd: '+str(cwd)+'\n$ '+shlex.join(list(map(str,args)))+'\nstdout:\n'+p.stdout+'stderr:\n'+p.stderr+'exit: '+str(p.returncode)+'\n\n')
    if code is not None:assert p.returncode==code, f'exit {p.returncode}, expected {code}: {p.stderr or p.stdout}'
    return p

def transcript_matches(pattern,text):
    lines=pattern.splitlines();obs=text.splitlines()
    def recur(i,j):
        if i==len(lines):return j==len(obs)
        if lines[i]=='...':return any(recur(i+1,k) for k in range(j,len(obs)+1))
        if j==len(obs):return False
        parts=re.split(r'(<[^>]+>)',lines[i]);exp=''.join('.+' if p.startswith('<') and p.endswith('>') else re.escape(p) for p in parts)
        return re.fullmatch(exp,obs[j]) is not None and recur(i+1,j+1)
    assert recur(0,0),'transcript mismatch: '+repr(text)

def filetree(repo,prefixes):
    result={}
    for prefix in prefixes:
        base=repo/prefix
        if not base.exists():continue
        for p in sorted(base.rglob('*')):
            key=str(p.relative_to(repo))
            if p.is_symlink():result[key]=('link',os.readlink(p))
            elif p.is_file():result[key]=('file',hashlib.sha256(p.read_bytes()).hexdigest())
    return result
