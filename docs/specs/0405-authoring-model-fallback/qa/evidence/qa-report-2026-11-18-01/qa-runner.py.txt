def bareenv(home):
    e={k:v for k,v in os.environ.items() if k in ('PATH','TMPDIR','LANG','LC_ALL','RUSTUP_HOME','CARGO_HOME')}
    e.update(HOME=str(home), XDG_CONFIG_HOME=str(home/'config'), XDG_CACHE_HOME=str(home/'cache'), XDG_DATA_HOME=str(home/'data'), GIT_CONFIG_NOSYSTEM='1',GIT_CONFIG_GLOBAL='/dev/null',GIT_TERMINAL_PROMPT='0',GOPROXY='off',GOSUMDB='off',GOTOOLCHAIN='local',CLAUDE_CODE_SESSION_ID='qa-0405-01')
    return e

def logrun(args,cwd,e,log,expected=None,policy=None):
    cmd=list(map(str,args))
    invoke=['/usr/bin/sandbox-exec','-f',str(policy),*cmd] if policy else cmd
    r=subprocess.run(invoke,cwd=cwd,env=e,text=True,stdout=subprocess.PIPE,stderr=subprocess.PIPE)
    with (EVID/log).open('a') as f:
        f.write('$ '+__import__('shlex').join(cmd)+'\nstdout:\n'+r.stdout+'stderr:\n'+r.stderr+'exit: '+str(r.returncode)+'\n')
    if expected is not None: assert r.returncode==expected,(cmd,r.returncode,r.stdout,r.stderr)
    return r

def finish(rid,status,detail):

def installed(repo):
    result={}
    for folder in ('.agents','.claude'):
        for p in (repo/folder).rglob('*'):
            if p.is_symlink():result[p.relative_to(repo).as_posix()]=('link',os.readlink(p))
            elif p.is_file():result[p.relative_to(repo).as_posix()]=('file',p.read_bytes())
    return result

def closure():

        e['GIT_DIR']=str(Path(gd).resolve());e['GIT_WORK_TREE']=str(ROOT)
        common=subprocess.check_output(['git','rev-parse','--path-format=absolute','--git-common-dir'],cwd=ROOT,env=e,text=True).strip()

        policy.write_text('(version 1)\n(allow default)\n(deny network*)\n(deny file-read* (subpath '+json.dumps(str(REALHOME))+'))\n(deny file-write*)\n(allow file-write* (subpath '+json.dumps(str(home))+'))\n(allow file-write* (literal "/dev/null"))\n')
        # The repository's .git file names its parent checkout under the user home.
        # Allow only that repository Git administration, never user tool state.
        with policy.open('a') as pf:
            for gitpath in {common,str(Path(common).resolve()),gd,str(Path(gd).resolve())}:
                pf.write('(allow file-read* (subpath '+json.dumps(gitpath)+'))\n')
        note('closure.txt','Harness correction: allow this repository Git administration named by its .git file; real user tool state and instruction files remain denied, as do network and worktree writes.')
        note('closure.txt','Auxiliary qa-report acceptance remains unavailable under the self-imposed policy: parent Git path metadata denied. No further retry; independent structure/link checks below remain runnable.')
        ar=logrun(['./bin/roundfix','archive',SPEC,'--plan','--offline'],ROOT,e,'closure.txt',None,policy)
        note('closure.txt','Offline archive preview exit '+str(ar.returncode)+'. Preview only; no archive, judge, provider or network call.')

        shutil.copyfile(EVID/'secret-audit.go.txt',src/'main.go')

def audit_report(e):

    while not lines[start].startswith('| # |'):start+=1
    end=start
    while end<len(lines) and lines[end].startswith('|'):end+=1
    rows=[line.split('|')[1:4] for line in lines[start+2:end]]
    assert len(rows)==22,len(rows)
    assert len({r[0].strip() for r in rows})==22
    statuses=[r[1].strip() for r in rows]
    assert sum(s.startswith('carried (') for s in statuses)==15
    assert statuses.count('pass')==6
    assert statuses.count('blocked (environment: no open Pull Request)')==1
    assert all(s=='pass' or s.startswith('carried (') or s=='blocked (environment: no open Pull Request)' for s in statuses)
    for rid in ['Q03-record','Q04-isolation','Q05-base','Q11-config','Q12-baseline','V01','PR01']:
        chunk=t.split('### '+rid+'\n',1)[1].split('\n### ',1)[0]
        assert '```yaml\ninputs:\n  - kind:' in chunk,rid
    for rid in ['Q05-base','Q11-config','PR01']:
        assert 'kind: commit_range' in t.split('### '+rid+'\n',1)[1].split('\n### ',1)[0]
    assert t.count('\n---\n')==1
    assert 'verdict: partial' in t and 'status: closed' in t and 'rows_blocked_environment: 1' in t
    assert 'evidence_snapshots:' not in t
    assert 'auditing_binary: "0.147.1 (35cad72e, built 2026-11-18 10:56:25 -0300)"' in t

    links=re.findall(r'\]\(([^)]+)\)',t)
