// Fresh QA 0405 rerun scanner: imports the current audited repository scanner.
package main
import (
 "fmt"
 "os"
 "path/filepath"
 "roundfix/internal/secretscan"
)
func main() {
 count, hits := 0, 0
 for _, root := range os.Args[1:] {
  err := filepath.WalkDir(root, func(path string, d os.DirEntry, err error) error {
   if err != nil {return err}; if d.IsDir() {return nil}
   data, err := os.ReadFile(path); if err != nil {return err}; count++
   for _, match := range secretscan.ScanContent(path,data) {fmt.Println(match.String()); hits++}
   return nil
  }); if err != nil {fmt.Fprintln(os.Stderr,err);os.Exit(2)}
 }
 fmt.Printf("Scanned %d files; Secret Matches: %d\n",count,hits)
 if hits>0 {os.Exit(1)}
}
