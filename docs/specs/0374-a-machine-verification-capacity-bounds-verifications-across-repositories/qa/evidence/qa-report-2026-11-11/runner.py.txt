def run(args, cwd=ROOT, environ=None, timeout=2700):
    p = subprocess.run(args, cwd=cwd, env=environ, text=True, capture_output=True, timeout=timeout)
    return p, 'COMMAND: ' + ' '.join(map(str,args)) + '\nEXIT: ' + str(p.returncode) + '\nSTDOUT:\n' + p.stdout + '\nSTDERR:\n' + p.stderr

def save(name, text):
    (EVIDENCE / name).write_text(text)

def status(row, value):
    text = REPORT.read_text()
    text, n = re.subn(r'(?m)^\| ' + re.escape(row) + r' \| pending \|', '| ' + row + ' | ' + value + ' |', text)
    assert n == 1, (row,n)
    REPORT.write_text(text)

@contextlib.contextmanager
def isolated():
    with tempfile.TemporaryDirectory() as home, tempfile.TemporaryDirectory() as tmp:
        e = os.environ.copy()
        for key in list(e):
            if key.endswith('_KEY') or 'TOKEN' in key or key.startswith('ROUNDFIX_'):
                e.pop(key, None)
        # Preserve GIT_CONFIG_KEY_n: those are Git configuration names, not credentials.
        for key in ['GOCACHE', 'GOMODCACHE']:
            e[key] = subprocess.check_output(['go', 'env', key], text=True).strip()
        e.update(HOME=home, TMPDIR=tmp, PATH=str(ROOT / 'bin/qa0374') + os.pathsep + e['PATH'])
        yield Path(home), Path(tmp), e

def cli(args, cwd=ROOT, environ=None):
    # Keep the public command spelling ./bin/roundfix, including from scratch repos.
    return run(['./bin/roundfix'] + args, cwd=cwd, environ=environ)

def scratch(tmp, e):
    repo = tmp / 'repo'; repo.mkdir()

    assert p.returncode == 0, log

def snapshot(directory):
    return {str(p.relative_to(directory)): (p.read_bytes(),p.stat().st_size,p.stat().st_mtime_ns)
            for p in directory.rglob('*') if p.is_file()}
