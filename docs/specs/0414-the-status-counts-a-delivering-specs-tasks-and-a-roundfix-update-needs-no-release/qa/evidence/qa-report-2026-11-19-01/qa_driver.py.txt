def run(args,cwd=ROOT,env=None,ok=None):
 p=subprocess.run(args,cwd=cwd,env=env,stdin=subprocess.DEVNULL,stdout=subprocess.PIPE,stderr=subprocess.PIPE,text=True)
 if ok is not None and p.returncode not in ok: raise RuntimeError(f'{args}: exit {p.returncode}\n{p.stdout}\n{p.stderr}')
 return p

def logged(name,args,cwd=ROOT,env=None,ok=(0,)):
