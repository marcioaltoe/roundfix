def command(args,cwd=ROOT,env=None):

def run(row,label,args,cwd=ROOT,env=None,expect=0,failnames=()):
 print(row+' '+label+': running',flush=True)
 p=command(args,cwd,env)
 body='$ '+shlex.join(args)+'\ncwd: '+str(cwd)+'\nstdout:\n'+p.stdout+'\nstderr:\n'+p.stderr+'\nexit: '+str(p.returncode)+'\n'
 ok=p.returncode==expect if expect is not None else p.returncode!=0
 if failnames: ok=ok and all(re.search(r'--- FAIL: '+re.escape(n)+r'(?: |$)',p.stdout+'\n'+p.stderr,re.M) for n in failnames)
 path=EV/(row+'-'+label+'.txt'); path.write_text(body)
 print(row+' '+label+': '+('PASS' if ok else 'FAIL')+' (exit '+str(p.returncode)+')',flush=True)
 return ok,path,p
