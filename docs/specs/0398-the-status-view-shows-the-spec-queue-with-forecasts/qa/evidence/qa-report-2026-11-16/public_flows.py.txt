def product(row,label,args,repo,env,binary=None):

def snap(paths):
 result=[]
 for label,path in paths:
  for p in sorted(path.rglob('*')):
   if p.is_file():data=p.read_bytes();result.append(f'{label}/{p.relative_to(path)} {len(data)} {hashlib.sha256(data).hexdigest()}')
 return '\n'.join(result)+'\n'
