
func TestQARemeasureAdopterFailureShape(t *testing.T) {
 for _, configured := range []bool{false, true} {
  name := "empty"; if configured { name = "configured" }
  t.Run(name, func(t *testing.T) {
   f, p, _ := priorFixture(t)
   prior := recordPriorCommit(t, f, p, priorReportName(1), priorReportContent("fail"), QACommitMessage(p.Spec.Slug, "fail"), true)
   if configured { p.FormatCommand = qaFormatterScript(t, "for path do printf '\\nformatted\\n' >> \"$path\"; done\n") }
   p.RepositoryVerification = `find docs/specs/` + p.Spec.Slug + `/qa -type f -exec sh -c 'for path do grep -aq formatted "$path" || exit 1; done' check {} +`
   runner := &taskFakeRunner{calls: f.calls, gitRoot: f.gitRoot, qaReport: qaReportForTest(spec.VerdictPass)}
   e := f.engine(t, runner, ExecVerifier{}, GitCommitter{}, GitWorktreeSnapshotter{})
   result, err := e.TaskCycle(context.Background(), p); if err != nil { t.Fatal(err) }
   report, err := spec.ReadQAReport(p.Spec.Dir); if err != nil { t.Fatal(err) }
   t.Logf("configured=%t prior_run_commit=%s verdict=%s blocked_precondition=%d agent_prompts=%d", configured, prior.Commit, result.QAVerdict, report.RowsBlockedPrecondition, len(runner.qaPrompts))
   if configured {
    if result.QAVerdict != spec.VerdictPass || report.RowsBlockedPrecondition != 0 || len(runner.qaPrompts) != 1 { t.Fatalf("configured outcome: %+v report=%+v", result, report) }
    for _, path := range prior.Files { if !strings.Contains(qaCommittedBytes(t,p,path),"formatted") { t.Fatalf("unformatted %s",path) } }
    assertQAFormatOutcome(t,f,"commit","formatted",2)
   } else {
    if result.QAVerdict != spec.VerdictFail || report.RowsBlockedPrecondition != 1 || len(runner.qaPrompts) != 0 { t.Fatalf("empty outcome: %+v report=%+v",result,report) }
    if strings.Contains(qaCommittedBytes(t,p,prior.Report),"formatted") { t.Fatal("empty command rewrote imported report") }
    assertQAFormatOutcome(t,f,"","",0)
   }
  })
 }
}
