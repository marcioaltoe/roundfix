def home_snapshot(home):
    result={}
    for path in [Path(home),*sorted(Path(home).rglob('*'))]:
        st=path.lstat(); entry={'mode':st.st_mode,'mtime_ns':st.st_mtime_ns}
        if path.is_symlink():entry['link']=os.readlink(path)
        elif path.is_file():entry['sha256']=hashlib.sha256(path.read_bytes()).hexdigest()
        result[str(path.relative_to(home))]=entry
    return result

def transcript_matches(block,p):
    block=__import__('textwrap').dedent(block).rstrip(' ')
    output=block.split('stdout:\n',1)[1].split('stderr:\n',1)
    wantout=output[0]; wanterr=output[1].rsplit('exit: ',1)[0]
    exitcode=int(output[1].rsplit('exit: ',1)[1].strip())
    def match(pattern,actual):
        tokens=re.split(r'(<[^>]+>)',pattern)
        regex=''.join('[^\\n]+' if re.fullmatch(r'<[^>]+>',x) else re.escape(x) for x in tokens)
        return re.fullmatch(regex,actual) is not None
    return match(wantout,p.stdout) and match(wanterr,p.stderr) and p.returncode==exitcode
