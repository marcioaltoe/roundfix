package main
import (
 "encoding/json"
 "fmt"
 "io/fs"
 "os"
 "path/filepath"
 "roundfix/internal/secretscan"
 "roundfix/internal/spec"
 "roundfix/internal/speccheck"
)
func main() {
 if len(os.Args)<3 { panic("mode and path required") }
 switch os.Args[1] {
 case "governed":
  result:=map[string]bool{}
  for _,p:=range os.Args[2:] { result[p]=speccheck.GovernedPath(p) }
  if err:=json.NewEncoder(os.Stdout).Encode(result);err!=nil {panic(err)}
 case "report":
  root,err:=os.Getwd();if err!=nil{panic(err)}
  p:=os.Args[2]
  report,err:=spec.ReadQAReportFile(p);if err!=nil{panic(err)}
  if err:=spec.QAReportEligibility(filepath.Dir(filepath.Dir(p)),report);err!=nil{panic(err)}
  fmt.Printf("QA eligibility: qualifying %s; environment=%d declared=%d finding=%d\n",report.Verdict,report.RowsBlockedEnvironment,report.RowsBlockedDeclared,report.RowsBlockedFinding)
  findings,err:=speccheck.ReportShapeFindings(root,p);if err!=nil{panic(err)}
  if len(findings)>0 { for _,f:=range findings {fmt.Printf("%+v\n",f)};os.Exit(1) }
  fmt.Println("Report shape and evidence links: pass")
  count:=0
  err=filepath.WalkDir(filepath.Dir(p),func(path string,entry fs.DirEntry,err error)error{
   if err!=nil{return err};if entry.IsDir(){return nil}
   b,err:=os.ReadFile(path);if err!=nil{return err}
   for _,m:=range secretscan.ScanContent(path,b){fmt.Println(m.String());count++}
   return nil
  });if err!=nil{panic(err)}
  fmt.Printf("Secret matches: %d\n",count);if count>0{os.Exit(1)}
 default: panic("unknown mode")
 }
}
