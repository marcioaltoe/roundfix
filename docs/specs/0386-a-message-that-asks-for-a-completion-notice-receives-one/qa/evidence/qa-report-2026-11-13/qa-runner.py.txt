def call(args,cwd=ROOT,env=None):

def checked(args,cwd=ROOT,env=None):

def record(name,args,cwd=ROOT,env=None):
    p=call(args,cwd,env)
    content='command: '+shlex.join(args)+'\nexit: '+str(p.returncode)+'\nstdout:\n'+p.stdout+'\nstderr:\n'+p.stderr
    (E/(name+'.txt')).write_text(content)
    print(name+': exit '+str(p.returncode),flush=True)
    return p
