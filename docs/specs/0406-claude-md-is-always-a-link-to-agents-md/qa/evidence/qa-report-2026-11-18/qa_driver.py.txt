def isolated(home,tmp):
    e=dict(ORIGINAL)
    for k in list(e):
        if k in ("GH_TOKEN","GITHUB_TOKEN","TYPESAFE_API_KEY","CLAUDE_CODE_SESSION_ID") or "API_KEY" in k or k.startswith("AI_MEMORY_") or k in ("GIT_DIR","GIT_WORK_TREE","GIT_INDEX_FILE","GIT_CONFIG_GLOBAL","GIT_AUTHOR_NAME","GIT_AUTHOR_EMAIL","GIT_COMMITTER_NAME","GIT_COMMITTER_EMAIL"):
            e.pop(k,None)
    e.update(HOME=str(home),TMPDIR=str(tmp),GOMODCACHE=MODCACHE,GOCACHE=BUILDCACHE,GOPROXY="off",GOSUMDB="off",GIT_CONFIG_NOSYSTEM="1",GIT_CONFIG_GLOBAL=os.devnull)

    e.setdefault("CARGO_HOME",str(pathlib.Path(ORIGINAL["HOME"])/".cargo"))

def run(cmd,name,cwd=ROOT,env=None,expect=None,only=None):
    with tempfile.TemporaryDirectory(prefix="rf0406-home-") as h, tempfile.TemporaryDirectory(prefix="rf0406-tmp-") as t:
        e=env or isolated(h,t)
        p=subprocess.run([str(x) for x in cmd],cwd=cwd,env=e,stdin=subprocess.DEVNULL,text=True,capture_output=True)
    out=p.stdout if only is None else "\n".join(x for x in p.stdout.splitlines() if only in x)+"\n"
    err=p.stderr if only is None else ""
    (EVID/(name+".txt")).write_text("command: "+" ".join(str(x) for x in cmd)+"\ncwd: "+str(cwd)+"\nstdout:\n"+out+"\nstderr:\n"+err+"\nexit: "+str(p.returncode)+"\n")
    print(name+": exit "+str(p.returncode),flush=True)
    if expect is not None and p.returncode not in expect: raise AssertionError(name+": unexpected exit; see evidence")
    return p
