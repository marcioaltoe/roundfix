package agent
import ("strings"; "testing"; "roundfix/internal/judge")
func TestQA0267AgentCharacterization(t *testing.T) {
 t.Parallel()
 q,err:=judge.Load(); if err!=nil {t.Fatal(err)}
 base:=[]string{"QA_CHARACTERIZATION_CONTROL=present"}
 for _,name:=range q.KeyVariables(){base=append(base,name+"=fixture")}
 output:=acpxCommandEnv(base,nil); count:=0; control:=false
 for _,entry:=range output {name,_,_:=strings.Cut(entry,"="); if name=="QA_CHARACTERIZATION_CONTROL"{control=true}; for _,key:=range q.KeyVariables(){if name==key{count++}}}
 if count!=0 || !control {t.Fatalf("judge count=%d control present=%t",count,control)}
 t.Logf("Agent environment: judge variables=%d; control present=%t; names=%s",count,control,strings.Join(q.KeyVariables(),", "))
}
