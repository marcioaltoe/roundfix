package secretscan
import ("os"; "os/exec"; "path/filepath"; "strings"; "testing"; "bytes"; "bufio"; "io"; "strconv")
func TestQA0267TrackedContentAndHistory(t *testing.T) {
 t.Parallel()
 root,err:=filepath.Abs("../.."); if err!=nil {t.Fatal(err)}
 cmd:=exec.CommandContext(t.Context(),"git","ls-files","-z"); cmd.Dir=root
 paths,err:=cmd.Output(); if err!=nil {t.Fatal(err)}
 count:=0; total:=0
 var refs strings.Builder
 for _,path:=range strings.Split(string(paths),"\x00") {if path!="" {refs.WriteString("HEAD:"+path+"\n")}}
 cmd=exec.CommandContext(t.Context(),"git","cat-file","--batch");cmd.Dir=root;cmd.Stdin=strings.NewReader(refs.String())
 blobs,err:=cmd.Output();if err!=nil {t.Fatal(err)};reader:=bufio.NewReader(bytes.NewReader(blobs))
 for _,path:=range strings.Split(string(paths),"\x00") {if path=="" {continue};total++;header,err:=reader.ReadString('\n');if err!=nil{t.Fatal(err)};parts:=strings.Fields(header);if len(parts)!=3||parts[1]!="blob"{t.Fatal("missing tracked blob",path)};size,err:=strconv.Atoi(parts[2]);if err!=nil{t.Fatal(err)};data:=make([]byte,size);if _,err:=io.ReadFull(reader,data);err!=nil{t.Fatal(err)};if _,err:=reader.ReadByte();err!=nil{t.Fatal(err)};for _,m:=range ScanContent(path,data){count++;t.Log(m.String())}}
 t.Logf("tracked blobs=%d; Secret Matches=%d",total,count)
 if count!=0 {t.Fatal("tracked content carries Secret Matches")}
 if qa:=os.Getenv("QA0267_EVIDENCE_ROOT");qa!="" {files:=0;count:=0;err:=filepath.WalkDir(qa,func(path string,d os.DirEntry,e error)error{if e!=nil{return e};if d.IsDir(){return nil};files++;data,e:=os.ReadFile(path);if e!=nil{return e};for _,m:=range ScanContent(path,data){count++;t.Log(m.String())};return nil});if err!=nil{t.Fatal(err)};t.Logf("QA report/evidence files=%d; Secret Matches=%d",files,count);if count!=0{t.Fatal("QA output carries Secret Matches")}}
}
