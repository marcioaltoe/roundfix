package daemon
import ("context"; "os"; "path/filepath"; "strings"; "testing"; "roundfix/internal/judge")
func TestQA0267VerificationCharacterization(t *testing.T) {
 // Sequential: this overlay characterization sets runtime fixture Judge Key Variables in the isolated test process.
 q,err:=judge.Load(); if err!=nil {t.Fatal(err)}
 checks:=[]string{}
 for _,name:=range q.KeyVariables(){t.Setenv(name,"fixture"); checks=append(checks,`[ "${`+name+`+x}" = "" ]`)}
 t.Setenv("QA_CHARACTERIZATION_CONTROL","present")
 checks=append(checks,`[ "${QA_CHARACTERIZATION_CONTROL+x}" = "x" ]`)
 dir:=t.TempDir(); out:=filepath.Join(dir,"verify.log")
 _,err=(ExecVerifier{}).Verify(context.Background(),VerifyRequest{WorkDir:dir,Command:strings.Join(checks," && "),OutputPath:out})
 if err!=nil {t.Fatal("verification environment assertion failed")}
 for _,name:=range q.KeyVariables(){if _,ok:=os.LookupEnv(name); !ok {t.Fatal("probe parent lost fixture")}}
 t.Logf("Verification environment: judge variables=0; control present=true; names=%s",strings.Join(q.KeyVariables(),", "))
}
