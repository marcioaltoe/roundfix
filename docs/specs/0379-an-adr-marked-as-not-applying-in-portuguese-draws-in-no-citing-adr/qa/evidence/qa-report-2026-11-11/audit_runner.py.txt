def show(rev,path):
 p,out=run(['git','show',rev+':'+path]);assert p.returncode==0,out;return p.stdout

def audit_history():
 commits=subprocess.check_output(['git','rev-list','--reverse',BASE+'..'+HEAD],text=True).split()
 log='commit_range: '+BASE+'..'+HEAD+'\n';paths=set();okay=True

  declared=set(re.findall(r'^- (?:interface|creates|instruction): `([^`]+)`',tasktext,re.M))
  if '## Recorded paths' in tasktext:
   declared.update(re.findall(r'`([^`]+)`',tasktext.split('## Recorded paths',1)[1].split('\n## ',1)[0]))
  outside=changed-declared-{taskpath};okay=okay and not outside
  log+='Task '+task+' declared '+str(len(declared))+' paths; out-of-scope: '+repr(sorted(outside))+'\n'

  p,out=run([str(executable)]);log+=out;okay=okay and p.returncode==0
  governed=[line.split('\t')[0] for line in p.stdout.splitlines() if line.endswith('\ttrue')]

 log+='Authorization byte-identical at Delivery Base and audited head; operative grant predates Tasks.\n'
 excluded=[p for p in paths if p in ['Makefile','go.mod','go.sum','.roundfixrc.yml','skills-lock.json','CHANGELOG.md','internal/app/version.go','internal/speccheck/constraints.go','docs/references/skill-coverage.json'] or p.startswith(('.github/workflows/','dist/npm/','internal/baseline/','docs/agents/'))]
 okay=okay and not excluded;log+='Forbidden changed paths: '+repr(excluded)+'\n'
 for number in ['0093','0168','0389']:
  for path in ROOT.glob('docs/adr/'+number+'-*'):
   rel=str(path.relative_to(ROOT));same=show(BASE,rel)==show(HEAD,rel);okay=okay and same;log+=rel+' byte-identical: '+str(same)+'\n'

  def settlement(t):

 okay=okay and '**Amended by [ADR-0389]' in after and 'updated_at: 2026-11-11T00:00:00Z' in after and 'updated_at: 2026-11-09T12:00:00Z' in before

 p,out=run(['git','diff',BASE+'..'+HEAD,'--','CONTEXT.md',rel,'.agents/skills/write-techspec/references/techspec-template.md']);log+=out
