SP=ROOT/'docs/specs'/SPEC

ENV=os.environ.copy()
ENV.update(GOPROXY='off',GOSUMDB='off',GOMAXPROCS='2',GIT_CONFIG_NOSYSTEM='1')
for k in list(ENV):
 if k.endswith('API_KEY'):del ENV[k]
MOD=subprocess.check_output(['go','env','GOMODCACHE'],text=True).strip()

def run(args,cwd=ROOT,env=ENV):
 p=subprocess.run(args,cwd=cwd,env=env,text=True,capture_output=True)
 desc=shlex.join(args) if isinstance(args,list) else args
 out='cwd: '+str(cwd)+'\ncommand: '+desc+'\nstdout:\n'+p.stdout+'\nstderr:\n'+p.stderr+'\nexit: '+str(p.returncode)+'\n'
 return p,out

def save(name,text):
 (E/(name+'.txt')).write_text(text)

def settle(row,status,detail,files=()):
 t=REPORT.read_text();t=re.sub(r'(?m)^(\| '+re.escape(row)+r' \| )[^|]+(\|)',lambda m:m[1]+status+' '+m[2],t)
 anchor='\n### '+row+'\n';start=t.index(anchor);end=t.find('\n### ',start+len(anchor));end=len(t) if end<0 else end
 section=t[start:end]
 section+='\n'+detail+'\n'
 if files:section+='\nEvidence: '+', '.join('['+f+'](evidence/qa-report-2026-11-11/'+f+')' for f in files)+'.\n'
 t=t[:start]+section+t[end:];REPORT.write_text(t)
 print(row,status,detail,flush=True)

def tests(name,selection=None,cwd=ROOT):
 args=['go','test','-count=1','-timeout','10m','-v','./internal/speccheck']
 if selection:args+=['-run',selection]
 p,out=run(args,cwd);save(name,out);return p,out

def commit(cwd,msg):
 for args in [['git','add','--all'],['git','-c','user.name=QA Fixture','-c','user.email=qa@example.invalid','-c','core.hooksPath=/dev/null','commit','-qm',msg]]:
  p,out=run(args,cwd);assert p.returncode==0,out

def init(cwd):
 p,out=run(['git','init','-q'],cwd);assert p.returncode==0,out

def archive(rev,cwd):
 data=subprocess.check_output(['git','archive',rev],cwd=ROOT)
 with tarfile.open(fileobj=io.BytesIO(data)) as t:t.extractall(cwd,filter='data')
 init(cwd);commit(cwd,'Fresh QA archive '+rev)

@contextlib.contextmanager
def cli_env():
 with tempfile.TemporaryDirectory() as h,tempfile.TemporaryDirectory() as t:
  env=ENV.copy();env.update(HOME=h,TMPDIR=t,GOMODCACHE=MOD,GOCACHE=CACHE)
  yield env

def check_case(binary,cwd):
 with cli_env() as env:
  return run([str(binary),'spec','check','0001-adr-related-seeds','--strict'],cwd,env)

def write(cwd,path,text):
 p=cwd/path;p.parent.mkdir(parents=True,exist_ok=True);p.write_text(text)

  write(cwd,'docs/adr/'+n+'-fixture-decision.md','---\nstatus: accepted\n---\n\n# Fixture decision '+n+'\n\n'+citation+'\n')
 commit(cwd,'Fixture ADRs precede the Spec')

def phase_tests():
 settle('Q01','pass','make build exited 0. Daemon supplied make verify-changed pass (exit 0) at '+HEAD+'; not rerun. Fresh binary version recorded in frontmatter.',['build.txt','version.txt','precondition.txt'])
 p,out=tests('focused','^('+'|'.join(NAMES)+')$')
 okay=p.returncode==0 and all('--- PASS: '+n in out for n in NAMES)
 settle('Q02','pass' if okay else 'fail','All six named tests and every Portuguese/guard subtest must pass; observed focused exit '+str(p.returncode)+'.',['focused.txt'])
 p,out=tests('speccheck-package')
 settle('Q03','pass' if p.returncode==0 else 'fail','Whole internal/speccheck package, uncached verbose run: exit '+str(p.returncode)+'.',['speccheck-package.txt'])
 # Execute authored commands byte for byte, without generator flags.
 evidence=[];okay=True;summary=[]
 for task in ['task_01','task_02']:
  text=(SP/(task+'.md')).read_text().split('## Verification\n',1)[1].split('\n## ',1)[0]
  cmds=re.findall(r'^- `(.*)` — expected:',text,re.M)
  for i,cmd in enumerate(cmds,1):
   assert not re.search(r'(?<![\w-])(?:go test)[^\n]*-record',cmd)
   p,out=run(['sh','-c',cmd]);name=task+'-verification-'+str(i)+'.txt';(E/name).write_text(out);evidence.append(name)
   okay=okay and p.returncode==0;summary.append(task+' command '+str(i)+': exit '+str(p.returncode))
 settle('Q07','pass' if okay else 'fail','Authored non-QA Verification commands executed; '+'; '.join(summary)+'.',evidence)
