import os, json, subprocess, tempfile
from pathlib import Path
root=Path.cwd(); binary=root/'bin/roundfix'; evidence=root/'docs/specs/0230-retire-the-jev-router/qa/evidence/2026-10-05-retirement'
records=[]
rule='OpenAI and Anthropic models run only through the codex and claude subscriptions'
models=['roundfix-openrouter/typesafe/jev-router','openrouter/openai/gpt-6.1-sol','openrouter/anthropic/claude-opus-5.5','openrouter/openrouter/auto']
with tempfile.TemporaryDirectory(prefix='roundfix-0230-qa-',dir='/private/tmp') as tmp:
 home=Path(tmp)/'home'; repo=Path(tmp)/'repo'; home.mkdir();repo.mkdir();(home/'.roundfix').mkdir()
 env={'PATH':'/usr/bin:/bin:/opt/homebrew/bin','HOME':str(home),'GIT_CONFIG_GLOBAL':'/dev/null','GIT_CONFIG_NOSYSTEM':'1','LANG':'en_US.UTF-8'}
 subprocess.run(['git','init','-q',str(repo)],env=env,check=True)
 project=repo/'.roundfixrc.yml'; user=home/'.roundfix/config.yml'
 original='# preserve this config\nwatch:\n  max_rounds: 4\n'
 def call(label,args):
  p=subprocess.run([str(binary)]+args,cwd=repo,env=env,capture_output=True,text=True,timeout=20)
  records.append({'label':label,'command':['./bin/roundfix']+args,'cwd':str(repo),'home':str(home),'exit':p.returncode,'stdout':p.stdout,'stderr':p.stderr})
  (evidence/'cli-flows.json').write_text(json.dumps(records,indent=2)+'\n')
  return p
 def profile(model,runtime='opencode'):
  return f'profiles:\n  docs:\n    preferred: {{runtime: {runtime}, model: "{model}", reasoning_effort: high}}\n    fallbacks: [{{runtime: claude, model: opus, reasoning_effort: high}}]\n'
 for model in models:
  message='names the retired Jev Router' if model.startswith('roundfix-openrouter/') else 'can reach OpenAI or Anthropic models through OpenRouter'
  for scope in ['project','user']:
   project.write_text(original);user.write_text('')
   target=project if scope=='project' else user;target.write_text(profile(model))
   for repeat in range(2):
    p=call(f'{scope} refusal {model} fresh read {repeat+1}',['profiles','show'])
    assert p.returncode==2 and f'profiles.docs.preferred.model "{model}" {message}; {rule}' in p.stderr,(p.returncode,p.stderr)
  user.write_text('');project.write_text(original)
  fragment=repo/'profile.yml';fragment.write_text(profile(model))
  before=project.read_bytes()
  for repeat in range(2):
   p=call(f'configure refusal {model} repeat {repeat+1}',['profiles','configure','--scope','project','--file',str(fragment)])
   want=f'roundfix: profiles failed: read profiles file "{fragment}": profiles.docs.preferred.model "{model}" {message}; {rule}\nRun \'roundfix profiles --help\' for usage.\n'
   assert p.returncode==2 and p.stdout=='' and p.stderr==want,(p.returncode,p.stdout,p.stderr,want)
   assert project.read_bytes()==before
  confirmation=call(f'configure refusal preserved config {model}',['profiles','show'])
  assert confirmation.returncode==0,confirmation.stderr
 for runtime,model in [('opencode','openrouter/deepseek/deepseek-v4-pro'),('codex','gpt-6.1-sol')]:
  for scope in ['user','project']:
   project.write_text(original);user.write_text('');target=project if scope=='project' else user;target.write_text(profile(model,runtime))
   for repeat in range(2):
    p=call(f'{scope} allowed {model} fresh read {repeat+1}',['profiles','show'])
    assert p.returncode==0 and model in p.stdout,(p.returncode,p.stdout,p.stderr)
 warning='config: jev.router_min_credit_usd is deprecated and ignored; the Jev Router was retired, so remove it\n'
 for scope in ['user','project','both']:
  for value in ['0','20','-1','[]','{}','null']:
   project.write_text(original);user.write_text('')
   content='jev:\n  router_min_credit_usd: '+value+'\n'
   if scope in ['user','both']:user.write_text(content)
   if scope in ['project','both']:project.write_text(content)
   for repeat in range(2):
    p=call(f'deprecated {scope} {value} fresh read {repeat+1}',['profiles','show'])
    assert p.returncode==0 and p.stderr==warning,(p.returncode,p.stderr)
 print(f'{len(records)} public CLI observations passed; transcript stdout, stderr and exits matched; configure state preserved after repeat and fresh public read')
