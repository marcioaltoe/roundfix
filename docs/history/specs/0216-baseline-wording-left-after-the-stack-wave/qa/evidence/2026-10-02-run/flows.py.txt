def run(args,cwd=None,name=None):
 p=subprocess.run(args,cwd=cwd or root,env=env,capture_output=True,text=True)
 with log.open('a') as f:f.write('COMMAND '+repr(args)+'\nCWD '+str(cwd or root)+'\nEXIT '+str(p.returncode)+'\nSTDOUT\n'+p.stdout+'\nSTDERR\n'+p.stderr+'\n')
 if name:(scratch/(name+'.stdout')).write_text(p.stdout);(scratch/(name+'.stderr')).write_text(p.stderr)
 print(name or args[:2],p.returncode,flush=True)
 return p
