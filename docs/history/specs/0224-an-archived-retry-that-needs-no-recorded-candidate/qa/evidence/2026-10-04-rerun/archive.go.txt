func TestRunStartFindsAQueueStartedRunByItsRepository(t *testing.T) {
	for _, legacy := range []bool{false, true} {
		t.Run(fmt.Sprintf("legacy_repository_root=%v", legacy), func(t *testing.T) {
			home, repo := newImplementWorkspace(t, []implementSeed{{id: "task_01"}})
			workflow := newItemRecoveryWorkflowForRepository(t, home, repo)
			worktree, branch := linkedArchivedRetryItem(t, repo)
			run := createArchivedRetryRun(t, workflow, worktree, branch)
			if legacy {
				db, err := sql.Open("sqlite", store.DatabasePath(home))
				if err != nil {
					t.Fatal(err)
				}
				defer db.Close()
				if _, err := db.ExecContext(t.Context(), "UPDATE runs SET repository_root = '' WHERE id = ?", run.ID); err != nil {
					t.Fatal(err)
				}
			}
			got, err := workflow.RunStart(t.Context(), repo, run.ID)
			if err != nil || got != run.HeadSHA {
				t.Fatalf("start=%q err=%v, want %q for linked item worktree", got, err, run.HeadSHA)
			}
		})
	}
}

func TestRunStartRefusesARunOfAnotherRepository(t *testing.T) {
	home, repo := newImplementWorkspace(t, []implementSeed{{id: "task_01"}})
	workflow := newItemRecoveryWorkflowForRepository(t, home, repo)
	worktree, branch := linkedArchivedRetryItem(t, repo)
	run := createArchivedRetryRun(t, workflow, worktree, branch)
	otherRepo := t.TempDir()
	gittest.InitRepo(t, otherRepo, "--initial-branch=main")
	otherWorktree := filepath.Join(t.TempDir(), "other-item")
	gittest.Run(t, otherRepo, "-c", "user.name=Roundfix Test", "-c", "user.email=roundfix-test@example.com", "-c", "commit.gpgsign=false", "commit", "--allow-empty", "-m", "test: other repository")
	gittest.Run(t, otherRepo, "worktree", "add", "-b", "test/other-item", otherWorktree)
	otherRun := createArchivedRetryRun(t, workflow, otherWorktree, "test/other-item")
	got, err := workflow.RunStart(t.Context(), repo, otherRun.ID)
	want := fmt.Sprintf("Run %q has no Implement start head for repository %q", otherRun.ID, repo)
	if got != "" || err == nil || err.Error() != want {
		t.Fatalf("start=%q err=%v, want %q", got, err, want)
	}
	// A same-repository control proves that refusal is scoped to the repository,
	// rather than rejecting every Run started in a linked worktree.
	if got, err := workflow.RunStart(t.Context(), repo, run.ID); err != nil || got != run.HeadSHA {
		t.Fatalf("same-repository control: start=%q err=%v, want %q", got, err, run.HeadSHA)
	}
}

func testArchivedRetryOfAQueueStartedRunReturnsToReview(t *testing.T, blocker string) {
	t.Helper()
	home, repo := newImplementWorkspace(t, []implementSeed{
		{id: "task_01", status: string(spec.StatusCompleted)},
		implementQAGateSeed(string(spec.StatusFailed), "task_01"),
	})
	workflow := newItemRecoveryWorkflowForRepository(t, home, repo)
	worktree, branch := linkedArchivedRetryItem(t, repo)
	run := createArchivedRetryRun(t, workflow, worktree, branch)
	queue, err := workflow.store.CreateDeliveryQueue(t.Context(), repo, []string{implementTestSlug})
	if err != nil {
		t.Fatal(err)
	}
	if _, _, _, err := workflow.store.RecordDeliveryQueueItemWorktree(t.Context(), repo, implementTestSlug, branch, worktree); err != nil {
		t.Fatal(err)
	}
	if err := workflow.store.SetDeliveryQueueItemWorktreeProvisioned(t.Context(), repo, implementTestSlug, true); err != nil {
		t.Fatal(err)
	}
	item := queue.Items[0]
	item.Stage = store.DeliveryStageParked
	item.Blocker = blocker
	item.RunID = run.ID
	item.Branch = branch
	item.Worktree = worktree
	item.WorktreeProvisioned = true
	if err := workflow.store.UpdateDeliveryQueueItem(t.Context(), repo, item); err != nil {
		t.Fatal(err)
	}

	writeArchiveQAReport(t, worktree, spec.VerdictPartial, "rows_blocked_finding: 0", "rows_blocked_environment: 1")
	mustWrite(t, filepath.Join(worktree, "operator-evidence.md"), "The operator satisfied the environment-blocked row.\n")
	gittest.Run(t, worktree, "add", "-A")
	gittest.Run(t, worktree, "commit", "-m", "docs: operator QA evidence")
	setCommandEnvironmentForTest(t, home, worktree)
	var stdout, stderr bytes.Buffer
	if code := qaBuiltCLI(t, []string{"archive", implementTestSlug, "--qa-override", "--approval", "test maintainer request", "--reason", "retain the environment partial and operator evidence"}, &stdout, &stderr); code != exitOK {
		t.Fatalf("archive exit=%d stdout=%q stderr=%q", code, stdout.String(), stderr.String())
	}
	gittest.Run(t, worktree, "add", "-A")
	gittest.Run(t, worktree, "commit", "-m", "docs: archive with QA override")
	head := itemRecoveryHead(t, worktree)
	state, err := workflow.InspectItem(t.Context(), worktree, implementTestSlug)
	if err != nil || !state.Archived || !state.QAOverride || head == run.HeadSHA {
		t.Fatalf("archive state=%+v err=%v start=%s", state, err, run.HeadSHA)
	}
	flow := &parkTestDeliveryFlow{}
	engine := delivery.NewEngine(workflow.store, delivery.EngineDependencies{
		Workspace: workflow, Recovery: workflow, History: workflow,
		Revalidator: flow, PullRequests: flow,
	})
	result, err := engine.Retry(t.Context(), repo, implementTestSlug)
	if err != nil {
		t.Fatalf("retry operator archive of queue-started Run: %v", err)
	}
	queue, found, err := workflow.store.DeliveryQueue(t.Context(), repo)
	if err != nil || !found {
		t.Fatalf("read retried queue: found=%v err=%v", found, err)
	}
	got := queue.Items[0]
	if result.Stage != store.DeliveryStageReviewing || got.Stage != store.DeliveryStageReviewing || !slices.Equal(got.CandidateCommits, []string{head}) || got.Blocker != "" {
		t.Fatalf("result=%+v item=%+v, want reviewing with newest candidate %s", result, got, head)
	}
	if current := itemRecoveryHead(t, worktree); current != head {
		t.Fatalf("retry changed item head: %s, want %s", current, head)
	}
}

func linkedArchivedRetryItem(t *testing.T, repo string) (string, string) {
	t.Helper()
	path := filepath.Join(t.TempDir(), "item")
	branch := "test/archived-retry-item"
	gittest.Run(t, repo, "worktree", "add", "-b", branch, path)
	return path, branch
}

func createArchivedRetryRun(t *testing.T, workflow *commandDeliveryWorkflow, worktree, branch string) store.Run {
	t.Helper()
	run, err := workflow.store.CreateRun(t.Context(), store.CreateRunRequest{
		Kind: store.KindImplement, GitRoot: worktree, LocalBranch: branch,
		HeadSHA: itemRecoveryHead(t, worktree), SpecSlug: implementTestSlug,
		Agent: "codex", ArtifactDir: t.TempDir(), WorkDir: worktree,
	})
	if err != nil {
		t.Fatal(err)
	}
	return run
}
