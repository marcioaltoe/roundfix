func qaBuiltCLI(t *testing.T,args []string,stdout,stderr *bytes.Buffer) int {
 t.Helper(); o:=commandEnvironmentOverridesForTest(t)
 cmd:=exec.CommandContext(t.Context(),os.Getenv("QA_BINARY"),args...)
 cmd.Dir=o.workDir;cmd.Env=append(os.Environ(),"HOME="+o.homeDir);cmd.Stdout=stdout;cmd.Stderr=stderr
 err:=cmd.Run();if err==nil {t.Logf("built archive stdout=%q stderr=%q",stdout.String(),stderr.String());return 0}; if ex,ok:=err.(*exec.ExitError);ok{return ex.ExitCode()};t.Fatal(fmt.Errorf("built CLI: %w",err));return -1
}
