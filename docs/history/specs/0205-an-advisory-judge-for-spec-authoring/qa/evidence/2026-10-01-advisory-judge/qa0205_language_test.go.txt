package judge

import (
 "os"
 "path/filepath"
 "strings"
 "testing"
 "unicode"
)

func TestQA0205ArchiveLanguage(t *testing.T) {
 q,err:=Load(); if err!=nil {t.Fatal(err)}
 root,err:=filepath.Abs("../..");if err!=nil {t.Fatal(err)}
 total:=0; low:=1.0; lowPath:=""
 err=filepath.WalkDir(filepath.Join(root,"docs/history/specs"),func(path string,d os.DirEntry,e error)error {
  if e!=nil{return e};if d.IsDir() || (d.Name()!="_prd.md" && d.Name()!="_techspec.md"){return nil}; checkLanguage(t,q,path,&total,&low,&lowPath);return nil
 });if err!=nil{t.Fatal(err)}
 files,err:=filepath.Glob(filepath.Join(root,"docs/adr/*.md"));if err!=nil{t.Fatal(err)}
 adrs:=0
 for _,path:=range files {
  name:=filepath.Base(path);if len(name)<5 || name[4]!='-' {continue}
  _,accepted,e:=readADR(root,name[:4]);if e!=nil{t.Fatal(e)};if !accepted{continue}
  checkLanguage(t,q,path,&total,&low,&lowPath);adrs++
 }
 t.Logf("English artifacts: %d; accepted ADRs: %d; lowest English share: %.9f at %s",total,adrs,low,lowPath)
}
func checkLanguage(t *testing.T,q Questions,path string,total *int,low *float64,lowPath *string) {
 t.Helper();b,e:=os.ReadFile(path);if e!=nil{t.Fatal(e)}
 passed:=q.Language.isEnglish(string(b))
 _,body,_,e:=splitFrontMatter(string(b));if e!=nil{t.Fatal(e)}
 words:=strings.FieldsFunc(strings.ToLower(body),func(r rune)bool{return !unicode.IsLetter(r)})
 en,pt:=0,0;for _,w:=range words{for _,x:=range q.Language.EnglishWords{if w==x{en++;break}}}
 for _,w:=range words{for _,x:=range q.Language.PortugueseWords{if w==x{pt++;break}}}
 share:=float64(en)/float64(len(words));if !passed{t.Errorf("not English: %s; English=%d/%d (%.9f); Portuguese=%d/%d (%.9f)",path,en,len(words),share,pt,len(words),float64(pt)/float64(len(words)))};if share<*low{*low=share;*lowPath=path};*total++
}
