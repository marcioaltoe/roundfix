import os, json, tempfile, subprocess, pathlib, hashlib, shutil
root=pathlib.Path.cwd(); evidence=root/'docs/specs/0236-baseline-update-with-a-repository-profile/qa/evidence/2026-10-06-profile'
work=pathlib.Path(tempfile.mkdtemp(prefix='roundfix-qa0236-',dir='/private/tmp')); repo=work/'repo'; repo.mkdir(); home=work/'home'; home.mkdir()
env=os.environ.copy(); env.update(HOME=str(home),XDG_CONFIG_HOME=str(home/'config'),XDG_CACHE_HOME=str(home/'cache'),XDG_DATA_HOME=str(home/'data'),GIT_CONFIG_GLOBAL='/dev/null',GIT_CONFIG_NOSYSTEM='1')
(repo/'bin').mkdir(); (repo/'bin/roundfix').symlink_to(root/'bin/roundfix')
log=[]
def run(name,args,cwd=repo):
 p=subprocess.run(args,cwd=cwd,env=env,text=True,capture_output=True,timeout=60)
 log.append(dict(name=name,command=args,cwd=str(cwd),exit=p.returncode,stdout=p.stdout,stderr=p.stderr))
 (evidence/'flows.json').write_text(json.dumps(dict(scratch=str(work),home=str(home),steps=log),indent=2))
 print(name,p.returncode,p.stdout[:180].replace('\n',' | '),p.stderr[:120],flush=True)
 return p
run('git-init',['git','init','-q']); run('git-identity',['git','config','user.email','qa@example.invalid']); run('git-name',['git','config','user.name','QA'])
(repo/'.agents/skills/context7').mkdir(parents=True); (repo/'.agents/skills/context7/SKILL.md').write_text('# context7\n'); (repo/'.agents/skills/exa-web-search').mkdir(parents=True); (repo/'.agents/skills/exa-web-search/SKILL.md').write_text('# exa\n'); (repo/'Makefile').write_text('verify:\n\t@true\nverify-incremental:\n\t@true\n')
(repo/'README.md').write_text('Disposable QA repository.\n'); run('seed-add',['git','add','README.md','Makefile','.agents']); run('seed-commit',['git','commit','-qm','QA seed'])
assert run('profile-init',['./bin/roundfix','baseline','profile','init','--id','scratch-go','--from','go-cli-tui']).returncode==0
run('profile-add',['git','add','.roundfix']); run('profile-commit',['git','commit','-qm','QA profile'])
decisions=['preservation.mode=greenfield','language.generated=English','verification.gate=make verify','verification.incremental=make verify-incremental','branch.prefix=feat/','spec.scaffold=true','domain.layout=single-context','triage.external=false','autonomous.enabled=true','runtime.backend=codex gpt-5.5 xhigh','runtime.design=claude opus xhigh','secondbrain.enabled=false','repository.extension.enabled=false']
args=['./bin/roundfix','baseline','plan','--repo',str(repo),'--profile','scratch-go','--format','json']
for d in decisions: args+=['--decision',d]
p=run('plan',args); assert p.returncode==0; plan=json.loads(p.stdout); planpath=work/'plan.json'; planpath.write_text(p.stdout)
print('plan keys',list(plan),flush=True)
p=run('apply',['./bin/roundfix','baseline','apply','--repo',str(repo),'--plan',str(planpath),'--confirm-plan',plan['planDigest'],'--format','json']); assert p.returncode==0
run('adopt-add',['git','add','.']); run('adopt-commit',['git','commit','-qm','QA adopt'])
p=run('transcript1',['./bin/roundfix','baseline','update','--repo',str(repo),'--format','text']); assert p.returncode==0 and p.stderr=='' and p.stdout.startswith('Baseline update: current\nResult: the repository already matches the current Baseline catalog\n')
p=run('current-json',['./bin/roundfix','baseline','update','--repo',str(repo),'--format','json']); assert p.returncode==0 and json.loads(p.stdout)['state']=='current'
p=run('transcript2',['./bin/roundfix','baseline','skills','restore','--profile','missing-profile','--repo',str(repo),'--format','text']); assert p.returncode==2 and 'restore.profile-unresolved:' in p.stdout
p=run('transcript3',['./bin/roundfix','baseline','skills','restore','--profile','scratch-go','--skill','context7-cli','--source-dir',str(work/'missing'),'--repo',str(repo),'--format','text']); assert p.returncode==2 and 'restore.source-dir-invalid:' in p.stdout
p=run('restore-json',['./bin/roundfix','baseline','skills','restore','--profile','scratch-go','--skill','context7-cli','--source-dir',str(work/'missing'),'--repo',str(repo),'--format','json']); assert p.returncode==2
p=run('reconcile-json',['./bin/roundfix','baseline','skills','reconcile','--profile','scratch-go','--source','example/skills','--revision','a'*40,'--source-dir',str(work/'missing'),'--repo',str(repo),'--format','json']); assert p.returncode==2
shutil.copytree(root/'.agents/skills',repo/'.agents/skills',dirs_exist_ok=True); shutil.copy2(root/'skills-lock.json',repo/'skills-lock.json')
# Force owned refresh while preserving locked external installations.
shutil.rmtree(repo/'.agents/skills/roundfix')
p=run('refresh',['./bin/roundfix','baseline','update','--repo',str(repo),'--yes','--format','json']); assert p.returncode==0 and 'Unknown built-in' not in p.stdout+p.stderr
p=run('refresh-again',['./bin/roundfix','baseline','update','--repo',str(repo),'--format','json']); assert p.returncode==0 and json.loads(p.stdout)['state']=='current'
env['PATH']='/usr/bin:/bin'
p=run('doctor-offline',['./bin/roundfix','doctor']); assert 'Unknown built-in Baseline Profile' not in p.stdout+p.stderr and 'snapshot comparison unavailable' not in p.stdout+p.stderr
print('ALL FLOW ASSERTIONS PASSED',flush=True)
