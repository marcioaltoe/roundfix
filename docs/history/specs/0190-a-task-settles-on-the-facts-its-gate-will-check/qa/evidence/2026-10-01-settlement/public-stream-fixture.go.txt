package cli
import (
 "bytes"
 "context"
 "encoding/json"
 "os"
 "os/exec"
 "path/filepath"
 "strings"
 "testing"
 "roundfix/internal/spec"
)
func TestQAGatePublicStreamFixture(t *testing.T) {
 home, repo := newImplementWorkspace(t, []implementSeed{{id:"task_01",title:"Build the widget core"},implementQAGateSeed("","task_01")})
 writeUserConfig(t,home,"defaults:\n  verification: repository check\n")
 runner:=&implementFakeRunner{gitRoot:repo,statusByTask:map[string]spec.Status{"task_01":spec.StatusCompleted},qaReport:implementQAReport("pass")}
 withImplementCollaborators(t,runner)
 var out,diag bytes.Buffer
 code:=runCLIContext(t,context.Background(),[]string{"implement","--spec",implementTestSlug,"--no-input"},&out,&diag)
 if code!=exitOK {t.Fatalf("fixture implement exit %d: %s %s",code,out.String(),diag.String())}
 runID,events:=journaledRunEvents(t,home,diag.String())
 evidence:=os.Getenv("QA_0190_EVIDENCE")
 raw,_:=json.MarshalIndent(events,"","  ");if err:=os.WriteFile(filepath.Join(evidence,"stream-raw-journal.json"),raw,0644);err!=nil{t.Fatal(err)}
 binary:=os.Getenv("QA_0190_BINARY")
 var first []byte
 for index,filter:=range []string{"verification","verification","all"} {
  args:=[]string{"events",runID}; if filter!="all" {args=append(args,"--filter",filter)}
  command:=exec.Command(binary,args...);command.Dir=repo;command.Env=withEnvValue(os.Environ(),"HOME",home)
  var stderr bytes.Buffer;command.Stderr=&stderr
  data,err:=command.Output();if err!=nil {t.Fatalf("public replay: %v; %s",err,stderr.String())}
  if stderr.Len()!=0 {t.Fatalf("public replay stderr: %s",stderr.String())}
  if err:=os.WriteFile(filepath.Join(evidence,[]string{"stream-first.jsonl","stream-restart.jsonl","stream-all.jsonl"}[index]),data,0644);err!=nil{t.Fatal(err)}
  if index==0 {first=data};if index==1&&!bytes.Equal(first,data){t.Fatal("fresh-process replay changed persisted results")}
  phases:=map[string]map[string]bool{}
  for _,line:=range strings.Split(strings.TrimSpace(string(data)),"\n") {
   var record map[string]interface{};if err:=json.Unmarshal([]byte(line),&record);err!=nil{t.Fatal(err)}
   if record["category"]!="verification"||record["work_item"]!="task_01"{continue}
   if label,ok:=record["command"].(string);ok {if phases[label]==nil {phases[label]=map[string]bool{}};phase,_:=record["phase"].(string);phases[label][phase]=true}
  }
  for _,label:=range []string{"settlement check: spec consistency","settlement check: authorization"} {if !phases[label]["started"]||!phases[label]["command-passed"]{t.Fatalf("missing settlement label phases: %s: %v; data=%s",label,phases,data)}}
  if bytes.Contains(data,[]byte("repository check")){t.Fatal("repository command leaked into public stream")}
 }
 foundRepository:=false
 for _,entry:=range events {var payload map[string]interface{};json.Unmarshal(entry.Event.Payload,&payload);if payload["command"]=="repository check"&&payload["task"]=="task_01" {foundRepository=true}}
 if !foundRepository {t.Fatal("engine did not journal repository verification")}
 os.WriteFile(filepath.Join(evidence,"stream-commands.txt"),[]byte("Disposable gated graph; existing implementFakeRunner and fake collaborators drive actual Task engine.\nPublic binary: "+binary+"\nRun: "+runID+"\n./bin/roundfix events "+runID+" --filter verification (twice in fresh processes)\n./bin/roundfix events "+runID+"\nAll exits 0, stderr empty. Settlement labels started/command-passed; repository command is present in raw journal and intentionally redacted publicly.\nParity: fixture setup uses the in-process CLI with fake runner/verifier/committer/worktree/source; public stream uses the unmodified built binary.\n"),0644)
}
