package cli

import (
	"context"
	"encoding/json"
	"os"
	"roundfix/internal/preflight"
	"strings"
	"testing"
)

func TestQA0203HistoricalLedgerReplay(t *testing.T) {
	ledger := os.Getenv("QA0203_LEDGER")
	body, err := os.ReadFile(ledger)
	if err != nil {
		t.Fatal(err)
	}
	root := os.Getenv("QA0203_REPO")
	git := preflight.ExecGitRunner{}
	ctx := context.Background()
	roots, err := reviewCandidateSpecRoots("docs/specs", root)
	if err != nil {
		t.Fatal(err)
	}
	total, anchored, regions, fixedRegions := 0, 0, 0, 0
	for _, line := range strings.Split(strings.TrimSpace(string(body)), "\n") {
		var d struct {
			Head        string `json:"headCommit"`
			Text        string `json:"text"`
			Disposition string `json:"disposition"`
			At          string `json:"recordedAt"`
			Finding     string `json:"finding"`
		}
		if err := json.Unmarshal([]byte(line), &d); err != nil {
			t.Fatal(err)
		}
		// First 34 are the authoring corpus. Later append-only ledger entries are reported, not silently substituted.
		if d.At > "2026-09-30T18:19:29Z" {
			t.Logf("EXCLUDED later entry %s %s %s", d.At, d.Head, d.Finding)
			continue
		}
		total++
		if _, err := git.RunGit(ctx, root, "rev-parse", "--verify", d.Head+"^{commit}"); err != nil {
			t.Fatal(err)
		}
		base, err := git.RunGit(ctx, root, "merge-base", "origin/main", d.Head)
		if err != nil {
			t.Fatal(err)
		}
		base = strings.TrimSpace(base)
		diff, err := git.RunGit(ctx, root, "diff", "--no-ext-diff", "--no-textconv", "--no-color", base, d.Head, "--")
		if err != nil {
			t.Fatal(err)
		}
		a, ok := parseReviewFindingAnchor(d.Text)
		held := ok && indexReviewDiff(diff).holds(a)
		var rules []string
		if ok {
			rules, err = conventionRegions(ctx, reviewRepository{Root: root, Head: d.Head, Base: base, SpecRoots: roots, Git: git}, a)
			if err != nil {
				t.Fatal(err)
			}
		}
		if held {
			anchored++
		}
		if len(rules) > 0 {
			regions++
			if d.Disposition == "fixed" {
				fixedRegions++
			}
		}
		t.Logf("ENTRY %02d head=%s base=%s disposition=%s anchor=%+v held=%v regions=%v text=%q", total, d.Head, base, d.Disposition, a, held, rules, d.Text)
	}
	t.Logf("SUMMARY original corpus=%d anchored=%d inRegions=%d fixedInRegions=%d", total, anchored, regions, fixedRegions)
	if total != 34 || anchored != 34 || regions != 5 || fixedRegions != 0 {
		t.Fatalf("corpus differs from authored acceptance: total=%d anchored=%d regions=%d fixedRegions=%d", total, anchored, regions, fixedRegions)
	}
}
